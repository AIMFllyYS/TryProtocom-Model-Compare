  wheelRadius: 0.34,

const COG_R = (SPEC.cogTeeth * SPEC.toothPitch) / (2 * Math.PI);
const BB = { x: 0, y: 0.30 };

function ik2(hip, target, l1, l2) {
  let dx = target.x - hip.x;
  let dy = target.y - hip.y;
  let dist = Math.hypot(dx, dy);
  const max = l1 + l2 - 1e-5;
  const min = Math.abs(l1 - l2) + 1e-5;
  if (dist > max && dist > 0) {
    const s = max / dist;
    dx *= s;
    dy *= s;
    dist = max;
  } else if (dist < min) {
    const s = min / (dist || 1);
    dx *= s;
    dy *= s;
    dist = min;
  }
  const ang = Math.atan2(dy, dx);
  const cosA = (l1 * l1 + dist * dist - l2 * l2) / (2 * l1 * dist);
  const bend = Math.acos(Math.min(1, Math.max(-1, cosA)));
  const a1 = ang + bend;
  const a2 = ang - bend;
  const k1x = hip.x + Math.cos(a1) * l1;
  const k2x = hip.x + Math.cos(a2) * l1;
  const ka = k1x >= k2x ? a1 : a2;
  return {
    knee: { x: hip.x + Math.cos(ka) * l1, y: hip.y + Math.sin(ka) * l1 },
    ankle: { x: hip.x + dx, y: hip.y + dy }
  };
}

function solveLeg(phi, side) {
  const pedal = {
    x: BB.x + Math.cos(phi) * SPEC.crankLength,
    y: BB.y + Math.sin(phi) * SPEC.crankLength,
    z: side * PEDAL_Z
  };
  const hip = { x: HIP.x, y: HIP.y, z: side * HIP_Z };
  const lean = SPEC.tarsoLean;
  const ankle = {
    x: pedal.x - Math.sin(lean) * SPEC.tarso,
    y: pedal.y + Math.cos(lean) * SPEC.tarso,
    z: 0
  };
  const solved = ik2(hip, ankle, SPEC.femur, SPEC.tibio);
  const total = SPEC.femur + SPEC.tibio + SPEC.tarso;
  const knee = {
    x: solved.knee.x,
    y: solved.knee.y,
    z: hip.z + (pedal.z - hip.z) * (SPEC.femur / total)
  };
  ankle.z = hip.z + (pedal.z - hip.z) * ((SPEC.femur + SPEC.tibio) / total);
  const foot = { x: pedal.x, y: pedal.y, z: pedal.z };
  return {
    hip, knee, ankle, foot, pedal,
    pedalAngle: 0,
    footDist: Math.hypot(foot.x - pedal.x, foot.y - pedal.y, foot.z - pedal.z)
  };
}

function externalNormals(A, rA, B, rB) {
  const dx = B.x - A.x;
  const dy = B.y - A.y;
  const d = Math.hypot(dx, dy);
  const phi = Math.atan2(dy, dx);
  const alpha = Math.acos(Math.max(-1, Math.min(1, (rA - rB) / d)));
  return [1, -1].map((s) => {
    const th = phi + s * alpha;
    return { x: Math.cos(th), y: Math.sin(th) };
  });
}

function buildChainLoop() {
  const ns = externalNormals(BB, CHAINRING_R, REAR, COG_R).map((n) => ({
    n,
    a: Math.atan2(n.y, n.x),
    p1: { x: BB.x + n.x * CHAINRING_R, y: BB.y + n.y * CHAINRING_R },
    p2: { x: REAR.x + n.x * COG_R, y: REAR.y + n.y * COG_R }
  }));
  ns.sort((a, b) => b.p1.y - a.p1.y);
  const up = ns[0];
  const lo = ns[1];
  const pts = [];
  const cw = (a0, a1) => {
    let delta = a0 - a1;
    while (delta < 0) delta += Math.PI * 2;
    while (delta >= Math.PI * 2) delta -= Math.PI * 2;
    return delta;
  };
  const pushLine = (p, q, n) => {
    for (let i = 0; i < n; i++) {
      const t = i / n;
      pts.push({ x: p.x + (q.x - p.x) * t, y: p.y + (q.y - p.y) * t });
    }
  };
  const pushArc = (c, r, a0, a1, n) => {
    const delta = cw(a0, a1);
    for (let i = 0; i < n; i++) {
      const a = a0 - delta * (i / n);
      pts.push({ x: c.x + Math.cos(a) * r, y: c.y + Math.sin(a) * r });
    }
  };
  pushLine(up.p2, up.p1, 16);
  pushArc(BB, CHAINRING_R, up.a, lo.a, 18);
  pushLine(lo.p1, lo.p2, 16);
  pushArc(REAR, COG_R, lo.a, up.a, 12);
  const cum = [0];
  for (let i = 1; i < pts.length; i++) {
    cum.push(cum[i - 1] + Math.hypot(pts[i].x - pts[i - 1].x, pts[i].y - pts[i - 1].y));
  }
  const close = Math.hypot(pts[0].x - pts[pts.length - 1].x, pts[0].y - pts[pts.length - 1].y);
  return { pts, cum, close, length: cum[cum.length - 1] + close };
}

const chainLoop = buildChainLoop();

function pointOnChain(s) {
  const L = chainLoop.length;
  s = ((s % L) + L) % L;
  const pts = chainLoop.pts;
  const cum = chainLoop.cum;
  let i = 1;
  while (i < pts.length && cum[i] < s) i++;
  if (i >= pts.length) {
    const t = (s - cum[cum.length - 1]) / (chainLoop.close || 1);
    const a = pts[pts.length - 1];
    const b = pts[0];
    return { x: a.x + (b.x - a.x) * t, y: a.y + (b.y - a.y) * t };
  }
  const span = cum[i] - cum[i - 1] || 1;
  const t = (s - cum[i - 1]) / span;
  const a = pts[i - 1];
  const b = pts[i];
  return { x: a.x + (b.x - a.x) * t, y: a.y + (b.y - a.y) * t };
}

function aimBone(mesh, ax, ay, az, bx, by, bz) {
  aimDir.set(bx - ax, by - ay, bz - az);
  const len = aimDir.length();
  mesh.position.set((ax + bx) * 0.5, (ay + by) * 0.5, (az + bz) * 0.5);
  if (len < 1e-5) return;

  mesh.quaternion.setFromUnitVectors(Y_AXIS, aimDir);
  mesh.scale.set(mesh.userData.sx || 1, len, mesh.userData.sz || 1);
}

function ensureAudio() {
  if (audioCtx) {

  if (!AC) return;

  master.gain.value = wantAudio ? 0.9 : 0;
  master.connect(audioCtx.destination);
  const seconds = 2;
  const buffer = audioCtx.createBuffer(1, seconds * audioCtx.sampleRate, audioCtx.sampleRate);
  const data = buffer.getChannelData(0);
  let brown = 0;
  for (let i = 0; i < data.length; i++) {
    const white = Math.random() * 2 - 1;
    brown = brown * 0.98 + white * 0.02;
    data[i] = brown * 3.2;
  }
  const src = audioCtx.createBufferSource();
  src.buffer = buffer;
  src.loop = true;
  const hp = audioCtx.createBiquadFilter();
  hp.type = 'highpass';
  hp.frequency.value = 80;
  const lp = audioCtx.createBiquadFilter();
  lp.type = 'lowpass';
  lp.frequency.value = 420;
  windGain = audioCtx.createGain();
  windGain.gain.value = 0;
  src.connect(hp);
  hp.connect(lp);
  lp.connect(windGain);
  windGain.connect(master);
  src.start();
  if (audioCtx.state === 'suspended') audioCtx.resume();
}

function playClick(time) {
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  const filter = audioCtx.createBiquadFilter();
  filter.type = 'bandpass';
  filter.frequency.value = 1400 + Math.random() * 800;
  filter.Q.value = 4;
  osc.type = 'square';
  osc.frequency.setValueAtTime(900 + Math.random() * 500, time);
  osc.frequency.exponentialRampToValueAtTime(220, time + 0.02);
  gain.gain.setValueAtTime(0.045, time);
  gain.gain.exponentialRampToValueAtTime(0.0001, time + 0.028);
  osc.connect(filter);
  filter.connect(gain);
  gain.connect(master);
  osc.start(time);
  osc.stop(time + 0.03);
}

function playCall(time) {
  const gain = audioCtx.createGain();
  gain.connect(master);

  gain.gain.exponentialRampToValueAtTime(0.12, time + 0.04);

}

function updateAudio(dt) {
  if (!audioCtx || !wantAudio) {
    if (windGain) windGain.gain.setTargetAtTime(0, audioCtx.currentTime, 0.05);
    return;
  }
  const now = audioCtx.currentTime;
  const wind = state.speed <= 0.01 ? 0 : Math.min(0.2, (state.speed / SPEC.maxSpeed) * 0.16);
  windGain.gain.setTargetAtTime(wind, now, 0.08);
  if (state.speed > 0.02) {
    const hz = (state.speed / SPEC.wheelRadius) * SPEC.cogTeeth / (2 * Math.PI);
    clickAcc += hz * dt;
    let n = 0;
    while (clickAcc >= 1 && n < 5) {
      playClick(now + n * (1 / hz));
      clickAcc -= 1;
      n++;
    }
    if (clickAcc > 2) clickAcc = 0;
    if (state.rideTime >= nextCall) {
      playCall(now);
      nextCall = state.rideTime + 7.5;
    }
  }
}
