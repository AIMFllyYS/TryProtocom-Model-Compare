### 建模与物理实现思路

本实现选取美洲白鹈鹕（*Pelecanus erythrorhynchos*）。物理上严格遵循纯滚动条件 $\omega_{wheel} = v / R_{wheel}$ 与齿轮传动比约束 $\omega_{crank} = \omega_{wheel} \cdot (N_{cog}/N_{chainring})$；踏板采用重力自平衡保持水平；鸟类后肢按解剖学构建两段式运动链，通过解析逆运动学（IK）驱动后弯跗间关节，脚蹼牢固贴合踏板。2.5D（分层SVG）与3D（Three.js程序化网格+软阴影）共享动力学时钟，实现毫秒级平滑无缝切换与 Web Audio 实时声效合成。

  import * as THREE from 'three';
  import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

  /* ==========================================================================
     1. 物理常数与全局状态引用
     ========================================================================== */
  const WHEEL_RADIUS = 0.34;      // 车轮半径 (米)
  const CHAINRING_TEETH = 48;    // 牙盘齿数
  const COG_TEETH = 16;          // 飞轮齿数
  const GEAR_RATIO = CHAINRING_TEETH / COG_TEETH; // 3.0

  const CRANK_LENGTH_PX = 36;    // 2.5D SVG 曲柄长度 (像素)

  const LEG_L1_M = 0.23;         // 大腿骨+胫跗骨 (米)
  const LEG_L2_M = 0.21;         // 跗跖骨 (米)
  const LEG_L1_PX = 76;          // SVG 大腿骨像素
  const LEG_L2_PX = 72;          // SVG 跗跖骨像素

  const state = window.__state;
  state.lastTime = performance.now();
  state.audioEnabled = false;

  /* ==========================================================================
     2. 逆运动学 (2-Bone Analytical IK)
     ========================================================================== */
  /**
   * 2D 鸟类后肢逆运动学 (膝/踝关节后曲)
   */
  function solve2DBirdLegIK(hx, hy, px, py, l1, l2) {
    const dx = px - hx;
    const dy = py - hy;
    let dist = Math.hypot(dx, dy);
    dist = Math.max(Math.abs(l1 - l2) + 0.001, Math.min(l1 + l2 - 0.001, dist));

    const cosAlpha = (l1 * l1 + dist * dist - l2 * l2) / (2 * l1 * dist);
    const alpha = Math.acos(Math.max(-1, Math.min(1, cosAlpha)));

    const ux = dx / dist;
    const uy = dy / dist;

    // 鸟类踝关节向后屈曲 (SVG 中车头向右，向后即 X 偏小方向，垂直向量选 -uy, ux)
    const nx = -uy;
    const ny = ux;

    const kx = hx + l1 * Math.cos(alpha) * ux + l1 * Math.sin(alpha) * nx;
    const ky = hy + l1 * Math.cos(alpha) * uy + l1 * Math.sin(alpha) * ny;
    return { kx, ky };
  }

  /**
   * 3D 鸟类后肢逆运动学
   */
  function solve3DBirdLegIK(H, P, l1, l2, outJoint) {
    const D = new THREE.Vector3().subVectors(P, H);
    let dist = D.length();
    dist = Math.max(Math.abs(l1 - l2) + 0.001, Math.min(l1 + l2 - 0.001, dist));
    const u = D.clone().normalize();

    // 后曲参考基准向量 (沿 -X 延伸)
    const bendPref = new THREE.Vector3(-1, 0, 0);
    const proj = u.clone().multiplyScalar(bendPref.dot(u));
    const n = new THREE.Vector3().subVectors(bendPref, proj);
    if (n.lengthSq() < 0.0001) {
      n.set(0, 0, 1).cross(u);
    }
    n.normalize();

    const cosAlpha = (l1 * l1 + dist * dist - l2 * l2) / (2 * l1 * dist);
    const alpha = Math.acos(Math.max(-1, Math.min(1, cosAlpha)));

    outJoint.copy(H)
      .addScaledVector(u, l1 * Math.cos(alpha))
      .addScaledVector(n, l1 * Math.sin(alpha));
  }

  /* ==========================================================================
     3. Web Audio API 程序化实时声效引擎
     ========================================================================== */
  class BikeAudioEngine {
    constructor() {
      this.ctx = null;
      this.masterGain = null;
      this.windGain = null;
      this.windFilter = null;
      this.noiseNode = null;
      this.lastClickTooth = -1;
      this.pelicanTimer = null;
    }

    init() {
      if (this.ctx) return;
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (!AudioCtx) return;
      this.ctx = new AudioCtx();

      this.masterGain = this.ctx.createGain();
      this.masterGain.gain.setValueAtTime(0, this.ctx.currentTime);
      this.masterGain.connect(this.ctx.destination);

      // 1. 白噪声 (模拟路噪与迎面风噪)
      const bufferSize = this.ctx.sampleRate * 2;
      const noiseBuffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
      const output = noiseBuffer.getChannelData(0);
      for (let i = 0; i < bufferSize; i++) {
        output[i] = Math.random() * 2 - 1;
      }
      this.noiseNode = this.ctx.createBufferSource();
      this.noiseNode.buffer = noiseBuffer;
      this.noiseNode.loop = true;

      this.windFilter = this.ctx.createBiquadFilter();
      this.windFilter.type = 'bandpass';
      this.windFilter.frequency.setValueAtTime(400, this.ctx.currentTime);
      this.windFilter.Q.setValueAtTime(1.2, this.ctx.currentTime);

      this.windGain = this.ctx.createGain();
      this.windGain.gain.setValueAtTime(0, this.ctx.currentTime);

      this.noiseNode.connect(this.windFilter);
      this.windFilter.connect(this.windGain);
      this.windGain.connect(this.masterGain);
      this.noiseNode.start();

      this.schedulePelicanCall();
    }

    setMute(mute) {
      if (!this.ctx) return;
      const t = this.ctx.currentTime;
      if (mute) {
        this.masterGain.gain.linearRampToValueAtTime(0, t + 0.1);
      } else {
        this.masterGain.gain.linearRampToValueAtTime(0.55, t + 0.1);
      }
    }

    update(speed, crankAngle) {
      if (!this.ctx || this.ctx.state !== 'running') return;
      const t = this.ctx.currentTime;

      // 风噪与胎噪随速度上升
      const spdNorm = Math.min(1.0, speed / 10.0);
      const targetGain = speed > 0.05 ? spdNorm * 0.35 : 0;
      const targetFreq = 300 + spdNorm * 1200;
      this.windGain.gain.setTargetAtTime(targetGain, t, 0.05);
      this.windFilter.frequency.setTargetAtTime(targetFreq, t, 0.05);

      // 链条飞轮棘齿啮合咔哒声
      if (speed > 0.05) {
        const toothPhase = (crankAngle * CHAINRING_TEETH) / (2 * Math.PI);
        const currentTooth = Math.floor(toothPhase);
        if (currentTooth !== this.lastClickTooth) {
          this.lastClickTooth = currentTooth;
          this.playChainTick();
        }
      }
    }

    playChainTick() {
      if (!this.ctx) return;
      const osc = this.ctx.createOscillator();
      const gain = this.ctx.createGain();
      const filter = this.ctx.createBiquadFilter();

      filter.type = 'highpass';
      filter.frequency.setValueAtTime(2400, this.ctx.currentTime);

      osc.type = 'triangle';
      osc.frequency.setValueAtTime(1800 + Math.random() * 400, this.ctx.currentTime);

      const t = this.ctx.currentTime;
      gain.gain.setValueAtTime(0.04, t);
      gain.gain.exponentialRampToValueAtTime(0.0001, t + 0.015);

      osc.connect(filter);
      filter.connect(gain);
      gain.connect(this.masterGain);

      osc.start(t);
      osc.stop(t + 0.018);
    }

    // 偶发性低沉鹈鹕喉鸣 (艺术拟声：成鸟天然少语，低频共振管合成)
    schedulePelicanCall() {
      const nextDelay = 7000 + Math.random() * 9000;
      this.pelicanTimer = setTimeout(() => {
        if (state.speed > 0.5 && state.audioEnabled) {
          this.playPelicanCroak();
        }
        this.schedulePelicanCall();
      }, nextDelay);
    }

    playPelicanCroak() {
      if (!this.ctx) return;
      const t = this.ctx.currentTime;
      const osc1 = this.ctx.createOscillator();
      const osc2 = this.ctx.createOscillator();
      const formant = this.ctx.createBiquadFilter();
      const gain = this.ctx.createGain();

      osc1.type = 'sawtooth';
      osc1.frequency.setValueAtTime(115, t);
      osc1.frequency.linearRampToValueAtTime(85, t + 0.42);

      osc2.type = 'sine';
      osc2.frequency.setValueAtTime(172, t);
      osc2.frequency.linearRampToValueAtTime(130, t + 0.42);

      formant.type = 'bandpass';
      formant.frequency.setValueAtTime(420, t);
      formant.Q.setValueAtTime(3.5, t);

      gain.gain.setValueAtTime(0.0001, t);
      gain.gain.linearRampToValueAtTime(0.18, t + 0.06);
      gain.gain.exponentialRampToValueAtTime(0.001, t + 0.45);

      osc1.connect(formant);
      osc2.connect(formant);
      formant.connect(gain);
      gain.connect(this.masterGain);

      osc1.start(t);
      osc2.start(t);
      osc1.stop(t + 0.48);
      osc2.stop(t + 0.48);
    }
  }

    bill.position.set(0.16, 0, 0);
    bill.castShadow = true;
    billGroup.add(bill);

    // 牢固锚定在喙尖的下弯尖钩 (Hooked Nail)
    const hookGeo = new THREE.TorusGeometry(0.016, 0.006, 8, 12, Math.PI / 1.5);
    const hook = new THREE.Mesh(hookGeo, mats.beak);
    hook.rotation.z = -Math.PI / 2;
    hook.position.set(0.33, -0.018, 0);
    billGroup.add(hook);

    // 下喙与弹性喉囊 (Gular Pouch)
    const pouchGeo = new THREE.ConeGeometry(0.048, 0.25, 12);
    pouchGeo.scale(1.15, 1, 0.42);
    const pouchMesh = new THREE.Mesh(pouchGeo, mats.pouch);
    pouchMesh.rotation.z = -Math.PI / 2 + 0.12;
    pouchMesh.position.set(0.14, -0.042, 0);
    billGroup.add(pouchMesh);
    threeObjects.pouchMesh = pouchMesh;

    // 羽翼分层 (双翼扶在把手保持平衡)
    function createWing(zSide) {
      const wGroup = new THREE.Group();
      wGroup.position.set(-0.16, 0.72, zSide * 0.15);

      // 第 1 层：白色大覆羽 (Coverts)
      const covGeo = new THREE.BoxGeometry(0.24, 0.11, 0.03);
      const covMesh = new THREE.Mesh(covGeo, mats.pelicanWhite);
      covMesh.position.set(0.11, -0.02, zSide * 0.015);
      covMesh.rotation.z = -0.28;
      wGroup.add(covMesh);

      // 第 2 层：白色次级飞羽 (Secondaries)
      const secGeo = new THREE.BoxGeometry(0.2, 0.07, 0.025);
      const secMesh = new THREE.Mesh(secGeo, mats.pelicanWhite);
      secMesh.position.set(0.24, -0.04, zSide * 0.02);
      secMesh.rotation.z = -0.36;
      wGroup.add(secMesh);

      // 第 3 层：黑色初级飞羽 (Primaries) 特征黑尖
      const primGeo = new THREE.BoxGeometry(0.18, 0.06, 0.02);
      const primMesh = new THREE.Mesh(primGeo, mats.pelicanBlack);
      primMesh.position.set(0.38, -0.065, zSide * 0.025);
      primMesh.rotation.z = -0.44;
      wGroup.add(primMesh);

      pelicanGroup.add(wGroup);
      return wGroup;
    }
    createWing(1);
    createWing(-1);

    // 鸟类两段式后肢骨骼几何网格
    function createLegMeshes() {
      const thighMesh = new THREE.Mesh(new THREE.CylinderGeometry(0.024, 0.018, LEG_L1_M, 10), mats.pelicanWhite);
      thighMesh.castShadow = true;
      scene.add(thighMesh);

      const shinMesh = new THREE.Mesh(new THREE.CylinderGeometry(0.016, 0.012, LEG_L2_M, 10), mats.legs);
      shinMesh.castShadow = true;
      scene.add(shinMesh);

      // 全蹼足 (Totipalmate Foot): 4 趾带蹼
      const footGroup = new THREE.Group();
      const webGeo = new THREE.ConeGeometry(0.065, 0.11, 4);
      webGeo.scale(1, 0.15, 0.9);
      const webMesh = new THREE.Mesh(webGeo, mats.legs);
      webMesh.rotation.x = Math.PI / 2;
      webMesh.position.z = 0.035;
      footGroup.add(webMesh);

      for (let t = 0; t < 4; t++) {
        const toe = new THREE.Mesh(new THREE.CylinderGeometry(0.003, 0.002, 0.09, 6), mats.beak);
        toe.rotation.x = Math.PI / 2;
        toe.rotation.y = (t - 1.5) * 0.28;
        toe.position.set((t - 1.5) * 0.015, 0, 0.04);
        footGroup.add(toe);
      }
      scene.add(footGroup);

      return { thighMesh, shinMesh, footGroup };
    }

    const rightLeg = createLegMeshes();
    threeObjects.rightThighMesh = rightLeg.thighMesh;
    threeObjects.rightShinMesh = rightLeg.shinMesh;
    threeObjects.rightFootGroup = rightLeg.footGroup;

    const leftLeg = createLegMeshes();
    threeObjects.leftThighMesh = leftLeg.thighMesh;
    threeObjects.leftShinMesh = leftLeg.shinMesh;
    threeObjects.leftFootGroup = leftLeg.footGroup;

  function onWindowResize() {
    if (!renderer || !camera) return;
    camera.aspect = window.innerWidth / window.innerHeight;
    camera.updateProjectionMatrix();
    renderer.setSize(window.innerWidth, window.innerHeight);
  }

  function orientBone(mesh, p1, p2) {
    const mid = new THREE.Vector3().addVectors(p1, p2).multiplyScalar(0.5);
    mesh.position.copy(mid);
    mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), p2.clone().sub(p1).normalize());
  }

  function update3D(wheelAngle, crankAngle, speed, time) {
    if (!renderer) return;

    threeObjects.rearWheel.rotation.z = -wheelAngle;
    threeObjects.frontWheel.rotation.z = -wheelAngle;

    threeObjects.rightCrankGroup.rotation.z = -crankAngle;
    threeObjects.leftCrankGroup.rotation.z = -crankAngle + Math.PI;

    threeObjects.rightPedal.rotation.z = crankAngle;
    threeObjects.leftPedal.rotation.z = crankAngle - Math.PI;

    const rPedalPos = new THREE.Vector3();
    const lPedalPos = new THREE.Vector3();
    threeObjects.rightPedal.getWorldPosition(rPedalPos);
    threeObjects.leftPedal.getWorldPosition(lPedalPos);

    const rHip = new THREE.Vector3(-0.16, 0.65, 0.08);
    const lHip = new THREE.Vector3(-0.16, 0.65, -0.08);

    solve3DBirdLegIK(rHip, rPedalPos, LEG_L1_M, LEG_L2_M, threeObjects.rightLegJoint);
    solve3DBirdLegIK(lHip, lPedalPos, LEG_L1_M, LEG_L2_M, threeObjects.leftLegJoint);

    orientBone(threeObjects.rightThighMesh, rHip, threeObjects.rightLegJoint);
    orientBone(threeObjects.rightShinMesh, threeObjects.rightLegJoint, rPedalPos);
    threeObjects.rightFootGroup.position.copy(rPedalPos);
    threeObjects.rightFootGroup.rotation.z = 0;

    orientBone(threeObjects.leftThighMesh, lHip, threeObjects.leftLegJoint);
    orientBone(threeObjects.leftShinMesh, threeObjects.leftLegJoint, lPedalPos);
    threeObjects.leftFootGroup.position.copy(lPedalPos);
    threeObjects.leftFootGroup.rotation.z = 0;

    // 喉囊微动
    if (threeObjects.pouchMesh) {
      const pouchVib = speed > 0.05 ? Math.sin(time * 7) * 0.08 * Math.min(1.0, speed / 4.5) : 0;
      threeObjects.pouchMesh.scale.x = 1.15 + pouchVib;
      threeObjects.pouchMesh.scale.y = 1.0 + pouchVib * 0.5;
    }

    // 地面路标白线纯滚动移位
    const stripeSpacing = 1.8;
    const stripeSpan = 40 * stripeSpacing;
    const scrollX = (wheelAngle * WHEEL_RADIUS) % stripeSpacing;
    threeObjects.groundStripes.forEach((stripe, idx) => {
      let x = (idx - 20) * stripeSpacing - scrollX;
      if (x < -stripeSpan / 2) x += stripeSpan;
      if (x > stripeSpan / 2) x -= stripeSpan;
      stripe.position.x = x;
    });

    // 路桩滚动
    const postSpan = 80;
    threeObjects.sceneryItems.forEach((post, idx) => {
      let x = (idx - 10) * 4.0 - ((wheelAngle * WHEEL_RADIUS) % postSpan);
      if (x < -postSpan / 2) x += postSpan;
      if (x > postSpan / 2) x -= postSpan;
      post.position.x = x;
    });

    controls.update();
    renderer.render(scene, camera);
  }

     6. 主循环 (动画、物理约束与模式无缝切换)
     ========================================================================== */
  let fpsCounter = 0;
  let fpsTimer = performance.now();
  let lastFpsUpdate = performance.now();

  function animate(now) {
    requestAnimationFrame(animate);

    const dt = Math.min(0.1, (now - state.lastTime) / 1000);
    state.lastTime = now;

    // 物理学计算：无滑移纯滚动角速度
    // omega_wheel * R = v  ==>  omega_wheel = v / R
    const wheelOmega = state.speed / WHEEL_RADIUS;
    const crankOmega = wheelOmega * (COG_TEETH / CHAINRING_TEETH);

    // 累计转角单调递增 (不取模)
    state.totalWheelAngle += wheelOmega * dt;
    state.totalCrankAngle += crankOmega * dt;

    // 声音同步
    if (state.audioEnabled) {
      audioEngine.update(state.speed, state.totalCrankAngle);
    }

    // 渲染更新
    if (state.mode === '2.5d') {
      updateSVG(state.totalWheelAngle, state.totalCrankAngle, state.speed, now / 1000);
    } else {
      update3D(state.totalWheelAngle, state.totalCrankAngle, state.speed, now / 1000);
    }

    // FPS 计算与状态条刷新
    fpsCounter++;
    if (now - lastFpsUpdate >= 400) {
      const fps = Math.round((fpsCounter * 1000) / (now - lastFpsUpdate));
      document.getElementById('hud-fps').textContent = `${fps} FPS`;
      fpsCounter = 0;
      lastFpsUpdate = now;

      const rpm = ((crankOmega * 60) / (2 * Math.PI)).toFixed(1);
      document.getElementById('hud-cadence').textContent = `${rpm} RPM`;
    }
  }
