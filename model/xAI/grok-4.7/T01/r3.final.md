function solve2(hip, target, L1, L2, pole) {
  let dx = target.x - hip.x;
  let dy = target.y - hip.y;
  let dz = target.z - hip.z;
  let dist = Math.hypot(dx, dy, dz) || 1e-8;
  const max = L1 + L2 - 1e-5;
  const min = Math.abs(L1 - L2) + 1e-5;
  if (dist > max) { const k = max / dist; dx *= k; dy *= k; dz *= k; dist = max; }
  else if (dist < min) { const k = min / dist; dx *= k; dy *= k; dz *= k; dist = min; }
  const dirx = dx / dist, diry = dy / dist, dirz = dz / dist;
  const cosA = clamp((L1 * L1 + dist * dist - L2 * L2) / (2 * L1 * dist), -1, 1);
  const along = L1 * cosA;
  const h = Math.sqrt(Math.max(0, L1 * L1 - along * along));
  let px = pole.x - hip.x, py = pole.y - hip.y, pz = pole.z - hip.z;
  const dot = px * dirx + py * diry + pz * dirz;
  px -= dirx * dot; py -= diry * dot; pz -= dirz * dot;
  let pl = Math.hypot(px, py, pz);
  if (pl < 1e-8) { px = 0; py = 0; pz = 1; pl = 1; }
  px /= pl; py /= pl; pz /= pl;
  return {
    x: hip.x + dirx * along + px * h,
    y: hip.y + diry * along + py * h,
    z: hip.z + dirz * along + pz * h
  };
}

function cubic(p0, p1, p2, p3, t) {

    z: u * u * u * p0.z + 3 * u * u * t * p1.z + 3 * u * t * t * p2.z + t * t * t * p3.z
  };
}

function shortestSweep(from, to) {
  let d = to - from;
  while (d <= -Math.PI) d += TAU;
  while (d > Math.PI) d -= TAU;
  return d;
}
function longestSweep(from, to) {
  const s = shortestSweep(from, to);
  return s >= 0 ? s - TAU : s + TAU;
}

function buildChain() {
  const c1 = { x: GEO.bbX, y: GEO.bbY };
  const c2 = { x: 0, y: GEO.wheelRadius };
  const r1 = GEO.chainringRadius;
  const r2 = GEO.cogRadius;
  const dx = c2.x - c1.x;
  const dy = c2.y - c1.y;
  const dist = Math.hypot(dx, dy);
  const ang = Math.atan2(dy, dx);
  const phi = Math.acos(clamp((r1 - r2) / dist, -1, 1));
  const aA = ang + phi;
  const aB = ang - phi;
  const upper = (c1.y + Math.sin(aA) * r1) > (c1.y + Math.sin(aB) * r1) ? aA : aB;
  const lower = upper === aA ? aB : aA;
  const cu = { x: c2.x + Math.cos(upper) * r2, y: c2.y + Math.sin(upper) * r2 };
  const ru = { x: c1.x + Math.cos(upper) * r1, y: c1.y + Math.sin(upper) * r1 };
  const rl = { x: c1.x + Math.cos(lower) * r1, y: c1.y + Math.sin(lower) * r1 };
  const cl = { x: c2.x + Math.cos(lower) * r2, y: c2.y + Math.sin(lower) * r2 };
  const pts = [];
  function line(a, b, n) {
    for (let i = 0; i <= n; i++) {
      const t = i / n;
      pts.push({ x: a.x + (b.x - a.x) * t, y: a.y + (b.y - a.y) * t });
    }
  }
  function arc(cx, cy, r, a0, sweep, n) {
    for (let i = 0; i <= n; i++) {
      const a = a0 + sweep * (i / n);
      pts.push({ x: cx + Math.cos(a) * r, y: cy + Math.sin(a) * r });
    }
  }
  line(cu, ru, 10);
  arc(c1.x, c1.y, r1, upper, longestSweep(upper, lower), 24);
  line(rl, cl, 10);
  arc(c2.x, c2.y, r2, lower, shortestSweep(lower, upper), 12);
  const cum = [0];
  for (let i = 1; i < pts.length; i++) {
    cum.push(cum[i - 1] + Math.hypot(pts[i].x - pts[i - 1].x, pts[i].y - pts[i - 1].y));
  }
  return { pts, cum, length: cum[cum.length - 1] };
}
const chain = buildChain();

function pointOnChain(s) {
  const L = chain.length;
  s = mod(s, L);
  let i = 1;
  while (i < chain.cum.length - 1 && chain.cum[i] < s) i++;
  const i0 = i - 1;
  const span = chain.cum[i] - chain.cum[i0] || 1e-8;
  const t = clamp((s - chain.cum[i0]) / span, 0, 1);
  const a = chain.pts[i0];
  const b = chain.pts[i];
  return {
    x: a.x + (b.x - a.x) * t,
    y: a.y + (b.y - a.y) * t,
    ang: Math.atan2(b.y - a.y, b.x - a.x)
  };
}

function setLine(el, a, b) {
  const p = proj(a.x, a.y, a.z || 0);
  const q = proj(b.x, b.y, b.z || 0);
  el.setAttribute('x1', p[0]);
  el.setAttribute('y1', p[1]);
  el.setAttribute('x2', q[0]);
  el.setAttribute('y2', q[1]);
}
function polyAttr(el, pts) {
  el.setAttribute('points', pts.map(function (p) {
    return p[0].toFixed(1) + ',' + p[1].toFixed(1);
  }).join(' '));
}
function leaf(a, b, w) {
  const mx = (a[0] + b[0]) / 2;
  const my = (a[1] + b[1]) / 2;
  const dx = b[0] - a[0];
  const dy = b[1] - a[1];
  const L = Math.hypot(dx, dy) || 1;
  const nx = -dy / L * w;
  const ny = dx / L * w;
  return 'M ' + a[0].toFixed(1) + ' ' + a[1].toFixed(1) +
    ' Q ' + (mx + nx).toFixed(1) + ' ' + (my + ny).toFixed(1) + ' ' + b[0].toFixed(1) + ' ' + b[1].toFixed(1) +
    ' Q ' + (mx - nx * 0.4).toFixed(1) + ' ' + (my - ny * 0.4).toFixed(1) + ' ' + a[0].toFixed(1) + ' ' + a[1].toFixed(1) + ' Z';
}
function mix(a, b, t) {
  return { x: a.x + (b.x - a.x) * t, y: a.y + (b.y - a.y) * t, z: a.z + (b.z - a.z) * t };
}
function norm3(v) {
  const L = Math.hypot(v.x, v.y, v.z) || 1;
  return { x: v.x / L, y: v.y / L, z: v.z / L };
}
function add3(a, b, s) {
  return { x: a.x + b.x * s, y: a.y + b.y * s, z: a.z + b.z * s };
}
function cross(a, b) {
  return {
    x: a.y * b.z - a.z * b.y,
    y: a.z * b.x - a.x * b.z,
    z: a.x * b.y - a.y * b.x
  };
}
function featherSpan(elbow, wrist, hint, t, len, spread, lift) {
  const ax = norm3({ x: wrist.x - elbow.x, y: wrist.y - elbow.y, z: wrist.z - elbow.z });
  const side = norm3(cross(ax, hint));
  const base = mix(elbow, wrist, t);
  const tip = add3(add3(add3(base, ax, len * 0.35), side, spread), { x: 0, y: 1, z: 0 }, lift);
  return [base, tip];
}
function footWorld(pedal) {
  const s = pedal.z >= 0 ? 1 : -1;
  const y = pedal.y + 0.012;
  const z = pedal.z;
  return [
    { x: pedal.x - 0.036, y: y, z: z },
    { x: pedal.x + 0.030, y: y, z: z - s * 0.030 },
    { x: pedal.x + 0.072, y: y, z: z - s * 0.004 },
    { x: pedal.x + 0.034, y: y, z: z + s * 0.032 }
  ];
}
function paintLeg(leg, hip, knee, ankle, foot, pedal) {
  setLine(leg.thigh, hip, knee);
  setLine(leg.shin, knee, ankle);
  setLine(leg.tarsus, ankle, foot);
  const k = proj(knee.x, knee.y, knee.z);
  const a = proj(ankle.x, ankle.y, ankle.z);
  leg.knee.setAttribute('cx', k[0]);
  leg.knee.setAttribute('cy', k[1]);
  leg.ankle.setAttribute('cx', a[0]);
  leg.ankle.setAttribute('cy', a[1]);
  const toes = footWorld(pedal);
  polyAttr(leg.web, toes.map(function (p) { return proj(p.x, p.y, p.z); }));
  const mid = { x: pedal.x + 0.01, y: pedal.y + 0.012, z: pedal.z };
  toes.forEach(function (t, i) { setLine(leg.toes[i], mid, t); });
  if (leg.knob) {
    const knob = proj(ankle.x - 0.012, ankle.y, ankle.z);
    leg.knob.setAttribute('cx', knob[0]);
    leg.knob.setAttribute('cy', knob[1]);
  }
}
function paintWing(slots, elbow, wrist, outward) {
  const hintSec = { x: 0, y: -0.2, z: outward };
  const hintCov = { x: 0, y: 1, z: outward * 0.3 };
  slots.sec.forEach(function (el, i) {
    const t = 0.15 + i * 0.13;
    const pair = featherSpan(elbow, wrist, hintSec, t, 0.16 + i * 0.01, 0.07 + i * 0.012, -0.02);
    el.setAttribute('d', leaf(proj(pair[0].x, pair[0].y, pair[0].z), proj(pair[1].x, pair[1].y, pair[1].z), 16));
    el.setAttribute('fill', '#6e5642');
  });
  slots.pri.forEach(function (el, i) {
    const ax = norm3({ x: wrist.x - elbow.x, y: wrist.y - elbow.y, z: wrist.z - elbow.z });
    const side = norm3(cross(ax, { x: 0.2, y: 0.2, z: outward }));
    const base = mix(elbow, wrist, 0.78 + i * 0.04);
    const tip = add3(add3(wrist, ax, 0.05 + i * 0.03), side, 0.08 + i * 0.035);
    tip.y += 0.01 * i;
    el.setAttribute('d', leaf(proj(base.x, base.y, base.z), proj(tip.x, tip.y, tip.z), 13));
    el.setAttribute('fill', i > 2 ? '#2b221c' : '#3a2e24');
  });
  slots.cov.forEach(function (el, i) {
    const t = 0.1 + i * 0.16;
    const pair = featherSpan(elbow, wrist, hintCov, t, 0.10, 0.045 + i * 0.006, 0.03);
    el.setAttribute('d', leaf(proj(pair[0].x, pair[0].y, pair[0].z), proj(pair[1].x, pair[1].y, pair[1].z), 12));
    el.setAttribute('fill', '#a08a68');
  });
}

function tube(parent, a, b, r, mat) {
  const dir = new THREE.Vector3(b.x - a.x, b.y - a.y, b.z - a.z);
  const len = dir.length();
  const mesh = new THREE.Mesh(new THREE.CylinderGeometry(r, r, len, 10), mat);
  mesh.position.set((a.x + b.x) / 2, (a.y + b.y) / 2, (a.z + b.z) / 2);
  mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), dir.normalize());
  mesh.castShadow = true;
  mesh.receiveShadow = true;
  parent.add(mesh);
  return mesh;
}
function joint(parent, p, r, mat) {
  const m = new THREE.Mesh(new THREE.SphereGeometry(r, 12, 10), mat);
  m.position.set(p.x, p.y, p.z);
  m.castShadow = true;
  parent.add(m);
  return m;
}
function makeWheel(withCog) {
  const g = new THREE.Group();
  const major = GEO.wheelRadius - 0.017;
  const tire = new THREE.Mesh(new THREE.TorusGeometry(major, 0.017, 12, 40), M.tire);
  tire.castShadow = true;
  tire.receiveShadow = true;
  g.add(tire);
  const rim = new THREE.Mesh(new THREE.TorusGeometry(major - 0.02, 0.006, 8, 32), M.rim);
  rim.castShadow = true;
  g.add(rim);
  const hub = new THREE.Mesh(new THREE.CylinderGeometry(0.02, 0.02, 0.09, 12), M.hub);
  hub.rotation.x = Math.PI / 2;
  hub.castShadow = true;
  g.add(hub);
  const spokeLen = major - 0.04;
  const spokeGeo = new THREE.CylinderGeometry(0.0022, 0.0022, spokeLen, 5);
  for (let i = 0; i < 16; i++) {
    const a = (i / 16) * TAU;
    const s = new THREE.Mesh(spokeGeo, M.spoke);
    const dir = new THREE.Vector3(Math.cos(a), Math.sin(a), 0);
    s.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), dir);
    s.position.copy(dir.multiplyScalar(spokeLen / 2 + 0.02));
    g.add(s);
  }
  if (withCog) {
    const cog = sprocket3(GEO.cogRadius, GEO.cogTeeth, 0.01);
    cog.position.z = GEO.chainZ;
    g.add(cog);
  }
  return g;
}
function sprocket3(radius, teeth, width) {
  const g = new THREE.Group();
  const disc = new THREE.Mesh(new THREE.CylinderGeometry(radius * 0.7, radius * 0.7, width, 20), M.cog);
  disc.rotation.x = Math.PI / 2;
  disc.castShadow = true;
  g.add(disc);
  const toothGeo = new THREE.BoxGeometry(0.007, radius * 0.42, width);
  for (let i = 0; i < teeth; i++) {
    const a = (i / teeth) * TAU;
    const t = new THREE.Mesh(toothGeo, M.cog);
    t.position.set(Math.cos(a) * radius * 0.86, Math.sin(a) * radius * 0.86, 0);
    t.rotation.z = a;
    g.add(t);
  }
  return g;
}

function placeY(mesh, a, b) {
  const d = M._v1.set(b.x - a.x, b.y - a.y, b.z - a.z);
  const len = d.length();
  if (len < 1e-6) return;
  mesh.position.set((a.x + b.x) / 2, (a.y + b.y) / 2, (a.z + b.z) / 2);
  mesh.quaternion.setFromUnitVectors(M._y, d.multiplyScalar(1 / len));
  mesh.scale.set(1, len / mesh.userData.len, 1);
}
function makeFoot() {
  const g = new THREE.Group();
  const toes = [[-0.036, 0], [0.030, -0.030], [0.072, -0.004], [0.034, 0.032]];
  const c = [0.012, 0];
  const positions = [];
  function tri(a, b, d) {
    [a, b, d].forEach(function (p) { positions.push(p[0], 0, p[1]); });
  }
  tri(c, toes[0], toes[1]);
  tri(c, toes[1], toes[2]);
  tri(c, toes[2], toes[3]);
  tri(c, toes[3], toes[0]);
  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geo.computeVertexNormals();
  const web = new THREE.Mesh(geo, M.foot);
  web.castShadow = true;
  g.add(web);
  toes.forEach(function (t) {
    const len = Math.hypot(t[0] - c[0], t[1] - c[1]);
    const toe = new THREE.Mesh(new THREE.CylinderGeometry(0.006, 0.004, len, 5), M.foot);
    const ang = Math.atan2(t[1] - c[1], t[0] - c[0]);
    toe.rotation.z = ang - Math.PI / 2;
    toe.position.set((t[0] + c[0]) / 2, 0.004, (t[1] + c[1]) / 2);
    toe.castShadow = true;
    g.add(toe);
  });
  return g;
}

function applyCamera() {

  const sp = Math.sin(o.phi);

  );
  camera.lookAt(o.target);
}

function updateFeathers3(bucket, elbow, wrist, outward) {
  const hintSec = { x: 0, y: -0.15, z: outward };
  const hintCov = { x: 0, y: 1, z: outward * 0.25 };
  bucket.sec.forEach(function (mesh, i) {
    const pair = featherSpan(elbow, wrist, hintSec, 0.12 + i * 0.13, 0.24, 0.11 + i * 0.02, -0.02);
    placeX(mesh, pair[0], pair[1]);
  });
  bucket.cov.forEach(function (mesh, i) {
    const pair = featherSpan(elbow, wrist, hintCov, 0.12 + i * 0.15, 0.16, 0.06 + i * 0.012, 0.03);
    placeX(mesh, pair[0], pair[1]);
  });
  const ax = norm3({ x: wrist.x - elbow.x, y: wrist.y - elbow.y, z: wrist.z - elbow.z });
  const side = norm3(cross(ax, { x: 0.2, y: 0.15, z: outward }));
  bucket.pri.forEach(function (mesh, i) {
    const base = mix(elbow, wrist, 0.8);
    const tip = add3(add3(wrist, ax, 0.08 + i * 0.04), side, 0.10 + i * 0.04);
    placeX(mesh, base, tip);
  });
}

const audio = { ready: false, muted: false, ctx: null, lastTooth: 0, rideTime: 0, nextCall: 5 };
function initAudio() {
  const Ctx = window.AudioContext || window.webkitAudioContext;
  if (!Ctx || audio.ready) return;
  const ctx = new Ctx();
  const master = ctx.createGain();
  master.gain.value = 1;
  master.connect(ctx.destination);
  const nBuf = ctx.createBuffer(1, ctx.sampleRate * 2, ctx.sampleRate);
  const nd = nBuf.getChannelData(0);
  let brown = 0;
  for (let i = 0; i < nd.length; i++) {
    brown = (brown + (Math.random() * 2 - 1) * 0.08) * 0.97;
    nd[i] = brown * 2.4;
  }
  const noise = ctx.createBufferSource();
  noise.buffer = nBuf;
  noise.loop = true;
  const filt = ctx.createBiquadFilter();
  filt.type = 'lowpass';
  filt.frequency.value = 420;
  const wind = ctx.createGain();
  wind.gain.value = 0;
  noise.connect(filt);
  filt.connect(wind);
  wind.connect(master);
  noise.start();
  const cBuf = ctx.createBuffer(1, Math.floor(ctx.sampleRate * 0.03), ctx.sampleRate);
  const cd = cBuf.getChannelData(0);
  for (let i = 0; i < cd.length; i++) {
    const t = i / ctx.sampleRate;
    const env = Math.exp(-t * 160);
    cd[i] = ((Math.random() * 2 - 1) * 0.7 + Math.sin(TAU * 2100 * t) * 0.45) * env;
  }
  audio.ctx = ctx;
  audio.master = master;
  audio.wind = wind;
  audio.filter = filt;
  audio.click = cBuf;
  audio.ready = true;
  audio.lastTooth = Math.floor((state.wheelAngle / TAU) * GEO.cogTeeth);
  if (ctx.state === 'suspended') ctx.resume().catch(function () {});
}
function playCall(time) {
  [150, 230, 310].forEach(function (f, i) {
    const o = audio.ctx.createOscillator();
    const g = audio.ctx.createGain();
    o.type = i === 0 ? 'sine' : 'triangle';
    o.frequency.setValueAtTime(f * 1.15, time);
    o.frequency.exponentialRampToValueAtTime(f * 0.82, time + 0.28);
    const peak = i === 0 ? 0.18 : 0.05;
    g.gain.setValueAtTime(0.0001, time);
    g.gain.exponentialRampToValueAtTime(peak, time + 0.03);
    g.gain.exponentialRampToValueAtTime(0.0001, time + 0.3);
    o.connect(g);
    g.connect(audio.master);
    o.start(time);
    o.stop(time + 0.32);
  });
}
function updateAudio(dt) {
  if (!audio.ready || audio.ctx.state !== 'running') return;
  const t = audio.ctx.currentTime;
  audio.master.gain.setTargetAtTime(audio.muted ? 0 : 1, t, 0.04);
  const windAmp = audio.muted ? 0 : Math.min(0.2, state.speed * 0.026);
  audio.wind.gain.setTargetAtTime(windAmp, t, 0.08);
  audio.filter.frequency.setTargetAtTime(260 + state.speed * 70, t, 0.1);
  const idx = Math.floor((state.wheelAngle / TAU) * GEO.cogTeeth);
  let n = idx - audio.lastTooth;
  if (n < 0) n = 0;
  if (!audio.muted && state.speed > 0.04 && n > 0) {
    if (n > 4) n = 2;
    const rate = Math.max(1, (state.speed / GEO.wheelRadius) * GEO.cogTeeth / TAU);
    const gain = Math.min(0.34, 1.05 / Math.sqrt(rate));
    for (let k = 0; k < n; k++) {
      const src = audio.ctx.createBufferSource();
      src.buffer = audio.click;
      const g = audio.ctx.createGain();
      g.gain.value = gain * (0.75 + Math.random() * 0.25);
      src.connect(g);
      g.connect(audio.master);
      src.start(t + k * (0.45 / rate));
    }
  }
  audio.lastTooth = idx;
  if (!audio.muted && state.speed > 0.45) {
    audio.rideTime += dt;
    if (audio.rideTime >= audio.nextCall) {
      playCall(t + 0.02);
      audio.rideTime = 0;
      audio.nextCall = 6 + Math.random() * 7;
    }
  }
}
