function buildChainPath() {
  const front = { x: BB.x, y: BB.y };
  const rear = { x: REAR.x, y: REAR.y };
  const dx = rear.x - front.x, dy = rear.y - front.y;
  const d = Math.hypot(dx, dy);
  const vx = dx / d, vy = dy / d;
  const c = (Rf - Rc) / d;
  const h = Math.sqrt(Math.max(0, 1 - c * c));
  const tans = [];
  for (const sign of [1, -1]) {
    const nx = vx * c - sign * vy * h;
    const ny = vy * c + sign * vx * h;
    tans.push({
      p1: { x: front.x + Rf * nx, y: front.y + Rf * ny },
      p2: { x: rear.x + Rc * nx, y: rear.y + Rc * ny }
    });
  }
  tans.sort((a, b) => (b.p1.y + b.p2.y) - (a.p1.y + a.p2.y));
  const upper = tans[0], lower = tans[1];
  const pts = [];
  function push(x, y) { pts.push({ x, y }); }
  function addLine(a, b, n) {
    for (let i = 0; i < n; i++) {
      const t = i / n;
      push(a.x + (b.x - a.x) * t, a.y + (b.y - a.y) * t);
    }
  }
  function addArc(cx, cy, r, a0, a1, n) {
    let sweep = a0 - a1;
    while (sweep < 0) sweep += Math.PI * 2;
    while (sweep >= Math.PI * 2) sweep -= Math.PI * 2;
    for (let i = 0; i < n; i++) {
      const a = a0 - sweep * (i / n);
      push(cx + Math.cos(a) * r, cy + Math.sin(a) * r);
    }
  }
  const aUpF = Math.atan2(upper.p1.y - front.y, upper.p1.x - front.x);
  const aLoF = Math.atan2(lower.p1.y - front.y, lower.p1.x - front.x);
  const aLoR = Math.atan2(lower.p2.y - rear.y, lower.p2.x - rear.x);
  const aUpR = Math.atan2(upper.p2.y - rear.y, upper.p2.x - rear.x);
  addLine(upper.p2, upper.p1, 16);
  addArc(front.x, front.y, Rf, aUpF, aLoF, 24);
  addLine(lower.p1, lower.p2, 16);
  addArc(rear.x, rear.y, Rc, aLoR, aUpR, 12);
  let s = 0;
  pts[0].s = 0;
  for (let i = 1; i < pts.length; i++) {
    s += Math.hypot(pts[i].x - pts[i - 1].x, pts[i].y - pts[i - 1].y);
    pts[i].s = s;
  }
  const close = Math.hypot(pts[0].x - pts[pts.length - 1].x, pts[0].y - pts[pts.length - 1].y);
  return { pts, length: s + close };
}

function sampleChain(dist) {

}

function featherPoints(base, tip, width) {
  const p = project(base), q = project(tip);
  const dx = q.x - p.x, dy = q.y - p.y;
  const L = Math.hypot(dx, dy) || 1;
  const nx = -dy / L * width * 0.5, ny = dx / L * width * 0.5;
  const mx = p.x * 0.35 + q.x * 0.65, my = p.y * 0.35 + q.y * 0.65;
  return p.x + "," + p.y + " " + (mx + nx) + "," + (my + ny) + " " + q.x + "," + q.y + " " + (mx - nx) + "," + (my - ny);
}

function placeTube(mesh, a, b) {
  const dx = b.x - a.x, dy = b.y - a.y, dz = b.z - a.z;
  const len = Math.hypot(dx, dy, dz);
  if (len < 1e-5) return;
  mesh.position.set((a.x + b.x) / 2, (a.y + b.y) / 2, (a.z + b.z) / 2);
  mesh.scale.set(1, len, 1);
  T.dir.set(dx / len, dy / len, dz / len);
  if (T.dir.y < -0.999) mesh.quaternion.set(1, 0, 0, 0);
  else mesh.quaternion.setFromUnitVectors(T.up, T.dir);
}
function placeTubeOverlap(mesh, a, b, extra) {
  const dx = b.x - a.x, dy = b.y - a.y, dz = b.z - a.z;
  const len = Math.hypot(dx, dy, dz) || 1e-5;
  const nx = dx / len, ny = dy / len, nz = dz / len;
  placeTube(
    mesh,
    { x: a.x - nx * extra, y: a.y - ny * extra, z: a.z - nz * extra },
    { x: b.x + nx * extra, y: b.y + ny * extra, z: b.z + nz * extra }
  );
}
function orientFeather(mesh, base, tip, normal, width) {
  const dx = tip.x - base.x, dy = tip.y - base.y, dz = tip.z - base.z;
  const len = Math.hypot(dx, dy, dz) || 1e-5;
  T.dir.set(dx / len, dy / len, dz / len);
  T.z.set(normal.x, normal.y, normal.z).normalize();
  T.y.crossVectors(T.z, T.dir);
  if (T.y.lengthSq() < 1e-8) T.y.set(0, 1, 0);
  T.y.normalize();
  T.z.crossVectors(T.dir, T.y).normalize();
  mesh.position.set(base.x + dx * 0.5, base.y + dy * 0.5, base.z + dz * 0.5);
  mesh.scale.set(len, width / 0.74, 1);
  T.m.makeBasis(T.dir, T.y, T.z);
  mesh.quaternion.setFromRotationMatrix(T.m);
}
function staticTube(a, b, radius, mat, parent) {
  const dx = b.x - a.x, dy = b.y - a.y, dz = b.z - a.z;
  const len = Math.hypot(dx, dy, dz) || 0.001;
  const mesh = new THREE.Mesh(new THREE.CylinderGeometry(radius, radius, len, 10), mat);
  mesh.position.set((a.x + b.x) / 2, (a.y + b.y) / 2, (a.z + b.z) / 2);
  const dir = new THREE.Vector3(dx / len, dy / len, dz / len);
  if (dir.y < -0.9999) mesh.quaternion.set(1, 0, 0, 0);
  else mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), dir);
  mesh.castShadow = true;
  mesh.receiveShadow = true;
  parent.add(mesh);
  return mesh;
}

  return x + 0.45;
}

function buildSprocket(teeth, radius, thickness) {
  const g = new THREE.Group();
  const disc = new THREE.Mesh(new THREE.CylinderGeometry(radius * 0.78, radius * 0.78, thickness, 28), metalMat);
  disc.rotation.x = Math.PI / 2;
  disc.castShadow = true;
  g.add(disc);
  const toothGeo = new THREE.BoxGeometry(0.012, 0.007, thickness);
  const inst = new THREE.InstancedMesh(toothGeo, metalMat, teeth);
  const dummy = new THREE.Object3D();
  for (let i = 0; i < teeth; i++) {
    const a = (i / teeth) * Math.PI * 2;
    dummy.position.set(Math.cos(a) * radius, Math.sin(a) * radius, 0);
    dummy.rotation.set(0, 0, a);
    dummy.updateMatrix();
    inst.setMatrixAt(i, dummy.matrix);
  }
  g.add(inst);
  return g;
}

function bindOrbit(dom) {
  const sph = { theta: 0, phi: 1, r: 3 };
  const offset = camera.position.clone().sub(camTarget);
  sph.r = offset.length();
  sph.phi = Math.acos(clamp(offset.y / sph.r, -1, 1));
  sph.theta = Math.atan2(offset.x, offset.z);
  const pointers = new Map();
  let pinch = null;
  function apply() {
    sph.phi = clamp(sph.phi, 0.2, 1.4);
    sph.r = clamp(sph.r, 1.25, 7.2);
    const sp = Math.sin(sph.phi);
    camera.position.set(
      camTarget.x + sph.r * sp * Math.sin(sph.theta),
      camTarget.y + sph.r * Math.cos(sph.phi),
      camTarget.z + sph.r * sp * Math.cos(sph.theta)
    );
    camera.lookAt(camTarget);
  }
  apply();
  function pinchDist() {
    const a = [...pointers.values()];
    if (a.length < 2) return null;
    return Math.hypot(a[0].x - a[1].x, a[0].y - a[1].y);
  }
  dom.addEventListener("pointerdown", (e) => {
    dom.setPointerCapture(e.pointerId);
    pointers.set(e.pointerId, { x: e.clientX, y: e.clientY, lx: e.clientX, ly: e.clientY });
    pinch = pinchDist();
  });
  dom.addEventListener("pointermove", (e) => {
    const p = pointers.get(e.pointerId);
    if (!p) return;
    if (pointers.size === 1) {
      sph.theta -= (e.clientX - p.lx) * 0.0052;
      sph.phi -= (e.clientY - p.ly) * 0.004;
      apply();
    } else {
      p.x = e.clientX;
      p.y = e.clientY;
      const d = pinchDist();
      if (pinch && d) {
        sph.r *= pinch / d;
        apply();
      }
      pinch = d;
    }
    p.lx = e.clientX;
    p.ly = e.clientY;
    p.x = e.clientX;
    p.y = e.clientY;
  });
  function endPtr(e) {
    pointers.delete(e.pointerId);
    pinch = pinchDist();
  }
  dom.addEventListener("pointerup", endPtr);
  dom.addEventListener("pointercancel", endPtr);
  dom.addEventListener("wheel", (e) => {
    e.preventDefault();
    sph.r *= Math.exp(e.deltaY * 0.001);
    apply();
  }, { passive: false });
  dom.addEventListener("contextmenu", (e) => e.preventDefault());
}

function armAudio() {
  if (audio) return;
  const ctx = new AudioContext();
  const master = ctx.createGain();
  master.gain.value = muted ? 0 : 0.85;
  master.connect(ctx.destination);
  const nLen = Math.floor(ctx.sampleRate * 1.5);
  const noiseBuf = ctx.createBuffer(1, nLen, ctx.sampleRate);
  const nd = noiseBuf.getChannelData(0);
  for (let i = 0; i < nLen; i++) nd[i] = Math.random() * 2 - 1;
  const clickBuf = ctx.createBuffer(1, Math.floor(ctx.sampleRate * 0.02), ctx.sampleRate);
  const cd = clickBuf.getChannelData(0);
  for (let i = 0; i < cd.length; i++) cd[i] = (Math.random() * 2 - 1) * Math.exp(-i / (ctx.sampleRate * 0.003));
  function loopNoise(filterType, freq, q) {
    const src = ctx.createBufferSource();
    src.buffer = noiseBuf;
    src.loop = true;
    const f = ctx.createBiquadFilter();
    f.type = filterType;
    f.frequency.value = freq;
    if (q) f.Q.value = q;
    const g = ctx.createGain();
    g.gain.value = 0;
    src.connect(f); f.connect(g); g.connect(master);
    src.start();
    return { gain: g, filter: f };
  }
  const wind = loopNoise("bandpass", 480, 0.55);
  const tire = loopNoise("lowpass", 170, 0.7);
  audio = { ctx, master, windGain: wind.gain, windFilter: wind.filter, tireGain: tire.gain, clickBuf };
  if (ctx.state === "suspended") ctx.resume();
  hintEl.hidden = true;
}
function playClick(time) {
  const src = audio.ctx.createBufferSource();

function playCall(time) {

function updateAudio(dt, crankOmega) {
  const toothRate = CHAINRING_TEETH * crankOmega / (2 * Math.PI);
  chainPhase += toothRate * dt;
  if (audio && !muted && speed > 0.02) {
    let n = 0;
    while (chainPhase >= 1 && n < 4) {
      chainPhase -= 1;
      playClick(audio.ctx.currentTime);
      n++;
    }
  }
  if (chainPhase >= 1) chainPhase %= 1;
  if (speed > 0.15) callCd -= dt;
  if (callCd <= 0) {
    callCd = 9 + Math.random() * 7;
    if (audio && !muted) playCall(audio.ctx.currentTime);
  }
  if (!audio) return;
  const now = audio.ctx.currentTime;
  audio.windGain.gain.setTargetAtTime(Math.min(0.22, speed * 0.028), now, 0.06);
  audio.tireGain.gain.setTargetAtTime(Math.min(0.14, speed * 0.018), now, 0.06);
  audio.windFilter.frequency.setTargetAtTime(300 + speed * 75, now, 0.1);
}
