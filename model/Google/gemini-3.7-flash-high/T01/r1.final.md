### 建模与物理实现思路

采用单源物理状态引擎同时驱动 2.5D SVG 与 3D Three.js 双模式渲染。基于无滑移滚动公式（v = ω_wheel × R_wheel）与牙盘飞轮传动比（48T/16T，曲柄角速度为后轮 1/3）严格进行单调转角积分；踏板受自由旋转与重力约束保持水平；腿部通过两段式逆运动学（2-Bone Avian IK）精确求解鸟类向后屈伸的踝关节（Intertarsal Joint），使全蹼足（*Pelecanus erythrorhynchos*）四趾平贴踏板；长喙带尖端弯钩，下附动态弹簧喉囊随踏频微幅起伏；双翼体现初级、次级飞羽与覆羽分层；Web Audio API 实时合成链条咔哒、风噪与低沉喉音，双模式无缝平滑切换并完整暴露评测接口。

  <script type="module">
    import * as THREE from 'three';
    import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

    /* =========================================================
       1. Physical Constants & Core State Engine
       ========================================================= */
    const PHYSICS = {
      WHEEL_RADIUS: 0.34,       // meters (approx. 700x28C wheel)
      WHEELBASE: 1.05,          // meters
      CHAINRING_TEETH: 48,      // Front Chainring
      COG_TEETH: 16,            // Rear Cog
      GEAR_RATIO: 16 / 48,      // crank angular velocity / wheel angular velocity = 1/3
      CRANK_LENGTH: 0.17,       // meters (170 mm)
      BB_POS: { x: -0.10, y: 0.28, z: 0 }, // Bottom Bracket position (3D coordinates in meters)
      SADDLE_POS: { x: -0.22, y: 0.72, z: 0 },
      HANDLEBAR_POS: { x: 0.48, y: 0.86, z: 0 },
      // Avian Leg Lengths for 2-Bone IK
      LEG_L1: 0.28,             // Thigh (Femur + Drumstick to Ankle)
      LEG_L2: 0.26              // Shank (Tarsus from Ankle to Foot)
    };

    // Shared Dynamic State between 2.5D and 3D Modes
    const state = {
      mode: '2.5d',             // '2.5d' | '3d'
      speed: 4.0,               // m/s
      targetSpeed: 4.0,
      distance: 0.0,            // meters traveled
      wheelAngle: 0.0,          // accumulated radians (monotonic, no modulo)
      crankAngle: 0.0,          // accumulated radians (monotonic, no modulo)
      pedalAngles: [0.0, 0.0],  // left, right pedal tilt relative to ground
      footToPedal: [0.0, 0.0],  // left, right foot contact distance to pedal center (meters)
      hipToSaddle: 0.0,         // distance from hip to saddle (meters)
      cadenceRPM: 0.0,
      fps: 60,
      pouchJiggle: 0.0
    };

    /* =========================================================
       2. Real-Time Web Audio Engine
       ========================================================= */
    class BikeAudioEngine {
      constructor() {
        this.ctx = null;
        this.isMuted = true;
        this.windGain = null;
        this.windFilter = null;
        this.noiseNode = null;
        this.lastClickCrankAngle = 0;
        this.lastPelicanCallTime = 0;
        this.initialized = false;
      }

      init() {
        if (this.initialized) return;
        const AudioCtx = window.AudioContext || window.webkitAudioContext;
        if (!AudioCtx) return;
        this.ctx = new AudioCtx();

        // Master Gain
        this.masterGain = this.ctx.createGain();
        this.masterGain.gain.value = this.isMuted ? 0.0 : 0.6;
        this.masterGain.connect(this.ctx.destination);

        // Continuous Wind & Tire Friction Pink Noise Generator
        const bufferSize = this.ctx.sampleRate * 2;
        const noiseBuffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const output = noiseBuffer.getChannelData(0);
        let lastOut = 0.0;
        for (let i = 0; i < bufferSize; i++) {
          const white = Math.random() * 2 - 1;
          output[i] = (lastOut + 0.02 * white) / 1.02;
          lastOut = output[i];
          output[i] *= 3.2;
        }

        this.noiseNode = this.ctx.createBufferSource();
        this.noiseNode.buffer = noiseBuffer;
        this.noiseNode.loop = true;

        this.windFilter = this.ctx.createBiquadFilter();
        this.windFilter.type = 'lowpass';
        this.windFilter.frequency.value = 220;

        this.windGain = this.ctx.createGain();
        this.windGain.gain.value = 0.0;

        this.noiseNode.connect(this.windFilter);
        this.windFilter.connect(this.windGain);
        this.windGain.connect(this.masterGain);
        this.noiseNode.start();

        this.initialized = true;
      }

      unlock() {
        if (!this.initialized) {
          this.init();
        }
        if (this.ctx && this.ctx.state === 'suspended') {
          this.ctx.resume();
        }
      }

      playChainClick(intensity = 1.0) {
        if (!this.initialized || this.isMuted || this.ctx.state !== 'running') return;
        const now = this.ctx.currentTime;

        const osc = this.ctx.createOscillator();
        const clickGain = this.ctx.createGain();
        const filter = this.ctx.createBiquadFilter();

        osc.type = 'triangle';
        osc.frequency.setValueAtTime(1400 + Math.random() * 500, now);
        osc.frequency.exponentialRampToValueAtTime(320, now + 0.012);

        filter.type = 'bandpass';
        filter.frequency.value = 1800;
        filter.Q.value = 3.5;

        clickGain.gain.setValueAtTime(0.08 * intensity, now);
        clickGain.gain.exponentialRampToValueAtTime(0.0001, now + 0.012);

        osc.connect(filter);
        filter.connect(clickGain);
        clickGain.connect(this.masterGain);

        osc.start(now);
        osc.stop(now + 0.014);
      }

      playPelicanCall() {
        this.unlock();
        if (!this.initialized || this.ctx.state !== 'running') return;
        const now = this.ctx.currentTime;

        // Formant synthesis for guttural pelican croak
        const carrier = this.ctx.createOscillator();
        const mod = this.ctx.createOscillator();
        const modGain = this.ctx.createGain();
        const formant1 = this.ctx.createBiquadFilter();
        const formant2 = this.ctx.createBiquadFilter();
        const vocalGain = this.ctx.createGain();

        carrier.type = 'sawtooth';
        carrier.frequency.setValueAtTime(80, now);
        carrier.frequency.linearRampToValueAtTime(65, now + 0.45);

        mod.type = 'sine';
        mod.frequency.setValueAtTime(26, now);
        modGain.gain.setValueAtTime(55, now);
        mod.connect(carrier.frequency);

        formant1.type = 'bandpass';
        formant1.frequency.setValueAtTime(360, now);
        formant1.Q.value = 4.0;

        formant2.type = 'bandpass';
        formant2.frequency.setValueAtTime(780, now);
        formant2.Q.value = 5.0;

        vocalGain.gain.setValueAtTime(0.0, now);
        vocalGain.gain.linearRampToValueAtTime(0.3, now + 0.08);
        vocalGain.gain.exponentialRampToValueAtTime(0.001, now + 0.5);

        carrier.connect(formant1);
        carrier.connect(formant2);
        formant1.connect(vocalGain);
        formant2.connect(vocalGain);
        vocalGain.connect(this.masterGain);

        carrier.start(now);
        mod.start(now);
        carrier.stop(now + 0.55);
        mod.stop(now + 0.55);
      }

      update(currentSpeed, currentCrankAngle) {
        if (!this.initialized || this.ctx.state !== 'running') return;

        const normalizedSpeed = Math.min(currentSpeed / 10.0, 1.0);
        if (this.windGain && this.windFilter) {
          const targetGain = this.isMuted ? 0 : normalizedSpeed * 0.22;
          this.windGain.gain.setTargetAtTime(targetGain, this.ctx.currentTime, 0.05);
          this.windFilter.frequency.setTargetAtTime(160 + normalizedSpeed * 850, this.ctx.currentTime, 0.05);
        }

        // Chain clicks at 48 teeth / revolution
        if (currentSpeed > 0.05 && !this.isMuted) {
          const toothStep = (Math.PI * 2) / PHYSICS.CHAINRING_TEETH;
          const deltaAngle = currentCrankAngle - this.lastClickCrankAngle;
          if (deltaAngle >= toothStep) {
            this.playChainClick(Math.min(1.0, 0.4 + normalizedSpeed * 0.6));
            this.lastClickCrankAngle = currentCrankAngle;
          }
        }

        // Occasional natural pelican vocalization
        const now = performance.now();
        if (currentSpeed > 0.5 && !this.isMuted && now - this.lastPelicanCallTime > 14000 && Math.random() < 0.006) {
          this.playPelicanCall();
          this.lastPelicanCallTime = now;
        }
      }

    /* =========================================================
       3. Avian 2-Bone Inverse Kinematics (IK) Solver
       ========================================================= */
    function solveAvianLegIK(hipX, hipY, targetX, targetY, L1, L2) {
      const dx = targetX - hipX;
      const dy = targetY - hipY;
      let dist = Math.sqrt(dx * dx + dy * dy);

      // Clamp distance to avoid singularity
      const maxDist = L1 + L2 - 0.001;
      const minDist = Math.abs(L1 - L2) + 0.001;
      dist = Math.max(minDist, Math.min(maxDist, dist));

      // Law of Cosines
      const cosAlpha = (L1 * L1 + dist * dist - L2 * L2) / (2 * L1 * dist);
      const alpha = Math.acos(Math.max(-1, Math.min(1, cosAlpha)));
      const baseAngle = Math.atan2(dy, dx);

      // Avian ankle bends backward (posteriorly, towards -X)
      const bone1Angle = baseAngle - alpha;
      const ankleX = hipX + L1 * Math.cos(bone1Angle);
      const ankleY = hipY + L1 * Math.sin(bone1Angle);

      return {
        ankleX,
        ankleY,
        footX: targetX,
        footY: targetY
      };
    }

      update(state) {
        // 1. Wheel & Crank Rotations
        if (this.rearWheelGroup) this.rearWheelGroup.rotation.z = -state.wheelAngle;
        if (this.frontWheelGroup) this.frontWheelGroup.rotation.z = -state.wheelAngle;
        if (this.chainringMesh) this.chainringMesh.rotation.y = -state.crankAngle;

        // 2. Cranks & Horizontal Level Pedals
        const bb = PHYSICS.BB_POS;
        const cLen = PHYSICS.CRANK_LENGTH;

        const rightPedalPos = new THREE.Vector3(
          bb.x + cLen * Math.cos(state.crankAngle),
          bb.y + cLen * Math.sin(state.crankAngle),
          0.10
        );
        const leftPedalPos = new THREE.Vector3(
          bb.x - cLen * Math.cos(state.crankAngle),
          bb.y - cLen * Math.sin(state.crankAngle),
          -0.10
        );

        if (this.crankRightMesh) {
          this.crankRightMesh.rotation.z = state.crankAngle;
        }
        if (this.crankLeftMesh) {
          this.crankLeftMesh.rotation.z = state.crankAngle + Math.PI;
        }

        // Pedals stay strictly horizontal relative to ground
        if (this.pedalRightGroup) {
          this.pedalRightGroup.position.copy(rightPedalPos);
          this.pedalRightGroup.rotation.set(0, 0, 0);
        }
        if (this.pedalLeftGroup) {
          this.pedalLeftGroup.position.copy(leftPedalPos);
          this.pedalLeftGroup.rotation.set(0, 0, 0);
        }

        // 3. Infinite Scrolling Road Texture
        if (this.roadTexture) {
          this.roadTexture.offset.x = -(state.distance / 12.0) % 1.0;
        }

        // 4. Pelican Torso Cadence Bobbing
        const bobY = Math.sin(state.crankAngle * 2) * 0.016;
        if (this.pelicanGroup) {
          this.pelicanGroup.position.y = bobY;
          this.pelicanGroup.rotation.z = Math.sin(state.crankAngle) * 0.02;
        }

        // Gular pouch oscillation
        if (this.pouchMesh) {
          const pouchScaleY = 0.7 + Math.sin(state.crankAngle * 2 - 0.5) * 0.15;
          this.pouchMesh.scale.set(1.8, pouchScaleY, 0.5);
        }

        // 5. 3D Avian Leg Inverse Kinematics Update
        const update3DLegIK = (legObj, targetPos, zHipOffset) => {
          const hipPos = new THREE.Vector3(PHYSICS.SADDLE_POS.x + 0.02, PHYSICS.SADDLE_POS.y + 0.04 + bobY, zHipOffset);
          const ik = solveAvianLegIK(hipPos.x, hipPos.y, targetPos.x, targetPos.y, PHYSICS.LEG_L1, PHYSICS.LEG_L2);

          const anklePos = new THREE.Vector3(ik.ankleX, ik.ankleY, (hipPos.z + targetPos.z) * 0.5);

          // Position & Orient Thigh
          const thighMid = hipPos.clone().add(anklePos).multiplyScalar(0.5);
          legObj.thighMesh.position.copy(thighMid);
          legObj.thighMesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), anklePos.clone().sub(hipPos).normalize());

          // Position & Orient Tarsus (Shank)
          const tarsusMid = anklePos.clone().add(targetPos).multiplyScalar(0.5);
          legObj.tarsusMesh.position.copy(tarsusMid);
          legObj.tarsusMesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), targetPos.clone().sub(anklePos).normalize());

          // Foot strictly resting on horizontal pedal platform (footToPedal = 0)
          legObj.footMesh.position.set(targetPos.x + 0.02, targetPos.y + 0.01, targetPos.z);
          legObj.footMesh.rotation.set(0, 0, -Math.PI / 2);
        };

        update3DLegIK(this.legRightMeshes, rightPedalPos, 0.07);
        update3DLegIK(this.legLeftMeshes, leftPedalPos, -0.07);

        // OrbitControls update & Render
        this.controls.update();
        this.renderer.render(this.scene, this.camera);
      }

      onResize() {
        if (!this.renderer || !this.camera) return;
        this.camera.aspect = window.innerWidth / window.innerHeight;
        this.camera.updateProjectionMatrix();
        this.renderer.setSize(window.innerWidth, window.innerHeight);
      }

    let lastTime = performance.now();

    function animate(now) {
      requestAnimationFrame(animate);

      const dt = Math.min((now - lastTime) / 1000, 0.1);
      lastTime = now;

      // FPS Monitoring
      frameCount++;
      if (now - fpsLastSample >= 500) {
        state.fps = Math.round((frameCount * 1000) / (now - fpsLastSample));
        statFPS.textContent = state.fps.toString();

      // Physics Integration (No-slip wheel rolling & Gear train)
      if (state.speed > 0) {
        const deltaDist = state.speed * dt;
        state.distance += deltaDist;

        const deltaWheelAngle = deltaDist / PHYSICS.WHEEL_RADIUS;
        const deltaCrankAngle = deltaWheelAngle * PHYSICS.GEAR_RATIO;

        // Monotonic accumulation without modulo
        state.wheelAngle += deltaWheelAngle;
        state.crankAngle += deltaCrankAngle;

      svgController.update(state);
      threeController.update(state);
      audioEngine.update(state.speed, state.crankAngle, dt);
    }
