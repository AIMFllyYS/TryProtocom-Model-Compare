### 建模与物理实现思路

本项目以美洲白鹈鹕（*Pelecanus erythrorhynchos*）为原型，通过两段式逆运动学（IK）解算鸟类骨骼链，踝关节向后凸起，四趾全蹼足贴合水平踏板。物理机制遵循车轮纯滚动无打滑（$\omega_{wheel} = v / r$）与齿轮传动比约束（$\omega_{crank} = \omega_{wheel} \times 16 / 48$），两模共享单调累加相位，实现毫秒级平滑无缝切换。音频全由 Web Audio API 实时合成。

    /**
     * 计算鸟类踝关节(向后凸起)的坐标与脚爪贴合
     * @param {Object} H 髋关节 {x, y}
     * @param {Object} P 脚踏中心目标 {x, y}
     * @param {number} l1 大腿段长度
     * @param {number} l2 小腿段长度
     * @returns {Object} 关节位置与误差 { ankle: {x, y}, foot: {x, y}, footDist: number }
     */
    function solveAvianIK2D(H, P, l1, l2) {
      const dx = P.x - H.x;
      const dy = P.y - H.y;
      const d = Math.max(Math.abs(l1 - l2) + 0.002, Math.min(l1 + l2 - 0.002, Math.hypot(dx, dy)));
      
      const cosAlpha = (l1 * l1 + d * d - l2 * l2) / (2 * l1 * d);
      const alpha = Math.acos(Math.max(-1, Math.min(1, cosAlpha)));
      const phi = Math.atan2(dy, dx);
      
      // 鸟类踝关节向后凸起 (Intertarsal joint bends backward towards rear of bike)
      const ankleAngle = phi - alpha;
      const Ax = H.x + l1 * Math.cos(ankleAngle);
      const Ay = H.y + l1 * Math.sin(ankleAngle);

      // 脚爪严格贴合目标踏板
      const footDist = Math.hypot(P.x - (Ax + l2 * Math.cos(Math.atan2(P.y - Ay, P.x - Ax))),
                                  P.y - (Ay + l2 * Math.sin(Math.atan2(P.y - Ay, P.x - Ax))));
      return {
        ankle: { x: Ax, y: Ay },
        foot: { x: P.x, y: P.y },
        footDist: footDist
      };
    }

    // 程序化生成道路贴图 (无需外部图片资源)
    function createProceduralRoad3D() {
      const roadCanvas = document.createElement('canvas');
      roadCanvas.width = 512;
      roadCanvas.height = 1024;
      const ctx = roadCanvas.getContext('2d');

      // 沥青底色
      ctx.fillStyle = '#1e293b';
      ctx.fillRect(0, 0, 512, 1024);
      // 路面颗粒杂色
      for (let i = 0; i < 8000; i++) {
        ctx.fillStyle = Math.random() > 0.5 ? '#334155' : '#0f172a';
        ctx.fillRect(Math.random() * 512, Math.random() * 1024, 2, 2);
      }
      // 两侧白线
      ctx.fillStyle = '#f8fafc';
      ctx.fillRect(40, 0, 16, 1024);
      ctx.fillRect(456, 0, 16, 1024);
      // 中间虚线 (沿 Y 轴平铺)
      for (let y = 0; y < 1024; y += 128) {
        ctx.fillRect(246, y + 24, 20, 80);
      }

      roadTexture3D = new THREE.CanvasTexture(roadCanvas);
      roadTexture3D.wrapS = THREE.RepeatWrapping;
      roadTexture3D.wrapT = THREE.RepeatWrapping;
      roadTexture3D.repeat.set(1, 10);

      const roadGeo = new THREE.PlaneGeometry(3.6, 60);
      const roadMat = new THREE.MeshStandardMaterial({
        map: roadTexture3D,
        roughness: 0.85,
        metalness: 0.1
      });
      roadMesh3D = new THREE.Mesh(roadGeo, roadMat);
      roadMesh3D.rotation.x = -Math.PI / 2;
      roadMesh3D.position.set(0, 0, 0);
      roadMesh3D.receiveShadow = true;
      scene3D.add(roadMesh3D);

      // 道路两侧草地 / 树木
      const grassGeo = new THREE.PlaneGeometry(40, 60);
      const grassMat = new THREE.MeshStandardMaterial({ color: 0x0284c7, roughness: 0.9 });
      const grass = new THREE.Mesh(grassGeo, grassMat);
      grass.rotation.x = -Math.PI / 2;
      grass.position.set(0, -0.01, 0);
      grass.receiveShadow = true;
      scene3D.add(grass);

      // 两侧循环路标/树柱
      for (let i = 0; i < 14; i++) {
        const side = i % 2 === 0 ? 1 : -1;
        const prop = new THREE.Group();
        const postGeo = new THREE.CylinderGeometry(0.04, 0.04, 1.2, 8);
        const postMat = new THREE.MeshStandardMaterial({ color: 0x78350f, roughness: 0.7 });
        const post = new THREE.Mesh(postGeo, postMat);
        post.position.y = 0.6;
        post.castShadow = true;
        prop.add(post);

        const bushGeo = new THREE.ConeGeometry(0.35, 0.8, 8);
        const bushMat = new THREE.MeshStandardMaterial({ color: 0x16a34a, roughness: 0.6 });
        const bush = new THREE.Mesh(bushGeo, bushMat);
        bush.position.y = 1.3;
        bush.castShadow = true;
        prop.add(bush);

        prop.position.set(side * 2.1, 0, (i - 7) * 4);
        scene3D.add(prop);
        roadsideProps3D.push(prop);
      }
    }

    // 创建 3D 自行车模型
    function createBicycle3D() {
      bikeGroup3D = new THREE.Group();
      scene3D.add(bikeGroup3D);

      const frameMat = new THREE.MeshStandardMaterial({
        color: 0x0891b2,
        metalness: 0.75,
        roughness: 0.25
      });
      const chromeMat = new THREE.MeshStandardMaterial({
        color: 0xe2e8f0,
        metalness: 0.9,
        roughness: 0.15
      });
      const blackRubberMat = new THREE.MeshStandardMaterial({
        color: 0x1e293b,
        roughness: 0.85
      });
      const leatherMat = new THREE.MeshStandardMaterial({
        color: 0x78350f,
        roughness: 0.7
      });

      // 辅助管件创建函数
      function addTube(p1, p2, radius, mat = frameMat) {
        const v1 = new THREE.Vector3(p1.x, p1.y, p1.z || 0);
        const v2 = new THREE.Vector3(p2.x, p2.y, p2.z || 0);
        const dist = v1.distanceTo(v2);
        const cyl = new THREE.CylinderGeometry(radius, radius, dist, 16);
        const mesh = new THREE.Mesh(cyl, mat);
        mesh.castShadow = true;
        mesh.position.copy(v1.clone().add(v2).multiplyScalar(0.5));
        mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), v2.clone().sub(v1).normalize());
        bikeGroup3D.add(mesh);
        return mesh;
      }

      // 车架五通 Shell
      const bbShell = new THREE.Mesh(new THREE.CylinderGeometry(0.03, 0.03, 0.16, 16), chromeMat);
      bbShell.rotation.x = Math.PI / 2;
      bbShell.position.set(POS_BB.x, POS_BB.y, 0);
      bikeGroup3D.add(bbShell);

      // 车架主三角几何管 (SI 单位米)
      const pBB = { x: POS_BB.x, y: POS_BB.y, z: 0 };
      const pSaddleJoint = { x: -0.20, y: 0.62, z: 0 };
      const pHeadTop = { x: 0.38, y: 0.74, z: 0 };
      const pHeadBottom = { x: 0.36, y: 0.65, z: 0 };
      const pRearAxle = { x: -WHEELBASE / 2, y: WHEEL_RADIUS, z: 0 };
      const pFrontAxle = { x: WHEELBASE / 2, y: WHEEL_RADIUS, z: 0 };

      // 下管 (Down Tube)
      addTube(pBB, pHeadBottom, 0.022);
      // 上管 (Top Tube)
      addTube(pSaddleJoint, pHeadTop, 0.018);
      // 立管 (Seat Tube)
      addTube(pBB, pSaddleJoint, 0.020);
      // 头管 (Head Tube)
      addTube(pHeadBottom, pHeadTop, 0.025);

      // 后下叉 (Chain Stays, 双侧)
      addTube(pBB, { x: pRearAxle.x, y: pRearAxle.y, z: 0.055 }, 0.012);
      addTube(pBB, { x: pRearAxle.x, y: pRearAxle.y, z: -0.055 }, 0.012);
      // 后上叉 (Seat Stays, 双侧)
      addTube(pSaddleJoint, { x: pRearAxle.x, y: pRearAxle.y, z: 0.055 }, 0.012);
      addTube(pSaddleJoint, { x: pRearAxle.x, y: pRearAxle.y, z: -0.055 }, 0.012);

      // 前叉 (Fork, 双侧)
      addTube(pHeadBottom, { x: pFrontAxle.x, y: pFrontAxle.y, z: 0.05 }, 0.014);
      addTube(pHeadBottom, { x: pFrontAxle.x, y: pFrontAxle.y, z: -0.05 }, 0.014);

      // 车把组 (Stem & Handlebars)
      const pStem = { x: 0.41, y: 0.78, z: 0 };
      addTube(pHeadTop, pStem, 0.016, chromeMat);
      // 横把
      const barGeo = new THREE.CylinderGeometry(0.012, 0.012, 0.44, 16);
      const barMesh = new THREE.Mesh(barGeo, chromeMat);
      barMesh.rotation.x = Math.PI / 2;
      barMesh.position.set(pStem.x, pStem.y, 0);
      bikeGroup3D.add(barMesh);
      // 把套 (Grips)
      const gripGeo = new THREE.CylinderGeometry(0.016, 0.016, 0.10, 16);
      const gripR = new THREE.Mesh(gripGeo, leatherMat);
      gripR.rotation.x = Math.PI / 2;
      gripR.position.set(pStem.x, pStem.y, 0.17);
      const gripL = new THREE.Mesh(gripGeo, leatherMat);
      gripL.rotation.x = Math.PI / 2;
      gripL.position.set(pStem.x, pStem.y, -0.17);
      bikeGroup3D.add(gripR, gripL);

      // 车座 (Saddle)
      addTube(pSaddleJoint, { x: POS_SADDLE.x, y: POS_SADDLE.y - 0.03, z: 0 }, 0.014, chromeMat);
      const saddleGeo = new THREE.ConeGeometry(0.09, 0.24, 5);
      const saddleMesh = new THREE.Mesh(saddleGeo, leatherMat);
      saddleMesh.rotation.x = Math.PI / 2;
      saddleMesh.rotation.y = Math.PI;
      saddleMesh.scale.set(1.4, 1, 0.4);
      saddleMesh.position.set(POS_SADDLE.x, POS_SADDLE.y, 0);
      saddleMesh.castShadow = true;
      bikeGroup3D.add(saddleMesh);

      // 牙盘 (48T) 与 飞轮 (16T)
      const chainringGeo = new THREE.CylinderGeometry(0.09, 0.09, 0.005, 32);
      const chainringMesh = new THREE.Mesh(chainringGeo, chromeMat);
      chainringMesh.rotation.x = Math.PI / 2;
      chainringMesh.position.set(POS_BB.x, POS_BB.y, 0.08);
      bikeGroup3D.add(chainringMesh);

      // 车轮构造器 (胎 + 圈 + 辐条)
      function build3DWheel() {
        const wheelGroup = new THREE.Group();
        // 轮胎
        const tireGeo = new THREE.TorusGeometry(WHEEL_RADIUS, 0.022, 16, 64);
        const tireMesh = new THREE.Mesh(tireGeo, blackRubberMat);
        tireMesh.castShadow = true;
        wheelGroup.add(tireMesh);

        // 车圈
        const rimGeo = new THREE.TorusGeometry(WHEEL_RADIUS - 0.02, 0.010, 12, 64);
        const rimMesh = new THREE.Mesh(rimGeo, chromeMat);
        wheelGroup.add(rimMesh);

        // 花鼓
        const hubGeo = new THREE.CylinderGeometry(0.025, 0.025, 0.09, 16);
        const hubMesh = new THREE.Mesh(hubGeo, chromeMat);
        hubMesh.rotation.x = Math.PI / 2;
        wheelGroup.add(hubMesh);

        // 16 根放射辐条
        const spokeMat = new THREE.MeshStandardMaterial({ color: 0xcccccc, metalness: 0.8 });
        for (let i = 0; i < 16; i++) {
          const ang = (i / 16) * Math.PI * 2;
          const spokeGeo = new THREE.CylinderGeometry(0.002, 0.002, WHEEL_RADIUS * 2 - 0.04, 6);
          const spoke = new THREE.Mesh(spokeGeo, spokeMat);
          spoke.rotation.z = ang;
          wheelGroup.add(spoke);
        }
        return wheelGroup;
      }

      // 后轮
      rearWheelGroup3D = build3DWheel();
      rearWheelGroup3D.position.set(pRearAxle.x, pRearAxle.y, 0);
      bikeGroup3D.add(rearWheelGroup3D);

      // 前轮
      frontWheelGroup3D = build3DWheel();
      frontWheelGroup3D.position.set(pFrontAxle.x, pFrontAxle.y, 0);
      bikeGroup3D.add(frontWheelGroup3D);

      // 右侧曲柄与踏板 (前景 Z > 0)
      rightCrankGroup3D = new THREE.Group();
      rightCrankGroup3D.position.set(POS_BB.x, POS_BB.y, 0.10);
      const crankArmGeo = new THREE.BoxGeometry(0.02, CRANK_LENGTH, 0.012);
      crankArmGeo.translate(0, -CRANK_LENGTH / 2, 0);
      const rightCrankArm = new THREE.Mesh(crankArmGeo, chromeMat);
      rightCrankArm.castShadow = true;
      rightCrankGroup3D.add(rightCrankArm);

      // 踏板平台 (保持水平，不随曲柄转动)
      const pedalGeo = new THREE.BoxGeometry(0.09, 0.018, 0.08);
      rightPedalMesh3D = new THREE.Mesh(pedalGeo, blackRubberMat);
      rightPedalMesh3D.position.set(0, -CRANK_LENGTH, 0.04);
      rightPedalMesh3D.castShadow = true;
      rightCrankGroup3D.add(rightPedalMesh3D);
      bikeGroup3D.add(rightCrankGroup3D);

      // 左侧曲柄与踏板 (背景 Z < 0, 相位差 PI)
      leftCrankGroup3D = new THREE.Group();
      leftCrankGroup3D.position.set(POS_BB.x, POS_BB.y, -0.10);
      const leftCrankArm = new THREE.Mesh(crankArmGeo, chromeMat);
      leftCrankArm.castShadow = true;
      leftCrankGroup3D.add(leftCrankArm);

      leftPedalMesh3D = new THREE.Mesh(pedalGeo, blackRubberMat);
      leftPedalMesh3D.position.set(0, -CRANK_LENGTH, -0.04);
      leftPedalMesh3D.castShadow = true;
      leftCrankGroup3D.add(leftPedalMesh3D);
      bikeGroup3D.add(leftCrankGroup3D);
    }

    // 创建 3D 鹈鹕模型 (美洲白鹈鹕 Pelecanus erythrorhynchos)
    function createPelican3D() {
      pelicanGroup3D = new THREE.Group();
      scene3D.add(pelicanGroup3D);

      const whiteFeatherMat = new THREE.MeshStandardMaterial({
        color: 0xffffff,
        roughness: 0.65,
        metalness: 0.05
      });
      const blackFeatherMat = new THREE.MeshStandardMaterial({
        color: 0x1e293b,
        roughness: 0.75,
        metalness: 0.05
      });
      const beakMat = new THREE.MeshStandardMaterial({
        color: 0xf97316,
        roughness: 0.35,
        metalness: 0.1
      });
      const pouchMat = new THREE.MeshStandardMaterial({
        color: 0xfb923c,
        roughness: 0.5,
        transparent: true,
        opacity: 0.92
      });
      const legMat = new THREE.MeshStandardMaterial({
        color: 0xea580c,
        roughness: 0.45
      });

      // 1. 身体 / 躯干 (Torso)
      const bodyGeo = new THREE.SphereGeometry(0.24, 32, 16);
      bodyGeo.scale(1.4, 0.9, 0.85);

      pelicanGroup3D.add(bodyMesh);

      const tailMesh = new THREE.Mesh(tailGeo, whiteFeatherMat);
      tailMesh.rotation.z = 1.3;
      tailMesh.position.set(POS_HIP.x - 0.18, POS_HIP.y + 0.04, 0);
      pelicanGroup3D.add(tailMesh);

      // 2. 双侧折叠翅膀 (分层：上层白色覆羽 + 下层黑色飞羽)
      function addWing(zSide) {
        const wingGroup = new THREE.Group();
        // 黑色初级/次级飞羽层 (Primaries & Secondaries)
        const blackWingGeo = new THREE.ConeGeometry(0.15, 0.55, 6);
        blackWingGeo.scale(0.8, 1, 0.3);

        blackWing.rotation.z = 1.35;
        blackWing.position.set(-0.12, -0.05, 0);
        wingGroup.add(blackWing);

        const whiteWingGeo = new THREE.SphereGeometry(0.20, 16, 12);
        whiteWingGeo.scale(1.2, 0.6, 0.4);
        const whiteWing = new THREE.Mesh(whiteWingGeo, whiteFeatherMat);
        whiteWing.position.set(0.04, 0.02, 0.02 * zSide);
        wingGroup.add(whiteWing);

        // 前翅握把腕关节 (延伸至把手)
        const armGeo = new THREE.CylinderGeometry(0.04, 0.035, 0.32, 12);
        const armMesh = new THREE.Mesh(armGeo, whiteFeatherMat);
        armMesh.rotation.z = -1.1;
        armMesh.rotation.y = 0.25 * zSide;
        armMesh.position.set(0.24, 0.02, 0.05 * zSide);
        wingGroup.add(armMesh);

        wingGroup.position.set(POS_HIP.x + 0.12, POS_HIP.y + 0.06, 0.18 * zSide);
        pelicanGroup3D.add(wingGroup);
      }
      addWing(1);   // 右翅
      addWing(-1);  // 左翅

      // 3. S 形颈部 (S-curved Neck)

        new THREE.Vector3(POS_HIP.x + 0.26, POS_HIP.y + 0.06, 0),
        new THREE.Vector3(POS_HIP.x + 0.38, POS_HIP.y + 0.15, 0),
        new THREE.Vector3(POS_HIP.x + 0.36, POS_HIP.y + 0.28, 0),

      ]);
      const neckGeo = new THREE.TubeGeometry(neckCurve, 20, 0.065, 12, false);

      neckMesh.castShadow = true;
      pelicanGroup3D.add(neckMesh);

      headGeo.scale(1.2, 1, 0.85);
      const headMesh = new THREE.Mesh(headGeo, whiteFeatherMat);

      pelicanGroup3D.add(headGroup);

      // 7. 两段式腿部组 (左右各一组，包含股骨段、踝关节、跗跖骨段、全蹼足)
      // 注意：组直接位于世界原点，方便进行全局三维 IK 坐标直接映射
      function build3DLegGroup(zSide) {
        const legGroup = new THREE.Group();

        thighMesh.castShadow = true;
        legGroup.add(thighMesh);

        // 踝关节球 (Intertarsal joint)
        const ankleGeo = new THREE.SphereGeometry(0.028, 12, 12);
        const ankleMesh = new THREE.Mesh(ankleGeo, legMat);
        legGroup.add(ankleMesh);

        // 小腿段网格 (跗跖骨)
        const shankGeo = new THREE.CylinderGeometry(0.024, 0.020, LEG_L2, 12);
        const shankMesh = new THREE.Mesh(shankGeo, legMat);
        shankMesh.castShadow = true;
        legGroup.add(shankMesh);

        // 全蹼足 (Totipalmate Foot: 4 趾紧密相连)
        const footGroup = new THREE.Group();
        const footPlateGeo = new THREE.BoxGeometry(0.10, 0.015, 0.08);
        const footPlate = new THREE.Mesh(footPlateGeo, legMat);
        footGroup.add(footPlate);

        // 4 趾前伸骨节
        for (let t = 0; t < 4; t++) {
          const toeGeo = new THREE.CylinderGeometry(0.005, 0.003, 0.075, 8);
          const toe = new THREE.Mesh(toeGeo, legMat);
          toe.rotation.x = Math.PI / 2;
          toe.position.set(0.04, 0, (t - 1.5) * 0.022);
          footGroup.add(toe);
        }
        legGroup.add(footGroup);

        scene3D.add(legGroup);
        return {
          group: legGroup,
          thigh: thighMesh,
          ankle: ankleMesh,
          shank: shankMesh,
          foot: footGroup,
          z: 0.14 * zSide
        };
      }

      legRightGroup3D = build3DLegGroup(1);   // 右腿 (Z > 0)
      legLeftGroup3D = build3DLegGroup(-1);   // 左腿 (Z < 0)
    }

    // 更新 3D 运动学与部件姿态
    function updateThree(dt) {
      if (!scene3D || !renderer3D) return;

      // 1. 地面道路贴图滚动 (前进物理感)
      if (roadTexture3D) {
        roadTexture3D.offset.y -= (currentSpeed * dt) / 6.0;
      }

      // 两侧景物后退
      for (const prop of roadsideProps3D) {
        prop.position.z -= currentSpeed * dt;
        if (prop.position.z < -25) {
          prop.position.z += 50;
        }
      }

      // 2. 车轮旋转 (无打滑，角速度 = v / r)
      if (rearWheelGroup3D) rearWheelGroup3D.rotation.z = -wheelAngle;
      if (frontWheelGroup3D) frontWheelGroup3D.rotation.z = -wheelAngle;

      // 3. 曲柄旋转 (顺时针旋转，推动自行车前进)
      if (rightCrankGroup3D) rightCrankGroup3D.rotation.z = -crankAngle;
      if (leftCrankGroup3D) leftCrankGroup3D.rotation.z = -(crankAngle + Math.PI);

      // 4. 踏板反向自转以保持水平 (不随曲柄一起倾斜)
      if (rightPedalMesh3D) rightPedalMesh3D.rotation.z = crankAngle;
      if (leftPedalMesh3D) leftPedalMesh3D.rotation.z = (crankAngle + Math.PI);

      // 更新曲柄与踏板的世界矩阵
      rightCrankGroup3D.updateMatrixWorld(true);
      leftCrankGroup3D.updateMatrixWorld(true);

      const pPedalWorldR = new THREE.Vector3();
      rightPedalMesh3D.getWorldPosition(pPedalWorldR);

      const pPedalWorldL = new THREE.Vector3();
      leftPedalMesh3D.getWorldPosition(pPedalWorldL);

      // 5. 3D 腿部两段式逆运动学更新 (脚爪紧紧咬合踏板平台)
      update3DLeg(legRightGroup3D, pPedalWorldR);
      update3DLeg(legLeftGroup3D, pPedalWorldL);

      // 6. 3D 喉囊惯性摆动
      if (pouchMesh3D) {
        const sway = Math.sin(crankAngle * 2) * 0.02 * Math.min(1.0, currentSpeed / 2.0);
        pouchMesh3D.position.y = -0.06 + sway;
        pouchMesh3D.scale.set(2.2, 0.9 + sway * 8, 0.65);
      }

      // 7. 更新 OrbitControls 并渲染
      if (orbitControls) orbitControls.update();
      renderer3D.render(scene3D, camera3D);
    }

    /**
     * 3D 腿部逆运动学姿态计算 (大腿、小腿与全蹼足无缝连接)
     */
    function update3DLeg(legData, pPedal) {
      const pHip = new THREE.Vector3(POS_HIP.x, POS_HIP.y, pPedal.z);

      // 2-Link Avian IK (踝关节向后凸起)
      const ik = solveAvianIK2D({ x: pHip.x, y: pHip.y }, { x: pPedal.x, y: pPedal.y }, LEG_L1, LEG_L2);
      const pAnkle = new THREE.Vector3(ik.ankle.x, ik.ankle.y, pPedal.z);

      // 定位大腿段 (Hip -> Ankle)
      const vThigh = new THREE.Vector3(pAnkle.x - pHip.x, pAnkle.y - pHip.y, 0);
      legData.thigh.position.set((pHip.x + pAnkle.x) * 0.5, (pHip.y + pAnkle.y) * 0.5, pPedal.z);
      legData.thigh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), vThigh.clone().normalize());

      // 定位踝关节球
      legData.ankle.position.set(pAnkle.x, pAnkle.y, pPedal.z);

      // 定位小腿段 (Ankle -> Foot)
      const vShank = new THREE.Vector3(pPedal.x - pAnkle.x, pPedal.y - pAnkle.y, 0);
      legData.shank.position.set((pAnkle.x + pPedal.x) * 0.5, (pAnkle.y + pPedal.y) * 0.5, pPedal.z);
      legData.shank.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), vShank.clone().normalize());

      // 定位全蹼足 (贴合踏板平台且保持水平)
      legData.foot.position.set(pPedal.x, pPedal.y + 0.012, pPedal.z);
      legData.foot.rotation.set(0, 0, 0);
    }

    /* ==========================================================================
       7. 轻量级全功能 OrbitControls 实现 (保证零 CDN 依赖、无报错)
       ========================================================================== */
    function setupCustomOrbitControls(camera, domElement) {
      const target = new THREE.Vector3(0, 0.45, 0);
      let isDragging = false;
      let prevMouseX = 0;
      let prevMouseY = 0;
      let spherical = new THREE.Spherical().setFromVector3(camera.position.clone().sub(target));

      function onPointerDown(e) {
        isDragging = true;
        prevMouseX = e.clientX;
        prevMouseY = e.clientY;
      }
      function onPointerMove(e) {
        if (!isDragging) return;
        const deltaX = e.clientX - prevMouseX;
        const deltaY = e.clientY - prevMouseY;
        prevMouseX = e.clientX;
        prevMouseY = e.clientY;

        // 左键旋转
        if (e.buttons === 1 || e.buttons === 0) {
          spherical.theta -= deltaX * 0.006;
          spherical.phi -= deltaY * 0.006;
          spherical.phi = Math.max(0.1, Math.min(Math.PI / 2 - 0.02, spherical.phi)); // 防止穿入地下
        }
        // 右键平移
        else if (e.buttons === 2) {
          target.x -= deltaX * 0.002;
          target.y += deltaY * 0.002;
        }
      }
      function onPointerUp() { isDragging = false; }
      function onWheel(e) {
        e.preventDefault();
        spherical.radius += e.deltaY * 0.003;
        spherical.radius = Math.max(1.2, Math.min(8.0, spherical.radius));
      }

      domElement.addEventListener('pointerdown', onPointerDown);
      window.addEventListener('pointermove', onPointerMove);
      window.addEventListener('pointerup', onPointerUp);
      domElement.addEventListener('wheel', onWheel, { passive: false });
      domElement.addEventListener('contextmenu', (e) => e.preventDefault());

      orbitControls = {
        update: function() {
          camera.position.setFromSpherical(spherical).add(target);
          camera.lookAt(target);
        },
        reset: function() {
          target.set(0, 0.45, 0);
          camera.position.set(2.4, 1.3, 2.6);
          spherical.setFromVector3(camera.position.clone().sub(target));
        }
      };
    }

    function resetCamera3D() {
      if (orbitControls) orbitControls.reset();
    }

    /* ==========================================================================
       8. Web Audio API 实时音频合成 (全代码合成，无外部音频)
       ========================================================================== */
    let audioCtx = null;
    let masterGain = null;
    let windNoiseNode = null;
    let windGainNode = null;
    let nextClickTime = 0;

    function initAudio() {
      if (audioCtx) return;
      const AudioContextClass = window.AudioContext || window.webkitAudioContext;
      audioCtx = new AudioContextClass();

      masterGain = audioCtx.createGain();
      masterGain.gain.setValueAtTime(0.0, audioCtx.currentTime);
      masterGain.connect(audioCtx.destination);

      // 1. 胎噪与破风声 (白噪声经滤波)
      const bufferSize = audioCtx.sampleRate * 2;
      const noiseBuffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
      const output = noiseBuffer.getChannelData(0);
      for (let i = 0; i < bufferSize; i++) {
        output[i] = Math.random() * 2 - 1;
      }

      windNoiseNode = audioCtx.createBufferSource();
      windNoiseNode.buffer = noiseBuffer;
      windNoiseNode.loop = true;

      const windFilter = audioCtx.createBiquadFilter();
      windFilter.type = 'bandpass';
      windFilter.frequency.setValueAtTime(450, audioCtx.currentTime);
      windFilter.Q.setValueAtTime(1.5, audioCtx.currentTime);

      windGainNode = audioCtx.createGain();
      windGainNode.gain.setValueAtTime(0.0, audioCtx.currentTime);

      windNoiseNode.connect(windFilter);
      windFilter.connect(windGainNode);
      windGainNode.connect(masterGain);
      windNoiseNode.start();
    }

    function toggleAudio() {
      initAudio();
      if (audioCtx.state === 'suspended') {
        audioCtx.resume();
      }
      isAudioActive = !isAudioActive;
      const btn = document.getElementById('btn-audio');
      const icon = document.getElementById('audio-icon');
      const text = document.getElementById('audio-text');

        btn.classList.remove('active');
        icon.innerText = '🔇';

    }

    // 播放单个链条齿轮微小咔哒声
    function playChainClick() {
      if (!isAudioActive || !audioCtx) return;
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      const filter = audioCtx.createBiquadFilter();

      filter.type = 'bandpass';
      filter.frequency.setValueAtTime(3200 + Math.random() * 600, audioCtx.currentTime);
      filter.Q.setValueAtTime(4.0, audioCtx.currentTime);

      osc.type = 'triangle';
      osc.frequency.setValueAtTime(2400 + Math.random() * 400, audioCtx.currentTime);

      gain.gain.setValueAtTime(0.06, audioCtx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + 0.015);

      osc.connect(filter);
      filter.connect(gain);
      gain.connect(masterGain);

      osc.start();
      osc.stop(audioCtx.currentTime + 0.016);
    }

    // 合成低沉短促鹈鹕喉部鸣叫 (FM 调频共鸣)
    function triggerPelicanCall(userTriggered = false) {
      initAudio();
      if (audioCtx.state === 'suspended') audioCtx.resume();
      if (!isAudioActive && !userTriggered) return;

      const now = audioCtx.currentTime;
      const carrier = audioCtx.createOscillator();
      const modulator = audioCtx.createOscillator();
      const modGain = audioCtx.createGain();
      const callGain = audioCtx.createGain();
      const formantFilter = audioCtx.createBiquadFilter();

      // FM 载波与调制波参数
      carrier.type = 'sawtooth';
      carrier.frequency.setValueAtTime(140, now);
      carrier.frequency.exponentialRampToValueAtTime(95, now + 0.35);

      modulator.type = 'sine';
      modulator.frequency.setValueAtTime(38, now);
      modGain.gain.setValueAtTime(65, now);

      modulator.connect(carrier.frequency);

      // 共振峰滤波 (模拟大型鸟类深沉喉音)
      formantFilter.type = 'bandpass';
      formantFilter.frequency.setValueAtTime(520, now);
      formantFilter.Q.setValueAtTime(3.0, now);

      callGain.gain.setValueAtTime(0.001, now);
      callGain.gain.linearRampToValueAtTime(0.25, now + 0.05);
      callGain.gain.exponentialRampToValueAtTime(0.0001, now + 0.4);

      carrier.connect(formantFilter);
      formantFilter.connect(callGain);
      callGain.connect(masterGain);

      carrier.start(now);
      modulator.start(now);
      carrier.stop(now + 0.42);
      modulator.stop(now + 0.42);
    }

    // 周期性更新持续音效
    let callTimer = 0;
    function updateAudio(dt) {
      if (!isAudioActive || !audioCtx) return;

      // 1. 胎噪与风噪随速度渐变
      if (windGainNode) {
        const targetWindGain = Math.min(0.28, (currentSpeed / 10.0) * 0.25);
        windGainNode.gain.linearRampToValueAtTime(targetWindGain, audioCtx.currentTime + 0.05);
      }

      // 2. 链条齿轮咔哒声 (踏频与齿数对应)
      if (currentSpeed > 0.05) {
        // 每秒通过齿数 = 曲柄角速度 / (2PI) * 48
        const teethPerSec = (currentSpeed / WHEEL_RADIUS) * (COG_TEETH / CHAINRING_TEETH) * (CHAINRING_TEETH / (2 * Math.PI));
        const clickInterval = 1.0 / Math.max(1, teethPerSec);
        if (audioCtx.currentTime >= nextClickTime) {
          playChainClick();
          nextClickTime = audioCtx.currentTime + clickInterval;
        }
      }

      // 3. 偶发自动鹈鹕低鸣 (15-28 秒随机间隔)
      callTimer += dt;
      if (callTimer > 20 && Math.random() < 0.005) {
        callTimer = 0;
        triggerPelicanCall();
      }
    }

    // 帧率统计
    let frameCount = 0;
    let fpsTime = performance.now();

    // 全局主循环
    function mainLoop(timestamp) {
      const dt = Math.min((timestamp - lastTimestamp) / 1000, 0.1);
      lastTimestamp = timestamp;

      // 1. 物理累计转角 (严格遵守物理公式，不打滑，单调累加)
      if (currentSpeed > 0) {
        // 后轮无打滑纯滚动: dTheta_wheel = (v / r_wheel) * dt
        const dWheel = (currentSpeed / WHEEL_RADIUS) * dt;
        wheelAngle += dWheel;
        // 传动比匹配: 曲柄转角 = 车轮转角 × (飞轮齿数 / 牙盘齿数)
        const dCrank = dWheel * (COG_TEETH / CHAINRING_TEETH);
        crankAngle += dCrank;
      }

      // 2. 渲染当前模式 (后台模式亦保持状态同步)
      updateSvg(dt);
      updateThree(dt);
      updateAudio(dt);

      // 3. 统计 FPS
      frameCount++;
      if (timestamp - fpsTime >= 500) {
        const fps = Math.round((frameCount * 1000) / (timestamp - fpsTime));
        if (fpsDisplay) fpsDisplay.innerText = fps;
        frameCount = 0;
        fpsTime = timestamp;
      }

      requestAnimationFrame(mainLoop);
    }
