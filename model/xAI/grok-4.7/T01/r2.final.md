function solveKnee(hip, ankle, out) {
  const dx = ankle.x - hip.x;
  const dy = ankle.y - hip.y;
  const dz = ankle.z - hip.z;
  const D = Math.hypot(dx, dy, dz) || 1e-8;
  const maxR = FEMUR + TIBIA - 1e-4;
  const minR = Math.abs(TIBIA - FEMUR) + 1e-4;
  const dist = Math.min(maxR, Math.max(minR, D));
  const inv = 1 / D;
  const dirx = dx * inv, diry = dy * inv, dirz = dz * inv;
  let cosA = (FEMUR * FEMUR + dist * dist - TIBIA * TIBIA) / (2 * FEMUR * dist);
  if (cosA > 1) cosA = 1;
  if (cosA < -1) cosA = -1;
  const sinA = Math.sqrt(Math.max(0, 1 - cosA * cosA));
  let px = 1, py = -0.65, pz = 0;
  const dot = px * dirx + py * diry + pz * dirz;
  px -= dirx * dot;
  py -= diry * dot;
  pz -= dirz * dot;
  const pl = Math.hypot(px, py, pz) || 1;

  out.x = hip.x + dirx * FEMUR * cosA + px * k;
  out.y = hip.y + diry * FEMUR * cosA + py * k;
  out.z = hip.z + dirz * FEMUR * cosA + pz * k;
}

function addWing(parent, z, far) {
  const shoulder = [0.04, 1.04];
  const elbow = [0.28, 0.99];
  const wrist = [0.54, 0.985];
  const arm = Math.atan2(elbow[1] - shoulder[1], elbow[0] - shoulder[0]);
  const covert = far ? "#6a5e56" : "#9a9086";
  const secondary = far ? "#4a382f" : "#3a2c26";
  const primary = far ? "#2a211c" : "#1a1411";
  const op = far ? "0.45" : "1";
  const armPath = openPath([shoulder, elbow, wrist], z);
  S("path", {
    d: armPath, fill: "none",
    stroke: far ? "#5c463c" : "#3e2e28",
    "stroke-width": far ? "16" : "22",
    "stroke-linecap": "round", "stroke-linejoin": "round",
    opacity: op
  }, parent);
  for (let i = 0; i < 5; i++) {
    const t = 0.15 + (i / 4) * 0.75;
    const bx = shoulder[0] + (elbow[0] - shoulder[0]) * t;
    const by = shoulder[1] + (elbow[1] - shoulder[1]) * t;
    S("path", {
      d: featherPath([bx, by], arm + 0.62, far ? 0.07 : 0.09, 0.046, z),
      fill: covert, opacity: op
    }, parent);
  }
  for (let i = 0; i < 6; i++) {
    const t = 0.12 + (i / 5) * 0.8;
    const bx = elbow[0] + (wrist[0] - elbow[0]) * t;
    const by = elbow[1] + (wrist[1] - elbow[1]) * t;
    S("path", {
      d: featherPath([bx, by], -1.25 - i * 0.05, (far ? 0.11 : 0.15) - i * 0.008, 0.05, z),
      fill: secondary, opacity: op
    }, parent);
  }
  for (let i = 0; i < 7; i++) {
    S("path", {
      d: featherPath(wrist, Math.PI + 0.7 + i * 0.06, (far ? 0.13 : 0.2) - i * 0.012, 0.04, z),
      fill: primary, opacity: op
    }, parent);
  }
  if (!far) {
    const grip = proj(0.56, 0.98, z);
    S("circle", { cx: fmt(grip.x), cy: fmt(grip.y), r: "8", fill: "#241c18" }, parent);
  }
}

function chainScreenPath() {
  const c1 = { x: BB_X, y: BB_Y };
  const c2 = { x: REAR_X, y: WHEEL_RADIUS };
  const r1 = CHAINRING_R, r2 = COG_R;
  const dx = c2.x - c1.x, dy = c2.y - c1.y;
  const dist = Math.hypot(dx, dy);
  const gamma = Math.atan2(dy, dx);
  const phi = Math.acos(clamp((r1 - r2) / dist, -1, 1));
  function tang(sign) {
    const a = gamma + sign * phi;
    return {
      a: a,
      p1: { x: c1.x + Math.cos(a) * r1, y: c1.y + Math.sin(a) * r1 },
      p2: { x: c2.x + Math.cos(a) * r2, y: c2.y + Math.sin(a) * r2 }
    };
  }
  const tA = tang(1), tB = tang(-1);
  const upper = tA.p1.y >= tB.p1.y ? tA : tB;
  const lower = upper === tA ? tB : tA;
  const pts = [];
  function arc(cx, cy, r, a0, a1, ccw, steps) {
    let sweep = a1 - a0;
    if (ccw) {
      while (sweep <= 0) sweep += Math.PI * 2;
    } else {
      while (sweep >= 0) sweep -= Math.PI * 2;
    }
    for (let i = 0; i <= steps; i++) {
      const a = a0 + sweep * (i / steps);
      const x = cx + Math.cos(a) * r;
      const y = cy + Math.sin(a) * r;
      const prev = pts[pts.length - 1];
      if (!prev || Math.hypot(prev.x - x, prev.y - y) > 1e-4) pts.push({ x: x, y: y });
    }
  }
  function midX(cx, r, a0, a1, ccw) {
    let sweep = a1 - a0;
    if (ccw) while (sweep <= 0) sweep += Math.PI * 2;
    else while (sweep >= 0) sweep -= Math.PI * 2;
    const a = a0 + sweep * 0.5;
    return cx + Math.cos(a) * r;
  }
  function line(a, b, steps) {
    for (let i = 0; i <= steps; i++) {
      const t = i / steps;
      const x = a.x + (b.x - a.x) * t;
      const y = a.y + (b.y - a.y) * t;
      const prev = pts[pts.length - 1];
      if (!prev || Math.hypot(prev.x - x, prev.y - y) > 1e-4) pts.push({ x: x, y: y });
    }
  }
  const ringCCW = midX(c1.x, r1, upper.a, lower.a, true) > midX(c1.x, r1, upper.a, lower.a, false);
  arc(c1.x, c1.y, r1, upper.a, lower.a, ringCCW, 18);
  line(lower.p1, lower.p2, 8);
  const cogCCW = midX(c2.x, r2, lower.a, upper.a, true) < midX(c2.x, r2, lower.a, upper.a, false);
  arc(c2.x, c2.y, r2, lower.a, upper.a, cogCCW, 12);
  line(upper.p2, upper.p1, 8);
  let d = "";
  for (let i = 0; i < pts.length; i++) {
    const s = proj(pts[i].x, pts[i].y, -0.02);
    d += (i ? " L " : "M ") + fmt(s.x) + " " + fmt(s.y);
  }
  return d + " Z";
}

function limbPoly(a, b, wa, wb) {
  const dx = b.x - a.x, dy = b.y - a.y;
  const len = Math.hypot(dx, dy) || 1;
  const nx = -dy / len, ny = dx / len;
  return "M " + fmt(a.x + nx * wa) + " " + fmt(a.y + ny * wa) +
    " L " + fmt(b.x + nx * wb) + " " + fmt(b.y + ny * wb) +
    " L " + fmt(b.x - nx * wb) + " " + fmt(b.y - ny * wb) +
    " L " + fmt(a.x - nx * wa) + " " + fmt(a.y - ny * wa) + " Z";
}

  dirVec.set(b.x - a.x, b.y - a.y, b.z - a.z);

}

function sampleChain() {
  const c1 = { x: BB_X, y: BB_Y };
  const c2 = { x: REAR_X, y: WHEEL_RADIUS };
  const r1 = CHAINRING_R, r2 = COG_R;
  const dx = c2.x - c1.x, dy = c2.y - c1.y;
  const dist = Math.hypot(dx, dy);
  const gamma = Math.atan2(dy, dx);
  const phi = Math.acos(clamp((r1 - r2) / dist, -1, 1));
  const angs = [gamma + phi, gamma - phi];
  const ts = angs.map(function (a) {
    return { a: a, y1: c1.y + Math.sin(a) * r1 };
  });
  const upper = ts[0].y1 >= ts[1].y1 ? ts[0] : ts[1];
  const lower = upper === ts[0] ? ts[1] : ts[0];
  const pts = [];
  function arc(cx, cy, r, a0, a1, ccw, steps) {
    let sweep = a1 - a0;
    if (ccw) while (sweep <= 0) sweep += Math.PI * 2;
    else while (sweep >= 0) sweep -= Math.PI * 2;
    for (let i = 0; i <= steps; i++) {
      const a = a0 + sweep * (i / steps);
      const x = cx + Math.cos(a) * r;
      const y = cy + Math.sin(a) * r;
      const prev = pts[pts.length - 1];
      if (!prev || Math.hypot(prev.x - x, prev.y - y) > 1e-4) pts.push({ x: x, y: y });
    }
  }
  function midX(cx, r, a0, a1, ccw) {
    let sweep = a1 - a0;
    if (ccw) while (sweep <= 0) sweep += Math.PI * 2;
    else while (sweep >= 0) sweep -= Math.PI * 2;
    const a = a0 + sweep * 0.5;
    return cx + Math.cos(a) * r;
  }
  function line(a, b, steps) {
    for (let i = 0; i <= steps; i++) {
      const t = i / steps;
      const x = a.x + (b.x - a.x) * t;
      const y = a.y + (b.y - a.y) * t;
      const prev = pts[pts.length - 1];
      if (!prev || Math.hypot(prev.x - x, prev.y - y) > 1e-4) pts.push({ x: x, y: y });
    }
  }
  function onCirc(c, r, a) { return { x: c.x + Math.cos(a) * r, y: c.y + Math.sin(a) * r }; }
  const ringCCW = midX(c1.x, r1, upper.a, lower.a, true) > midX(c1.x, r1, upper.a, lower.a, false);
  arc(c1.x, c1.y, r1, upper.a, lower.a, ringCCW, 16);
  line(onCirc(c1, r1, lower.a), onCirc(c2, r2, lower.a), 8);
  const cogCCW = midX(c2.x, r2, lower.a, upper.a, true) < midX(c2.x, r2, lower.a, upper.a, false);
  arc(c2.x, c2.y, r2, lower.a, upper.a, cogCCW, 10);
  line(onCirc(c2, r2, upper.a), onCirc(c1, r1, upper.a), 8);
  return pts;
}

function featherGeo(len, w) {
  const g = new THREE.BufferGeometry();
  const pos = new Float32Array([
    0, w * 0.45, 0,
    0, -w * 0.4, 0,
    len * 0.62, w * 0.62, 0.003,
    len, 0, 0.001,
    len * 0.58, -w * 0.5, 0.003
  ]);
  g.setAttribute("position", new THREE.BufferAttribute(pos, 3));
  g.setIndex([0, 2, 1, 1, 2, 4, 2, 3, 4]);
  g.computeVertexNormals();
  return g;
}

function addFeather(parent, x, y, z, dx, dy, dz, len, w, mat) {
  const m = new THREE.Mesh(featherGeo(len, w), mat);
  m.position.set(x, y, z);
  dirVec.set(dx, dy, dz);
  if (dirVec.lengthSq() < 1e-8) dirVec.set(1, 0, 0);
  dirVec.normalize();
  m.quaternion.setFromUnitVectors(xAxis, dirVec);
  parent.add(m);
}

function buildWing3(sign, covertMat, secMat, priMat, boneMat) {
  const group = new THREE.Group();
  const sx = 0.04, sy = 1.04, sz = 0.14 * sign;
  group.position.set(sx, sy, sz);
  const elbow = { x: 0.28 - sx, y: 0.99 - sy, z: 0.22 * sign - sz };
  const wrist = { x: 0.56 - sx, y: 0.98 - sy, z: 0.32 * sign - sz };
  const bone = new THREE.CylinderGeometry(1, 1, 1, 8);
  const hum = new THREE.Mesh(bone, boneMat);
  const fore = new THREE.Mesh(bone, boneMat);
  placeBone(hum, { x: 0, y: 0, z: 0 }, elbow, 0.028);
  placeBone(fore, elbow, wrist, 0.022);
  group.add(hum); group.add(fore);
  const membrane = new THREE.Shape();
  membrane.moveTo(0.02, 0.04);
  membrane.lineTo(elbow.x, elbow.y + 0.06);
  membrane.lineTo(wrist.x * 0.95, wrist.y + 0.03);
  membrane.lineTo(wrist.x * 0.72, wrist.y - 0.14);
  membrane.lineTo(elbow.x * 0.8, elbow.y - 0.11);
  membrane.lineTo(0.04, -0.04);
  membrane.closePath();
  const mem = new THREE.Mesh(new THREE.ExtrudeGeometry(membrane, { depth: 0.018, bevelEnabled: false }), secMat);
  mem.position.z = -0.01;
  mem.castShadow = true;
  group.add(mem);
  for (let i = 0; i < 5; i++) {
    const t = 0.2 + (i / 4) * 0.7;
    addFeather(group, elbow.x * t, elbow.y * t + 0.02, elbow.z * t, 0.45, 0.55, 0.1 * sign, 0.11, 0.06, covertMat);
  }
  for (let i = 0; i < 6; i++) {
    const t = 0.15 + (i / 5) * 0.75;
    const x = elbow.x + (wrist.x - elbow.x) * t;
    const y = elbow.y + (wrist.y - elbow.y) * t;
    const z = elbow.z + (wrist.z - elbow.z) * t;
    addFeather(group, x, y, z, -0.2, -0.95, 0.06 * sign, 0.2 - i * 0.01, 0.07, secMat);
  }
  for (let i = 0; i < 7; i++) {
    const t = i / 6;
    addFeather(group, wrist.x, wrist.y, wrist.z, -0.45, -0.85 - t * 0.12, sign * (0.04 + t * 0.06), 0.26 - t * 0.016, 0.055, priMat);
  }
  const grip = new THREE.Mesh(new THREE.SphereGeometry(0.03, 10, 8), boneMat);
  grip.position.set(wrist.x, wrist.y, wrist.z);
  grip.castShadow = true;
  group.add(grip);
  return group;
}

function buildFoot(sign, webMat, clawMat) {
  const group = new THREE.Group();
  group.rotation.x = -Math.PI / 2;
  const s = new THREE.Shape();
  const lat = function (v) { return v * sign; };
  s.moveTo(-0.028, 0);
  s.lineTo(0.02, lat(0.02));
  s.lineTo(0.06, lat(0.05));
  s.lineTo(0.11, lat(0.052));
  s.lineTo(0.125, lat(0.018));
  s.lineTo(0.128, lat(-0.008));
  s.lineTo(0.105, lat(-0.04));
  s.lineTo(0.055, lat(-0.052));
  s.lineTo(0.01, lat(-0.028));
  s.closePath();
  const geo = new THREE.ExtrudeGeometry(s, { depth: 0.012, bevelEnabled: false });
  const web = new THREE.Mesh(geo, webMat);
  web.castShadow = true;
  group.add(web);
  [[0.11, 0.052], [0.125, 0.018], [0.128, -0.008], [0.105, -0.04]].forEach(function (t) {
    const c = new THREE.Mesh(new THREE.ConeGeometry(0.008, 0.03, 5), clawMat);
    c.rotation.z = -Math.PI / 2;
    c.position.set(t[0], lat(t[1]), 0.01);
    group.add(c);
  });
  return group;
}

function updateCamera() {
  const sp = Math.sin(cam.phi);
  camera.position.set(
    cam.target.x + cam.radius * sp * Math.sin(cam.theta),
    cam.target.y + cam.radius * Math.cos(cam.phi),
    cam.target.z + cam.radius * sp * Math.cos(cam.theta)
  );
  camera.lookAt(cam.target);
}

function bindOrbit() {
  const canvas = renderer.domElement;
  const pointers = new Map();
  let pinchDist = 0;
  function pdist() {
    const pts = Array.from(pointers.values());
    if (pts.length < 2) return 0;
    return Math.hypot(pts[0].x - pts[1].x, pts[0].y - pts[1].y);
  }
  canvas.addEventListener("pointerdown", function (e) {
    if (state.mode !== "3d") return;
    pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
    canvas.setPointerCapture(e.pointerId);
    if (pointers.size === 2) pinchDist = pdist();
  });
  canvas.addEventListener("pointermove", function (e) {
    if (!pointers.has(e.pointerId) || state.mode !== "3d") return;
    const prev = pointers.get(e.pointerId);
    if (pointers.size === 1) {
      cam.theta -= (e.clientX - prev.x) * 0.005;
      cam.phi = clamp(cam.phi + (e.clientY - prev.y) * 0.004, 0.28, 1.4);
      updateCamera();
    } else if (pointers.size >= 2) {
      prev.x = e.clientX; prev.y = e.clientY;
      const d = pdist();
      if (pinchDist > 0 && d > 0) cam.radius = clamp(cam.radius * (pinchDist / d), 1.7, 9);
      pinchDist = d;
      updateCamera();
      return;
    }
    prev.x = e.clientX; prev.y = e.clientY;
  });
  function endPtr(e) { pointers.delete(e.pointerId); pinchDist = 0; }
  canvas.addEventListener("pointerup", endPtr);
  canvas.addEventListener("pointercancel", endPtr);
  canvas.addEventListener("wheel", function (e) {
    if (state.mode !== "3d") return;
    e.preventDefault();
    cam.radius = clamp(cam.radius * (e.deltaY > 0 ? 1.08 : 0.93), 1.7, 9);
    updateCamera();
  }, { passive: false });
}
