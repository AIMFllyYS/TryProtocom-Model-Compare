### 建模与物理实现思路
1. **传动与滚动无滑移**：车轮角速度 $\omega_w = v / R_w$（$R_w = 0.34\text{ m}$），曲柄角速度 $\omega_c = \omega_w \cdot (N_{cog}/N_{chainring})$（$16\text{T}/48\text{T} = 1/3$ 传动比），转角单调积分累加；踏板自适应重力保持水平。
2. **鸟类逆运动学（IK）**：大腿与小腿构成双骨骼运动链，髋关节固定于车座，基于余弦定理精确解析向后弯曲的鸟类踝关节（跗跖关节），全蹼足（四趾全连蹼）贴合踏板。
3. **生物特征还原**：严格依据褐鹈鹕（*Pelecanus occidentalis*）形态，包含S形曲颈、带尖钩长喙、风阻与惯性驱动的弹性下颌喉囊、分层羽翼。
4. **双模式与实时音效**：2.5D SVG与3D Three.js共享动力学状态实现无缝平滑切换；Web Audio API纯代码实时合成踏频链条齿合声、速度风噪胎噪与低沉鸣叫。

    function solveAvianLegIK(hip, pedal, L1, L2) {
      const dx = pedal.x - hip.x;
      const dy = pedal.y - hip.y;
      const dist2D = Math.sqrt(dx * dx + dy * dy);
      const clampedDist = Math.max(0.01, Math.min(dist2D, (L1 + L2) - 0.001));

      const cosAlpha = (L1 * L1 + clampedDist * clampedDist - L2 * L2) / (2 * L1 * clampedDist);
      const alpha = Math.acos(Math.max(-1, Math.min(1, cosAlpha)));
      const baseAngle = Math.atan2(dy, dx);

      // Avian ankle bends backwards (caudally / posterior)
      const ankleAngle = baseAngle - alpha;
      const ankleX = hip.x + L1 * Math.cos(ankleAngle);
      const ankleY = hip.y + L1 * Math.sin(ankleAngle);
      const ankleZ = (hip.z + pedal.z) * 0.5;

      return new THREE.Vector3(ankleX, ankleY, ankleZ);

    function createTube(p1, p2, radius, mat) {
      const dir = new THREE.Vector3().subVectors(p2, p1);
      const len = dir.length();
      const geo = new THREE.CylinderGeometry(radius, radius, len, 16);
      const mesh = new THREE.Mesh(geo, mat);
      mesh.castShadow = true;
      mesh.receiveShadow = true;

      const mid = new THREE.Vector3().addVectors(p1, p2).multiplyScalar(0.5);
      mesh.position.copy(mid);

      const up = new THREE.Vector3(0, 1, 0);
      const orient = dir.clone().normalize();
      const quat = new THREE.Quaternion().setFromUnitVectors(up, orient);
      mesh.setRotationFromQuaternion(quat);

      return mesh;
    }

    function update3DPouchGeometry(sway) {
      const vertices = [];
      const indices = [];

      const pStart = new THREE.Vector3(0.17, 0.38, 0);
      const pTip = new THREE.Vector3(0.52, 0.36, 0);
      const sag = 0.11 + sway * 0.035;

      const segments = 10;
      for (let i = 0; i <= segments; i++) {
        const t = i / segments;
        const x = pStart.x + (pTip.x - pStart.x) * t;
        const topY = pStart.y + (pTip.y - pStart.y) * t;
        const botY = topY - Math.sin(t * Math.PI) * sag;

        vertices.push(x, topY, 0);
        vertices.push(x, botY, 0.01 * Math.sin(t * Math.PI));
        vertices.push(x, botY, -0.01 * Math.sin(t * Math.PI));
      }

      for (let i = 0; i < segments; i++) {
        const r1 = i * 3;
        const r2 = (i + 1) * 3;
        indices.push(r1, r2, r1 + 1);
        indices.push(r2, r2 + 1, r1 + 1);
        indices.push(r1, r1 + 2, r2);
        indices.push(r2, r1 + 2, r2 + 2);
      }

      pouchGeo.setAttribute('position', new THREE.Float32BufferAttribute(vertices, 3));
      pouchGeo.setIndex(indices);
      pouchGeo.computeVertexNormals();
    }

    let lastTime = 0;
    let frameCount = 0;
    let lastFpsUpdate = performance.now();
    let accumulatedDistance = 0;

    function animate(currentTime) {
      requestAnimationFrame(animate);

      if (!lastTime) lastTime = currentTime;
      const dt = Math.max(0, Math.min((currentTime - lastTime) / 1000, 0.1));
      lastTime = currentTime;

      const v = simState.speed;
      const dDistance = v * dt;
      accumulatedDistance += dDistance;

      const dWheelAngle = dDistance / WHEEL_RADIUS;
      simState.wheelAngle += dWheelAngle;

      const dCrankAngle = dWheelAngle * (COG_TEETH / CHAINRING_TEETH);
      simState.crankAngle += dCrankAngle;

      const swayCadence = Math.sin(simState.crankAngle * 2);
      const swayWind = Math.sin(currentTime * 0.012) * Math.min(v / 8, 1);
      simState.pouchSway = swayCadence * 0.5 + swayWind * 0.5;

      updateAudio(dt);

      if (simState.mode === '3d') {
        rearWheel3D.rotation.z = -simState.wheelAngle;
        frontWheel3D.rotation.z = -simState.wheelAngle;

        leftCrank3D.rotation.z = -simState.crankAngle + Math.PI;
        rightCrank3D.rotation.z = -simState.crankAngle;

        leftPedal3D.rotation.z = simState.crankAngle - Math.PI;
        rightPedal3D.rotation.z = simState.crankAngle;

        const leftPedalWorld = new THREE.Vector3(
          BB_POS.x + CRANK_LENGTH * Math.sin(simState.crankAngle + Math.PI),
          BB_POS.y - CRANK_LENGTH * Math.cos(simState.crankAngle + Math.PI),
          -PEDAL_OFFSET_Z
        );

        const rightPedalWorld = new THREE.Vector3(
          BB_POS.x + CRANK_LENGTH * Math.sin(simState.crankAngle),
          BB_POS.y - CRANK_LENGTH * Math.cos(simState.crankAngle),
          PEDAL_OFFSET_Z
        );

        update3DLegIK(leftPedalWorld, rightPedalWorld);
        update3DPouchGeometry(simState.pouchSway);

        roadTexture.offset.y = -(accumulatedDistance / 6) % 1.0;

        palmInstances.forEach((p, idx) => {
          p.position.x = ((idx * 4.5 - accumulatedDistance * 1.5) % 63) - 31.5;
        });

        controls.update();
        renderer.render(scene, camera);

      } else {
        render25DSVG(simState.wheelAngle, simState.crankAngle, simState.pouchSway, accumulatedDistance);
      }

      const wheelVel = v / WHEEL_RADIUS;
      const cadenceRPM = (wheelVel / (GEAR_RATIO * 2 * Math.PI)) * 60;
      if (liveWheelVel) liveWheelVel.textContent = `${wheelVel.toFixed(2)} rad/s`;
      if (liveCadence) liveCadence.textContent = `${cadenceRPM.toFixed(1)} rpm`;
      if (liveWheelAngle) liveWheelAngle.textContent = `${simState.wheelAngle.toFixed(2)} rad`;
      if (liveCrankAngle) liveCrankAngle.textContent = `${simState.crankAngle.toFixed(2)} rad`;

      frameCount++;
      if (currentTime - lastFpsUpdate >= 250) {
        const fps = Math.round((frameCount * 1000) / (currentTime - lastFpsUpdate));
        if (fpsDisplay) fpsDisplay.textContent = `${fps} FPS`;
        frameCount = 0;
        lastFpsUpdate = currentTime;
      }
    }

    requestAnimationFrame(animate);

      renderer.setSize(window.innerWidth, window.innerHeight);
    });
