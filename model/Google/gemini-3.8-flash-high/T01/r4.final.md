### 建模与物理实现思路

本项目基于刚体动力学与解剖学约束建立统一仿真模型：
1. **传动物理**：设定轮径 $R=0.34\text{ m}$、牙盘 $44\text{T}$、飞轮 $16\text{T}$，根据无滑移滚动条件 $\omega_{wheel} = v / R$ 及齿比关系严格计算曲柄转速 $\omega_{crank} = \omega_{wheel} \times (16 / 44)$；踏板自平衡保持水平（相对地面倾角为 0）。
2. **生物学与 IK**：针对褐鹈鹕（*Pelecanus occidentalis*）建立两段式逆运动学（IK）链，后折关节严格遵循鸟类踝关节（跗间关节）解剖，足端为四趾全相连的“全蹼足”并精确贴合踏板；刻画 S 形颈、尖端钩状长喙、惯性起伏的大喉囊及三层分层翼羽。
3. **架构与音效**：2.5D SVG 与 3D Three.js 共享单调累加相位与状态，切换平滑无缝；Web Audio API 实时合成链条咔哒、破风滚阻与低沉喉鸣。

      init() {
        if (this.ctx) return;
        const AudioCtx = window.AudioContext || window.webkitAudioContext;
        this.ctx = new AudioCtx();

        // 1. 轮胎摩擦与风噪合成器 (基于粉红/白噪声缓冲区)
        const bufferSize = this.ctx.sampleRate * 2;
        const noiseBuffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const output = noiseBuffer.getChannelData(0);
        let b0 = 0, b1 = 0, b2 = 0;
        for (let i = 0; i < bufferSize; i++) {

          b0 = 0.99886 * b0 + white * 0.0555179;
          b1 = 0.99332 * b1 + white * 0.0750759;
          b2 = 0.96900 * b2 + white * 0.1538520;
          output[i] = (b0 + b1 + b2) * 0.3;
        }

        this.noiseNode = this.ctx.createBufferSource();
        this.noiseNode.buffer = noiseBuffer;
        this.noiseNode.loop = true;

        this.windFilter.Q.value = 1.2;

    /* ==========================================================================
       2. 逆运动学 (IK) 算法：两段式鸟类骨骼结构求解器
       ========================================================================== */
    /**
     * 2-Link 鸟类腿部逆运动学
     * @param {number} hx 髋关节 X
     * @param {number} hy 髋关节 Y
     * @param {number} fx 脚部接触点 X (在踏板上)
     * @param {number} fy 脚部接触点 Y (在踏板上)
     * @param {number} L1 股骨/胫跗骨段长度
     * @param {number} L2 跗跖骨段长度
     * @param {number} bendDir 弯曲方向 (-1 表示后弯，符合鸟类踝关节解剖结构)
     */
    function solveAvianLegIK(hx, hy, fx, fy, L1, L2, bendDir = -1) {
      const dx = fx - hx;
      const dy = fy - hy;
      let dist = Math.hypot(dx, dy);

      // 防止奇异点超限
      const maxDist = (L1 + L2) * 0.999;
      const minDist = Math.abs(L1 - L2) * 1.001;
      dist = Math.max(minDist, Math.min(maxDist, dist));

      const baseAngle = Math.atan2(dy, dx);
      // 余弦定理
      const cosAlpha = (L1 * L1 + dist * dist - L2 * L2) / (2 * L1 * dist);
      const alpha = Math.acos(Math.max(-1, Math.min(1, cosAlpha)));

      // 鸟类踝关节向后折屈
      const jointAngle = baseAngle + bendDir * alpha;
      const jx = hx + L1 * Math.cos(jointAngle);
      const jy = hy + L1 * Math.sin(jointAngle);

      return {
        hip: { x: hx, y: hy },
        ankle: { x: jx, y: jy },
        foot: { x: fx, y: fy }
      };
    }

    // 精简且高性能的 OrbitControls 原生支持
    class SimpleOrbitControls {
      constructor(camera, domElement) {
        this.camera = camera;
        this.domElement = domElement;
        this.target = new THREE.Vector3(0, 0.65, 0);
        this.radius = 2.8;
        this.theta = 0.35;
        this.phi = Math.PI / 2.35;
        this.isDragging = false;
        this.prevMouse = { x: 0, y: 0 };

        this.bindEvents();
        this.updateCamera();
      }

      bindEvents() {
        const dom = this.domElement;
        dom.addEventListener('mousedown', (e) => {
          this.isDragging = true;
          this.prevMouse = { x: e.clientX, y: e.clientY };
        });

        window.addEventListener('mousemove', (e) => {
          if (!this.isDragging) return;
          const dx = e.clientX - this.prevMouse.x;
          const dy = e.clientY - this.prevMouse.y;
          this.prevMouse = { x: e.clientX, y: e.clientY };

          this.theta -= dx * 0.008;
          this.phi = Math.max(0.1, Math.min(Math.PI / 2 - 0.02, this.phi - dy * 0.008));
          this.updateCamera();
        });

        window.addEventListener('mouseup', () => {
          this.isDragging = false;
        });

        dom.addEventListener('wheel', (e) => {
          e.preventDefault();
          this.radius = Math.max(1.2, Math.min(6.5, this.radius + e.deltaY * 0.003));
          this.updateCamera();
        }, { passive: false });

        let touchStartDist = 0;
        dom.addEventListener('touchstart', (e) => {
          if (e.touches.length === 1) {
            this.isDragging = true;
            this.prevMouse = { x: e.touches[0].clientX, y: e.touches[0].clientY };
          } else if (e.touches.length === 2) {
            touchStartDist = Math.hypot(
              e.touches[0].clientX - e.touches[1].clientX,
              e.touches[0].clientY - e.touches[1].clientY
            );
          }
        });

        dom.addEventListener('touchmove', (e) => {
          if (e.touches.length === 1 && this.isDragging) {
            const dx = e.touches[0].clientX - this.prevMouse.x;
            const dy = e.touches[0].clientY - this.prevMouse.y;
            this.prevMouse = { x: e.touches[0].clientX, y: e.touches[0].clientY };
            this.theta -= dx * 0.01;
            this.phi = Math.max(0.1, Math.min(Math.PI / 2 - 0.02, this.phi - dy * 0.01));
            this.updateCamera();
          } else if (e.touches.length === 2) {
            const curDist = Math.hypot(
              e.touches[0].clientX - e.touches[1].clientX,
              e.touches[0].clientY - e.touches[1].clientY
            );
            const delta = touchStartDist - curDist;
            this.radius = Math.max(1.2, Math.min(6.5, this.radius + delta * 0.005));
            touchStartDist = curDist;
            this.updateCamera();
          }
        });

        dom.addEventListener('touchend', () => { this.isDragging = false; });
      }

      updateCamera() {
        const x = this.target.x + this.radius * Math.sin(this.phi) * Math.sin(this.theta);
        const y = this.target.y + this.radius * Math.cos(this.phi);
        const z = this.target.z + this.radius * Math.sin(this.phi) * Math.cos(this.theta);
        this.camera.position.set(x, y, z);
        this.camera.lookAt(this.target);
      }
    }

    function createCylinderBetweenPoints(p1, p2, radius, material) {
      const v1 = new THREE.Vector3(p1.x, p1.y, p1.z || 0);
      const v2 = new THREE.Vector3(p2.x, p2.y, p2.z || 0);
      const dist = v1.distanceTo(v2);
      const cylGeo = new THREE.CylinderGeometry(radius, radius, dist, 12);
      const mesh = new THREE.Mesh(cylGeo, material);
      mesh.castShadow = true;
      mesh.receiveShadow = true;

      const mid = new THREE.Vector3().addVectors(v1, v2).multiplyScalar(0.5);
      mesh.position.copy(mid);
      mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), v2.clone().sub(v1).normalize());
      return mesh;
    }

      const pelicanGroup = new THREE.Group();

      bodyMesh.castShadow = true;

      rightLegMeshes3D = make3DLeg();

    }

    function update3DScene(dt) {
      const scrollDist = SimState.speed * dt;
      roadSegments3D.forEach((seg) => {
        seg.position.x -= scrollDist;
        if (seg.position.x < -24) {
          seg.position.x += 60;
        }
      });

      if (rearWheel3D) rearWheel3D.rotation.z = -SimState.wheelAngle;
      if (frontWheel3D) frontWheel3D.rotation.z = -SimState.wheelAngle;

      if (rightCrank3D) rightCrank3D.rotation.z = -SimState.crankAngle;
      if (leftCrank3D) leftCrank3D.rotation.z = -(SimState.crankAngle + Math.PI);

      if (rightPedal3D) rightPedal3D.rotation.z = SimState.crankAngle;
      if (leftPedal3D) leftPedal3D.rotation.z = SimState.crankAngle + Math.PI;

      if (pouchMesh3D) {
        const sway = Math.sin(SimState.crankAngle * 2) * 0.08 + Math.sin(performance.now() * 0.005) * 0.04;
        pouchMesh3D.scale.y = 0.7 + sway * Math.min(1, SimState.speed / 3);
      }

      const updateLeg3D = (legData, hipPos, pedalWorldPos) => {
        const ik = solveAvianLegIK(
          hipPos.x, hipPos.y,
          pedalWorldPos.x, pedalWorldPos.y,
          PELICAN_ANATOMY.thighLength,
          PELICAN_ANATOMY.shinLength,
          -1
        );

        const midThigh = new THREE.Vector3(
          (ik.hip.x + ik.ankle.x) / 2,
          (ik.hip.y + ik.ankle.y) / 2,
          hipPos.z
        );
        legData.thigh.position.copy(midThigh);
        const thighDir = new THREE.Vector3(ik.ankle.x - ik.hip.x, ik.ankle.y - ik.hip.y, 0).normalize();
        legData.thigh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), thighDir);

        legData.ankleJoint.position.set(ik.ankle.x, ik.ankle.y, hipPos.z);

        const midShin = new THREE.Vector3(
          (ik.ankle.x + ik.foot.x) / 2,
          (ik.ankle.y + ik.foot.y) / 2,
          pedalWorldPos.z
        );
        legData.shin.position.copy(midShin);
        const shinDir = new THREE.Vector3(ik.foot.x - ik.ankle.x, ik.foot.y - ik.ankle.y, pedalWorldPos.z - hipPos.z).normalize();
        legData.shin.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), shinDir);

        legData.footGroup.position.set(ik.foot.x, ik.foot.y + 0.015, pedalWorldPos.z);
        legData.footGroup.rotation.set(0, 0, 0);
      };

      const bb = BIKE_GEOM.bottomBracket;
      const cA = SimState.crankAngle;

      const rPedalWorld = {
        x: bb.x + CRANK_LENGTH * Math.sin(cA),
        y: bb.y - CRANK_LENGTH * Math.cos(cA),
        z: 0.11
      };

      const lPedalWorld = {
        x: bb.x + CRANK_LENGTH * Math.sin(cA + Math.PI),
        y: bb.y - CRANK_LENGTH * Math.cos(cA + Math.PI),
        z: -0.11
      };

      if (rightLegMeshes3D) updateLeg3D(rightLegMeshes3D, PELICAN_ANATOMY.hipRight, rPedalWorld);
      if (leftLegMeshes3D) updateLeg3D(leftLegMeshes3D, PELICAN_ANATOMY.hipLeft, lPedalWorld);

      if (renderer3D && scene3D && camera3D) {
        renderer3D.render(scene3D, camera3D);
      }
    }

    window.switchMode = function(newMode) {
      if (SimState.mode === newMode) return;

      const modeBtn = document.getElementById('btn-mode-toggle');

    let lastTime = performance.now();
    let frameCount = 0;
    let lastFpsUpdate = performance.now();

    window.updateSpeedUI = function() {
      const spd = SimState.speed;
      const kmh = (spd * 3.6).toFixed(1);
      const speedText = document.getElementById('speed-text');
      if (speedText) speedText.innerText = `${spd.toFixed(2)} m/s (${kmh} km/h)`;

      const slider = document.getElementById('speed-slider');
      if (slider && parseFloat(slider.value) !== spd) {
        slider.value = spd;
      }

      const wheelRps = spd / (2 * Math.PI * WHEEL_RADIUS);
      SimState.wheelRPM = wheelRps * 60;
      SimState.cadenceRPM = SimState.wheelRPM * (COG_TEETH / CHAINRING_TEETH);

      const cadenceVal = document.getElementById('cadence-val');
      const wheelRpmVal = document.getElementById('wheel-rpm-val');
      if (cadenceVal) cadenceVal.innerText = `${SimState.cadenceRPM.toFixed(1)} RPM`;
      if (wheelRpmVal) wheelRpmVal.innerText = `${SimState.wheelRPM.toFixed(1)} RPM`;
    };

    function mainAnimationLoop(now) {
      requestAnimationFrame(mainAnimationLoop);

      const dt = Math.min(0.1, (now - lastTime) / 1000);
      lastTime = now;

      // 物理积分计算 (无滑移严格传动比)
      if (SimState.speed > 0) {
        const dWheel = (SimState.speed / WHEEL_RADIUS) * dt;
        const dCrank = dWheel * (COG_TEETH / CHAINRING_TEETH);

        SimState.wheelAngle += dWheel;
        SimState.crankAngle += dCrank;
      }

      // 模式动画更新 (保持同一物理相位与速度，无缝切换)
      if (SimState.mode === '2.5d') {
        update2DScene(dt);
      } else {
        update3DScene(dt);
      }

      // 音效引擎节拍刷新
      soundEngine.update(now);

      // FPS 统计与刷新
      frameCount++;
      if (now - lastFpsUpdate >= 500) {
        const fps = Math.round((frameCount * 1000) / (now - lastFpsUpdate));
        const fpsDisplay = document.getElementById('fps-display');
        if (fpsDisplay) fpsDisplay.innerText = `${fps} FPS`;
        frameCount = 0;
        lastFpsUpdate = now;
      }
    }

      // 如果初始化前已被脚本调用切换模式

        if (nextMute) {
          audioIcon.innerText = '🔇';
          audioText.innerText = '音效: 关';
          btnAudio.classList.remove('active-mode');
        } else {
          audioIcon.innerText = '🔊';
          audioText.innerText = '音效: 开';
          btnAudio.classList.add('active-mode');
        }
      });

      const infoModal = document.getElementById('info-modal');
      document.getElementById('btn-info-toggle').addEventListener('click', () => {
        infoModal.classList.toggle('open');
      });
      document.getElementById('btn-close-info').addEventListener('click', () => {
        infoModal.classList.remove('open');
      });

      window.addEventListener('pointerdown', () => {
        if (!soundEngine.ctx && !soundEngine.isMuted) {
          soundEngine.init();
        }
      }, { once: true });

      requestAnimationFrame(mainAnimationLoop);
    });
  </script>
