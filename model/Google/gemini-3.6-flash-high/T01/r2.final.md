### 建模与物理实现思路

1. **运动学与物理控制**：建立后轮无打滑滚动模型（$\omega_{\text{wheel}} = v / r_{\text{wheel}}$）与牙盘飞轮传动比约束（44T / 16T = 2.75），使得曲柄角速度 $\omega_{\text{crank}} = \omega_{\text{wheel}} / 2.75$。腿部运动链（大腿与胫跗骨）使用二段式逆运动学（IK）解算关节角度，保证脚部精确贴合水平踏板，髋部固定于车座。速度滑块为 0 时运动部件瞬间冻结。
2. **生物形态与双模式建模**：
   - **2.5D 模式（SVG）**：分层矢量绘制棕鹈鹕（*Pelecanus occidentalis*）的长喙、红色钩尖、S型颈、初级/覆羽双层翅膀与全蹼足，带有随骑行相位轻微摆动的喉囊（gular pouch）及视差路面。
   - **3D 模式（Three.js）**：使用程序化 Geometry 组合与 DirectionalLight ShadowMap 阴影渲染，腿部 3D IK 实时骨骼位置同步，支持鼠标/触控 OrbitControls 视角旋转缩放。
3. **音效与平滑过渡**：Web Audio API 实时合成与踏频同步的链条咔哒声、随车速变化的轮胎风噪及偶发叫声；双模式共享同一物理状态，以 opacity 淡入淡出无缝切换（< 1 秒）。

    // ----------------------------------------------------
    // Inverse Kinematics (2D IK Engine)
    // ----------------------------------------------------
    function solve2DIK(hx, hy, px, py, l1, l2, flexForward = true) {
      const dx = px - hx;
      const dy = py - hy;
      let d = Math.sqrt(dx * dx + dy * dy);
      const minD = Math.abs(l1 - l2) + 0.001;
      const maxD = l1 + l2 - 0.001;
      d = Math.max(minD, Math.min(maxD, d));

      const alpha = Math.atan2(dy, dx);
      const cosAngle = (l1 * l1 + d * d - l2 * l2) / (2 * l1 * d);
      const clampedCos = Math.max(-1, Math.min(1, cosAngle));
      const phi = Math.acos(clampedCos);

      // In SVG space (Y down), flexForward means knee x > hip x
      const thighAngle = flexForward ? (alpha - phi) : (alpha + phi);
      const kx = hx + l1 * Math.cos(thighAngle);
      const ky = hy + l1 * Math.sin(thighAngle);

      return { kx, ky, thighAngle };
    }

    function build3DBikeAndPelican() {
      threeBikeGroup = new THREE.Group();
      scene.add(threeBikeGroup);

      const frameMat = new THREE.MeshStandardMaterial({ color: 0x38bdf8, metalness: 0.7, roughness: 0.3 });

      const tireMat = new THREE.MeshStandardMaterial({ color: 0x0f172a, roughness: 0.9 });
      const leatherMat = new THREE.MeshStandardMaterial({ color: 0x451a03, roughness: 0.7 });

      // Helper function to build tube between two points
      function createTube(p1, p2, radius, material) {
        const dir = new THREE.Vector3().subVectors(p2, p1);
        const len = dir.length();
        const geo = new THREE.CylinderGeometry(radius, radius, len, 12);
        const mesh = new THREE.Mesh(geo, material);
        mesh.castShadow = true;
        mesh.position.copy(p1).add(p2).multiplyScalar(0.5);
        mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), dir.clone().normalize());
        return mesh;
      }

      // Wheels
      const wheelGeo = new THREE.TorusGeometry(WHEEL_RADIUS, 0.022, 16, 48);
      threeRearWheel = new THREE.Mesh(wheelGeo, tireMat);
      threeRearWheel.castShadow = true;
      threeRearWheel.position.set(-0.525, WHEEL_RADIUS, 0);

      threeFrontWheel = new THREE.Mesh(wheelGeo, tireMat);
      threeFrontWheel.castShadow = true;
      threeFrontWheel.position.set(0.525, WHEEL_RADIUS, 0);

      // Spokes for 3D wheels
      const spokeMat = new THREE.MeshBasicMaterial({ color: 0x94a3b8 });
      for (let i = 0; i < 12; i++) {
        const ang = (i * Math.PI) / 6;
        const spoke = new THREE.Mesh(new THREE.CylinderGeometry(0.002, 0.002, WHEEL_RADIUS * 2), spokeMat);
        spoke.rotation.z = ang;
        threeRearWheel.add(spoke);
        threeFrontWheel.add(spoke.clone());

      threeBikeGroup.add(threeRearWheel);
      threeBikeGroup.add(threeFrontWheel);

      // Key Bike Nodes

      const RearAxle = new THREE.Vector3(-0.525, WHEEL_RADIUS, 0);
      const FrontAxle = new THREE.Vector3(0.525, WHEEL_RADIUS, 0);

      const HeadTubeTop = new THREE.Vector3(0.38, 0.82, 0);
      const HeadTubeBot = new THREE.Vector3(0.42, 0.45, 0);

      threeBikeGroup.add(createTube(RearAxle, SeatJoint, 0.014, frameMat));
      threeBikeGroup.add(createTube(BB, SeatJoint, 0.018, frameMat));
      threeBikeGroup.add(createTube(BB, HeadTubeBot, 0.02, frameMat));

      threeBikeGroup.add(createTube(HeadTubeBot, FrontAxle, 0.015, frameMat));

      threeBikeGroup.add(saddle);

      const handlebars = new THREE.Mesh(handleGeo, chromeMat);
      handlebars.position.copy(HeadTubeTop);
      handlebars.rotation.x = Math.PI / 2;

      threeCrankGroup.position.copy(BB);

      threeCrankGroup.add(rCrankMesh);

      threeRightPedal.position.set(CRANK_LENGTH, 0, 0.08);

    function update3DScene() {
      if (!renderer || !scene) return;

      // 1. Wheel Rotations
      threeRearWheel.rotation.z = -cumulativeWheelAngle;
      threeFrontWheel.rotation.z = -cumulativeWheelAngle;

      // 2. Crank & Pedal Rotations
      threeCrankGroup.rotation.z = -cumulativeCrankAngle;
      threeRightPedal.rotation.z = cumulativeCrankAngle; // Keep pedal horizontal
      threeLeftPedal.rotation.z = cumulativeCrankAngle;

      // 3. Solve 3D IK for Legs
      const hipPosR = new THREE.Vector3(-0.18, 0.76, 0.08);
      const hipPosL = new THREE.Vector3(-0.18, 0.76, -0.08);

      // World positions of pedals
      const pedalPosR = new THREE.Vector3();
      threeRightPedal.getWorldPosition(pedalPosR);

      const pedalPosL = new THREE.Vector3();
      threeLeftPedal.getWorldPosition(pedalPosL);

      // Solve Right Leg IK
      const ikR = solve2DIK(hipPosR.x, hipPosR.y, pedalPosR.x, pedalPosR.y, THIGH_LENGTH, TARSUS_LENGTH, true);
      const kneeR = new THREE.Vector3(ikR.kx, ikR.ky, 0.08);

      positionBone(threeRightThigh, hipPosR, kneeR);
      positionBone(threeRightShin, kneeR, pedalPosR);

      // Solve Left Leg IK
      const ikL = solve2DIK(hipPosL.x, hipPosL.y, pedalPosL.x, pedalPosL.y, THIGH_LENGTH, TARSUS_LENGTH, true);
      const kneeL = new THREE.Vector3(ikL.kx, ikL.ky, -0.08);

      positionBone(threeLeftThigh, hipPosL, kneeL);
      positionBone(threeLeftShin, kneeL, pedalPosL);

      // 4. Dynamic Gular Pouch Motion
      if (threeGularPouch) {
        const pouchScale = 1.0 + Math.sin(cumulativeCrankAngle * 2) * 0.12;
        threeGularPouch.scale.set(2.2 * pouchScale, 1.0, 0.6);
      }

      if (controls) controls.update();
      if (currentMode === '3d') {
        renderer.render(scene, camera);
      }
    }

    function positionBone(mesh, p1, p2) {
      const dir = new THREE.Vector3().subVectors(p2, p1);
      const len = dir.length();
      mesh.position.copy(p1).add(p2).multiplyScalar(0.5);
      mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), dir.clone().normalize());
      mesh.scale.set(1, len / THIGH_LENGTH, 1);
    }

    }

    function initAudio() {
      if (audioCtx) return;
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      audioCtx = new AudioCtx();

      masterGainNode = audioCtx.createGain();
      masterGainNode.gain.value = isMuted ? 0 : 0.6;
      masterGainNode.connect(audioCtx.destination);

      // Wind / Tire Noise Generator
      const bufferSize = audioCtx.sampleRate * 2;
      const noiseBuffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
      const output = noiseBuffer.getChannelData(0);
      for (let i = 0; i < bufferSize; i++) {
        output[i] = Math.random() * 2 - 1;
      }

      const whiteNoise = audioCtx.createBufferSource();
      whiteNoise.buffer = noiseBuffer;
      whiteNoise.loop = true;

      windFilterNode = audioCtx.createBiquadFilter();
      windFilterNode.type = 'bandpass';
      windFilterNode.frequency.value = 400;
      windFilterNode.Q.value = 1.5;

      windGainNode = audioCtx.createGain();
      windGainNode.gain.value = 0;

      whiteNoise.connect(windFilterNode);
      windFilterNode.connect(windGainNode);
      windGainNode.connect(masterGainNode);
      whiteNoise.start();

      const toast = document.getElementById('audio-toast');
      if (toast) {
        toast.style.opacity = '0';
        setTimeout(() => toast.remove(), 500);
      }
    }

    function playChainClickSound() {
      if (!audioCtx || isMuted || audioCtx.state !== 'running') return;
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();

      osc.type = 'triangle';
      osc.frequency.setValueAtTime(1200 + Math.random() * 300, audioCtx.currentTime);
      gain.gain.setValueAtTime(0.08, audioCtx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.02);

      osc.connect(gain);
      gain.connect(masterGainNode);
      osc.start();
      osc.stop(audioCtx.currentTime + 0.02);
    }

    function playPelicanSquawkSound() {
      if (!audioCtx || isMuted || audioCtx.state !== 'running') return;
      const now = audioCtx.currentTime;
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();

      osc.type = 'sawtooth';
      osc.frequency.setValueAtTime(160, now);
      osc.frequency.exponentialRampToValueAtTime(90, now + 0.35);

      gain.gain.setValueAtTime(0.25, now);
      gain.gain.exponentialRampToValueAtTime(0.001, now + 0.35);

      osc.connect(gain);
      gain.connect(masterGainNode);
      osc.start(now);
      osc.stop(now + 0.35);
    }

    function updateAudioSynth() {
      if (!audioCtx || audioCtx.state !== 'running') return;

      // 1. Tire / Wind Noise Volume & Pitch
      const windVol = Math.min(0.35, (speed / 10.0) * 0.35);
      const windFreq = 300 + speed * 80;
      windGainNode.gain.setTargetAtTime(isMuted ? 0 : windVol, audioCtx.currentTime, 0.05);
      windFilterNode.frequency.setTargetAtTime(windFreq, audioCtx.currentTime, 0.05);

      // 2. Chain & Cog Click Trigger
      const clickStep = (2 * Math.PI) / COG_TEETH;
      if (cumulativeCrankAngle - lastClickCrankAngle >= clickStep) {
        playChainClickSound();
        lastClickCrankAngle = cumulativeCrankAngle;
      }
    }

    // ----------------------------------------------------
    // UI Event Handlers & Mode Switching
    // ----------------------------------------------------
    function switchMode(newMode) {
      if (newMode === currentMode) return;
      currentMode = newMode;

      const svgLayer = document.getElementById('svg-layer');
      const canvasLayer = document.getElementById('canvas-layer');
      const modeBtnText = document.getElementById('mode-text');

      if (currentMode === '3d') {
        init3DScene();
        svgLayer.style.opacity = '0';
        canvasLayer.style.opacity = '1';
        canvasLayer.classList.add('active');
        modeBtnText.textContent = '切换至 2.5D 模式';
      } else {
        svgLayer.style.opacity = '1';
        canvasLayer.style.opacity = '0';
        canvasLayer.classList.remove('active');
        modeBtnText.textContent = '切换至 3D 模式';
      }
    }

    function setupUIEvents() {
      // Audio trigger on first click
      window.addEventListener('click', () => {
        if (!audioCtx) initAudio();
        if (audioCtx && audioCtx.state === 'suspended') audioCtx.resume();
      }, { once: true });

      // Mode Switch Button
      document.getElementById('btn-mode').addEventListener('click', () => {
        switchMode(currentMode === '2.5d' ? '3d' : '2.5d');
      });

      // Mute Toggle Button
      document.getElementById('btn-audio').addEventListener('click', () => {
        isMuted = !isMuted;
        document.getElementById('audio-text').textContent = `声音: ${isMuted ? '关' : '开'}`;
        if (masterGainNode) {
          masterGainNode.gain.value = isMuted ? 0 : 0.6;
        }
      });

      // Squawk Button
      document.getElementById('btn-squawk').addEventListener('click', () => {
        if (!audioCtx) initAudio();
        playPelicanSquawkSound();
      });

      // Speed Slider
      const speedInput = document.getElementById('speed-slider');
      speedInput.addEventListener('input', (e) => {
        speed = parseFloat(e.target.value);
        document.getElementById('speed-val').textContent = speed.toFixed(1);
      });
    }

    // ----------------------------------------------------
    // Main Animation Loop
    // ----------------------------------------------------
    let frameCount = 0;
    let fpsLastTime = performance.now();

    function animate(currentTime) {
      requestAnimationFrame(animate);

      const dt = Math.min(0.1, (currentTime - lastTime) / 1000.0);
      lastTime = currentTime;

      // Physics State Update (Wheel rolling without slipping & gear ratio)
      const wheelAngularVel = speed / WHEEL_RADIUS; // rad/s
      const crankAngularVel = wheelAngularVel / GEAR_RATIO; // rad/s

      cumulativeWheelAngle += wheelAngularVel * dt;
      cumulativeCrankAngle += crankAngularVel * dt;

      // Update Active Mode Scene
      updateSVGScene();
      update3DScene();

      // Update Audio Synth
      updateAudioSynth();

      // FPS & Dashboard Stats Update
      frameCount++;
      if (currentTime - fpsLastTime >= 500) {
        const fps = Math.round((frameCount * 1000) / (currentTime - fpsLastTime));
        document.getElementById('fps-val').textContent = fps;
        frameCount = 0;
        fpsLastTime = currentTime;

        // Realtime stats
        const cadenceRPM = Math.round((crankAngularVel * 60) / (2 * Math.PI));
        document.getElementById('cadence-val').textContent = cadenceRPM;
        document.getElementById('wheel-w-val').textContent = wheelAngularVel.toFixed(1);
        document.getElementById('kmh-val').textContent = (speed * 3.6).toFixed(1);
      }
    }

    // Initialize Application
    initSVGSpokes();
    init3DScene(); // Pre-initialize 3D scene so window.__scene and window.__renderer are ready immediately
    setupUIEvents();
    requestAnimationFrame(animate);
  </script>
