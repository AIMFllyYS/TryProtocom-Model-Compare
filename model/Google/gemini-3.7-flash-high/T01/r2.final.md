### 建模与物理实现思路
1. **生物解剖**：以美洲白鹈鹕（*Pelecanus erythrorhynchos*）为原型，实现初级/次级/覆羽三层分层羽毛、S型长颈、带钩长喙、弹性摆动喉囊及4趾全蹼足（Totipalmate）。
2. **两段式逆运动学（IK）**：髋部固定于车座，基于余弦定理求解向后突出的鸟类踝关节（Hock），使全蹼足精确贴合水平踏板。
3. **刚体动力学与传动**：严格遵循无打滑纯滚动（$v = \omega_{wheel} \cdot R_{wheel}$）与齿比传动（44T/16T = 2.75），踏板轴承自适应重力保持水平。
4. **无缝双模与音效**：共享单调累加相位，CSS/WebGL平滑跨模切换；Web Audio程序化实时合成链条咔哒声、风噪与喉囊鸣叫。

    /**
     * 鸟类腿部 2 段式解析逆运动学求解器
     * @param {number} x0 髋关节 X
     * @param {number} y0 髋关节 Y
     * @param {number} x2 脚/踏板接触点 X
     * @param {number} y2 脚/踏板接触点 Y
     * @param {number} l1 大腿长
     * @param {number} l2 小腿长
     * @param {number} bendDir 弯曲方向 (+1: 鸟类踝关节向后屈曲)
     */
    function solveLegIK2D(x0, y0, x2, y2, l1, l2, bendDir = 1) {
      const dx = x2 - x0;
      const dy = y2 - y0;
      let dist = Math.sqrt(dx * dx + dy * dy);
      // 约束距离在合法工作空间内
      const maxReach = (l1 + l2) * 0.999;
      const minReach = Math.abs(l1 - l2) * 1.001;
      dist = Math.max(minReach, Math.min(maxReach, dist));

      const baseAngle = Math.atan2(dy, dx);
      // 余弦定理求关节夹角
      const cosAlpha = (l1 * l1 + dist * dist - l2 * l2) / (2 * l1 * dist);
      const alpha = Math.acos(Math.max(-1, Math.min(1, cosAlpha)));

      // 鸟类踝关节 (Hock) 向后突出
      const theta1 = baseAngle - bendDir * alpha;
      const x1 = x0 + l1 * Math.cos(theta1);
      const y1 = y0 + l1 * Math.sin(theta1);

      return {
        hip: { x: x0, y: y0 },
        kneeOrAnkle: { x: x1, y: y1 },
        foot: { x: x2, y: y2 }
      };
    }

      buildEnvironment() {
        // 无限循环道路切片
        const roadMat = new THREE.MeshStandardMaterial({ color: 0x1e293b, roughness: 0.85, metalness: 0.1 });
        const lineMat = new THREE.MeshBasicMaterial({ color: 0xfbbf24 });
        const whiteLineMat = new THREE.MeshBasicMaterial({ color: 0xffffff });
        const sandMat = new THREE.MeshStandardMaterial({ color: 0xd97706, roughness: 0.9 });

        const tileLength = 30;
        for (let i = 0; i < 3; i++) {
          const tileGroup = new THREE.Group();

          // 柏油路面 (宽 4m)
          const roadGeo = new THREE.PlaneGeometry(tileLength, 4);
          const roadMesh = new THREE.Mesh(roadGeo, roadMat);
          roadMesh.rotation.x = -Math.PI / 2;
          roadMesh.receiveShadow = true;
          tileGroup.add(roadMesh);

          // 黄色中心虚线
          for (let d = -tileLength / 2 + 1; d < tileLength / 2; d += 3) {
            const dashGeo = new THREE.PlaneGeometry(1.5, 0.12);
            const dashMesh = new THREE.Mesh(dashGeo, lineMat);
            dashMesh.rotation.x = -Math.PI / 2;
            dashMesh.position.set(d, 0.005, 0);
            tileGroup.add(dashMesh);
          }

          // 两侧白线
          const edgeGeo = new THREE.PlaneGeometry(tileLength, 0.1);
          const edgeLeft = new THREE.Mesh(edgeGeo, whiteLineMat);
          edgeLeft.rotation.x = -Math.PI / 2;
          edgeLeft.position.set(0, 0.005, 1.8);
          const edgeRight = edgeLeft.clone();
          edgeRight.position.set(0, 0.005, -1.8);
          tileGroup.add(edgeLeft);
          tileGroup.add(edgeRight);

          // 两侧沙滩草地
          const sandGeo = new THREE.PlaneGeometry(tileLength, 40);
          const sandLeft = new THREE.Mesh(sandGeo, sandMat);
          sandLeft.rotation.x = -Math.PI / 2;
          sandLeft.position.set(0, -0.01, 22);
          sandLeft.receiveShadow = true;
          const sandRight = sandLeft.clone();
          sandRight.position.set(0, -0.01, -22);
          tileGroup.add(sandLeft);
          tileGroup.add(sandRight);

          tileGroup.position.x = (i - 1) * tileLength;
          this.scene.add(tileGroup);
          this.roadTiles.push(tileGroup);
        }

        // 沿途低多边形棕榈树与路桩

          this.scene.add(propGroup);
          this.sceneryProps.push(propGroup);
        }

        domElement.addEventListener('pointerdown', (e) => {

          initAudioOnGesture();

        window.addEventListener('pointermove', (e) => {
          if (!this.orbit.isDragging) return;
          const dx = e.clientX - this.orbit.prevMouseX;
          const dy = e.clientY - this.orbit.prevMouseY;
          this.orbit.prevMouseX = e.clientX;
          this.orbit.prevMouseY = e.clientY;

          this.orbit.theta -= dx * 0.008;
          this.orbit.phi = Math.max(0.2, Math.min(Math.PI / 2 - 0.05, this.orbit.phi - dy * 0.008));
          this.updateCameraPosition();
        });

        window.addEventListener('pointerup', () => {
          this.orbit.isDragging = false;
        });

        domElement.addEventListener('wheel', (e) => {
          e.preventDefault();
          this.orbit.radius = Math.max(1.2, Math.min(6.5, this.orbit.radius + e.deltaY * 0.003));
          this.updateCameraPosition();
        }, { passive: false });
      },

      updateCameraPosition() {
        const x = this.orbit.target.x + this.orbit.radius * Math.sin(this.orbit.phi) * Math.sin(this.orbit.theta);
        const y = this.orbit.target.y + this.orbit.radius * Math.cos(this.orbit.phi);
        const z = this.orbit.target.z + this.orbit.radius * Math.sin(this.orbit.phi) * Math.cos(this.orbit.theta);
        this.camera.position.set(x, y, z);
        this.camera.lookAt(this.orbit.target);
      },

      update(state, dt) {
        // 1. 道路切片与路景无限向前滚动
        const tileLength = 30;
        this.roadTiles.forEach(tile => {
          tile.position.x -= state.speed * dt;
          if (tile.position.x < -tileLength * 1.5) {
            tile.position.x += tileLength * 3;
          }
        });

        this.sceneryProps.forEach(prop => {
          prop.position.x -= state.speed * dt;
          if (prop.position.x < -45) {
            prop.position.x += 100;
          }
        });

        // 2. 车轮与传动系统旋转 (绕 Z 轴)
        if (this.rearWheelGroup) this.rearWheelGroup.rotation.z = -state.wheelAngle;
        if (this.frontWheelGroup) this.frontWheelGroup.rotation.z = -state.wheelAngle;
        if (this.chainringMesh) this.chainringMesh.rotation.z = -state.crankAngle;

        // 曲柄旋转
        if (this.crankLeftGroup) {
          this.crankLeftGroup.rotation.z = -state.crankAngle;
          // 踏板自转对消，始终保持世界水平!
          this.pedalLeftMesh.rotation.z = state.crankAngle;
        }
        if (this.crankRightGroup) {
          this.crankRightGroup.rotation.z = -(state.crankAngle + Math.PI);
          this.pedalRightMesh.rotation.z = state.crankAngle + Math.PI;
        }

        // 3. 计算 3D 腿部逆运动学 (IK)
        const pedals = calculatePedalPositions(state.crankAngle);
        const updateLegMesh = (legObj, hipPos, pedalPos, bendDir) => {
          const ik = solveLegIK2D(
            hipPos.x, hipPos.y,
            pedalPos.x, pedalPos.y,
            CONSTANTS.LEG_UPPER, CONSTANTS.LEG_LOWER,
            bendDir
          );

          const p0 = new THREE.Vector3(ik.hip.x, ik.hip.y, hipPos.z);
          const p1 = new THREE.Vector3(ik.kneeOrAnkle.x, ik.kneeOrAnkle.y, (hipPos.z + pedalPos.z) / 2);
          const p2 = new THREE.Vector3(ik.foot.x, ik.foot.y, pedalPos.z);

          // 大腿骨骼定位与定向
          const dir1 = new THREE.Vector3().subVectors(p1, p0);
          legObj.thigh.position.copy(p0).addScaledVector(dir1, 0.5);
          legObj.thigh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), dir1.clone().normalize());

          // 踝关节球
          legObj.hockJoint.position.copy(p1);

          // 小腿骨骼定位与定向
          const dir2 = new THREE.Vector3().subVectors(p2, p1);
          legObj.shank.position.copy(p1).addScaledVector(dir2, 0.5);
          legObj.shank.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), dir2.clone().normalize());

          // 全蹼足平放于踏板表面 (保持水平)
          legObj.foot.position.set(p2.x + 0.02, p2.y + 0.01, p2.z);
          legObj.foot.rotation.set(0, 0, 0);
        };

        const hipL = { x: CONSTANTS.HIP_POS.x, y: CONSTANTS.HIP_POS.y, z: 0.08 };
        const hipR = { x: CONSTANTS.HIP_POS.x, y: CONSTANTS.HIP_POS.y, z: -0.08 };
        updateLegMesh(this.legLeft, hipL, pedals.left, 1);
        updateLegMesh(this.legRight, hipR, pedals.right, 1);

        // 4. 鹈鹕头部与喉囊弹性动力学
        const bob = Math.sin(state.crankAngle * 2) * 0.015;
        this.headGroup.position.y = CONSTANTS.HIP_POS.y + 0.38 + bob;

        // 喉囊轻微摆动与形变
        const pouchScaleY = 1.0 + Math.sin(state.crankAngle * 2 - 0.4) * 0.18 + (state.speed / 12) * 0.12;
        this.pouchMesh.scale.set(1.0, pouchScaleY, 1.0);

        // 5. 渲染
        this.renderer.render(this.scene, this.camera);
      }
    };

      init() {

      setupTireNoise() {
        const bufferSize = this.ctx.sampleRate * 2;
        const noiseBuffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const output = noiseBuffer.getChannelData(0);
        let b0 = 0, b1 = 0, b2 = 0, b3 = 0, b4 = 0, b5 = 0, b6 = 0;
        // 粉红噪声发生器算法
        for (let i = 0; i < bufferSize; i++) {
          const white = Math.random() * 2 - 1;
          b0 = 0.99886 * b0 + white * 0.0555179;
          b1 = 0.99332 * b1 + white * 0.0750759;
          b2 = 0.96900 * b2 + white * 0.1538520;
          b3 = 0.86650 * b3 + white * 0.3104856;
          b4 = 0.55000 * b4 + white * 0.5329522;
          b5 = -0.7616 * b5 - white * 0.0168980;
          output[i] = (b0 + b1 + b2 + b3 + b4 + b5 + b6 + white * 0.5362) * 0.08;
          b6 = white * 0.115926;
        }

        const whiteNoiseSource = this.ctx.createBufferSource();
        whiteNoiseSource.buffer = noiseBuffer;
        whiteNoiseSource.loop = true;

        this.tireFilter = this.ctx.createBiquadFilter();
        this.tireFilter.type = "lowpass";
        this.tireFilter.frequency.setValueAtTime(300, this.ctx.currentTime);

        this.tireGain = this.ctx.createGain();
        this.tireGain.gain.setValueAtTime(0, this.ctx.currentTime);

        whiteNoiseSource.connect(this.tireFilter);
        this.tireFilter.connect(this.tireGain);
        this.tireGain.connect(this.masterGain);

        whiteNoiseSource.start(0);
      },

      playChainClick() {
        if (!this.ctx || state.audioMuted || state.speed < 0.1) return;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        const filter = this.ctx.createBiquadFilter();

        osc.type = "triangle";
        osc.frequency.setValueAtTime(2800 + Math.random() * 400, this.ctx.currentTime);

        filter.type = "bandpass";
        filter.frequency.value = 3200;
        filter.Q.value = 6;

        gain.gain.setValueAtTime(0.12, this.ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + 0.012);

        osc.connect(filter);
        filter.connect(gain);
        gain.connect(this.chainGain);

        osc.start();
        osc.stop(this.ctx.currentTime + 0.015);
      },

      playPelicanCroak() {
        if (!this.ctx) return;
        const now = this.ctx.currentTime;
        // FM 调制合成低沉喉囊鸣叫
        const carrier = this.ctx.createOscillator();
        const modulator = this.ctx.createOscillator();
        const modGain = this.ctx.createGain();
        const formant = this.ctx.createBiquadFilter();
        const envGain = this.ctx.createGain();

        carrier.type = "sawtooth";
        carrier.frequency.setValueAtTime(130, now);
        carrier.frequency.exponentialRampToValueAtTime(75, now + 0.28);

        modulator.type = "sine";
        modulator.frequency.setValueAtTime(42, now);
        modGain.gain.setValueAtTime(90, now);
        modulator.connect(carrier.frequency);

        formant.type = "bandpass";
        formant.frequency.setValueAtTime(680, now);
        formant.Q.setValueAtTime(3.5, now);

        envGain.gain.setValueAtTime(0.001, now);
        envGain.gain.linearRampToValueAtTime(0.35, now + 0.04);
        envGain.gain.exponentialRampToValueAtTime(0.001, now + 0.32);

        carrier.connect(formant);
        formant.connect(envGain);
        envGain.connect(this.masterGain);

        modulator.start(now);
        carrier.start(now);
        modulator.stop(now + 0.35);
        carrier.stop(now + 0.35);
      },

      update(state, dt) {
        if (!this.ctx || this.ctx.state !== 'running') return;

        // 1. 动态调节轮胎/风噪
        const speedRatio = Math.min(1.0, state.speed / 12);
        const targetCutoff = 250 + speedRatio * 1800;
        const targetVol = speedRatio > 0.01 ? Math.pow(speedRatio, 1.2) * 0.35 : 0;
        this.tireFilter.frequency.setTargetAtTime(targetCutoff, this.ctx.currentTime, 0.05);
        this.tireGain.gain.setTargetAtTime(targetVol, this.ctx.currentTime, 0.05);

        // 2. 链条点击声触发逻辑 (根据齿数与踏频)
        if (state.speed > 0.1) {
          const crankRps = state.speed / (CONSTANTS.WHEEL_RADIUS * CONSTANTS.GEAR_RATIO * 2 * Math.PI);
          const clickFreq = Math.max(1, crankRps * CONSTANTS.CHAINRING_TEETH);
          const clickInterval = 1 / clickFreq;

          if (this.ctx.currentTime >= this.nextClickTime) {
            this.playChainClick();
            this.nextClickTime = this.ctx.currentTime + clickInterval;
          }
        }

        // 3. 偶发鹈鹕叫声
        this.croakTimer += dt;
        if (state.speed > 1.0 && this.croakTimer > 15 + Math.random() * 10) {
          this.croakTimer = 0;
          this.playPelicanCroak();
        }
      },

      setMuted(muted) {
        if (!this.ctx) return;
        this.masterGain.gain.setTargetAtTime(muted ? 0 : 0.6, this.ctx.currentTime, 0.03);
      }
    };

    function initAudioOnGesture() {
      if (!audioSystem.ctx) {
        audioSystem.init();
      }
      if (audioSystem.ctx && audioSystem.ctx.state === 'suspended') {
        audioSystem.ctx.resume();
      }
    }

    function toggleAudio() {
      initAudioOnGesture();
      state.audioMuted = !state.audioMuted;
      audioSystem.setMuted(state.audioMuted);
      const icon = document.getElementById('sound-icon');
      const btn = document.getElementById('btn-sound');
      if (state.audioMuted) {
        icon.textContent = '🔇';
        btn.classList.remove('active');
      } else {
        icon.textContent = '🔊';
        btn.classList.add('active');
      }
    }

    function triggerPelicanCroak() {
      initAudioOnGesture();
      if (state.audioMuted) {
        toggleAudio();
      }
      audioSystem.playPelicanCroak();
    }

    /* ==========================================================================
       6. 交互控制与主循环调度

    function setAppMode(m) {
      if (m !== '2.5d' && m !== '3d') return;
      state.mode = m;
      if (m === '3d') {
        document.body.classList.add('mode-3d');

        document.getElementById('btn-mode-3d').classList.add('active');

        document.getElementById('btn-mode-3d').classList.remove('active');
        document.getElementById('btn-mode-2d').classList.add('active');
      }
    }

    function setAppSpeed(v) {
      state.targetSpeed = Math.max(0, Math.min(15, Number(v) || 0));
      state.speed = state.targetSpeed;
      document.getElementById('speed-slider').value = state.speed;
      updateSpeedUI();

    function onSpeedSliderChange(val) {
      initAudioOnGesture();
      state.targetSpeed = parseFloat(val);

    function updateSpeedUI() {
      const kmh = (state.speed * 3.6).toFixed(1);
      document.getElementById('speed-display').textContent = `${state.speed.toFixed(2)} m/s (${kmh} km/h)`;

      // 更新预设按钮状态
      const presets = document.querySelectorAll('.preset-btn');
      presets.forEach(p => p.classList.remove('active'));
    }

    function toggleInfoModal(open) {
      const modal = document.getElementById('info-modal');
      if (open) {
        modal.classList.add('open');
      } else {
        modal.classList.remove('open');
      }
    }

    // 主动画渲染循环
    function mainLoop(timestamp) {
      if (!state.lastTimestamp) state.lastTimestamp = timestamp;
      const rawDt = (timestamp - state.lastTimestamp) / 1000;
      state.lastTimestamp = timestamp;
      const dt = Math.min(0.1, rawDt); // 钳制单帧最大时间步

      // 物理积分计算 (无打滑纯滚动)
      if (state.speed > 0) {
        const deltaWheelAngle = (state.speed * dt) / CONSTANTS.WHEEL_RADIUS;
        const deltaCrankAngle = deltaWheelAngle / CONSTANTS.GEAR_RATIO;

        state.wheelAngle += deltaWheelAngle;
        state.crankAngle += deltaCrankAngle;
        state.distance += state.speed * dt;
      }

      // 同步更新 2.5D 与 3D 视图 (保持完全无缝相位同步)
      svgSystem.update(state, dt);
      threeSystem.update(state, dt);

      // 音频合成引擎更新
      audioSystem.update(state, dt);

      // FPS 与遥测监控计算
      state.frameCount++;
      state.fpsTimer += dt;
      if (state.fpsTimer >= 0.25) {
        state.fps = state.frameCount / state.fpsTimer;
        state.frameCount = 0;
        state.fpsTimer = 0;
        document.getElementById('fps-val').textContent = `${state.fps.toFixed(1)} FPS`;
        document.getElementById('dt-val').textContent = `${(rawDt * 1000).toFixed(1)} ms`;

        // 遥测面板数值
        const cadenceRPM = ((state.speed / (CONSTANTS.WHEEL_RADIUS * CONSTANTS.GEAR_RATIO)) * 60) / (2 * Math.PI);
        document.getElementById('tel-cadence').textContent = cadenceRPM.toFixed(1);
        document.getElementById('tel-crank').textContent = state.crankAngle.toFixed(2);
        document.getElementById('tel-wheel').textContent = state.wheelAngle.toFixed(2);
      }

      requestAnimationFrame(mainLoop);
    }

    // 初始化入口
    window.addEventListener('DOMContentLoaded', () => {
      svgSystem.init();
      threeSystem.init();
      updateSpeedUI();
      requestAnimationFrame(mainLoop);

      // 用户首次交互唤醒音频
      window.addEventListener('click', initAudioOnGesture, { once: true });
      window.addEventListener('touchstart', initAudioOnGesture, { once: true });
    });
  </script>
