### 建模与物理实现思路
本页面基于棕鹈鹕（*Pelecanus occidentalis*）生物特征与经典自行车传动结构进行双模式程序化建模。2.5D采用内联SVG多层分层与阴影，3D采用Three.js PBR材质与阴影渲染。物理上严格遵循无打滑滚动（\(\omega_{\text{wheel}} = v/r\)）与44T/16T传动比（\(\omega_{\text{crank}} = \omega_{\text{wheel}}/2.75\)）；两段式逆运动学（IK）解算两腿关节，使反向弯曲的踝关节符合鸟类骨骼，全蹼足贴合水平踏板；Web Audio API实时合成链条咔哒声、风噪与啼鸣；视角平滑过渡且保持动画相位连续。

  <script type="module">
    import * as THREE from 'three';
    import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

    // Expose THREE globally for bench access if needed
    window.THREE = THREE;
    window.OrbitControls = OrbitControls;

    /* =========================================================================
       1. CONSTANTS & GLOBAL STATE
       ========================================================================= */
    const CONFIG = {
      WHEEL_RADIUS: 0.34,       // meters
      CHAINRING_TEETH: 44,
      COG_TEETH: 16,
      CRANK_LENGTH: 0.17,       // meters
      WHEELBASE: 1.05,          // meters
      HIP_SADDLE_OFFSET: 0.0,   // meters (hip anchored at saddle)
      MAX_SPEED: 10.0,          // m/s
      DEFAULT_SPEED: 4.0,        // m/s
      SPECIES: 'Pelecanus occidentalis'
    };

    const GEAR_RATIO = CONFIG.CHAINRING_TEETH / CONFIG.COG_TEETH; // 2.75

    const state = {
      mode: '2.5d',             // '2.5d' | '3d'
      speed: CONFIG.DEFAULT_SPEED,
      wheelAngle: 0.0,          // accumulated radians (monotonic)
      crankAngle: 0.0,          // accumulated radians (monotonic)
      isMuted: false,
      audioInitialized: false
    };

    /* =========================================================================
       2. WEB AUDIO API SYNTHESIZER
       ========================================================================= */
    class BikeAudioEngine {
      constructor() {
        this.ctx = null;
        this.isMuted = false;
        this.windGain = null;
        this.lastClickAngle = 0;
        this.squawkTimer = 0;
      }

      init() {
        if (this.ctx) return;
        const AudioContext = window.AudioContext || window.webkitAudioContext;
        if (!AudioContext) return;
        this.ctx = new AudioContext();

        // 1. Wind & Tire Friction Noise Setup
        const bufferSize = 2 * this.ctx.sampleRate;
        const noiseBuffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const output = noiseBuffer.getChannelData(0);
        for (let i = 0; i < bufferSize; i++) {
          output[i] = Math.random() * 2 - 1;
        }
        const whiteNoise = this.ctx.createBufferSource();
        whiteNoise.buffer = noiseBuffer;
        whiteNoise.loop = true;

        const filter = this.ctx.createBiquadFilter();
        filter.type = 'lowpass';
        filter.frequency.value = 350;

        this.windGain = this.ctx.createGain();
        this.windGain.gain.value = 0;

        whiteNoise.connect(filter);
        filter.connect(this.windGain);
        this.windGain.connect(this.ctx.destination);
        whiteNoise.start();

        state.audioInitialized = true;
        const prompt = document.getElementById('audio-prompt');
        if (prompt) prompt.style.opacity = '0';
      }

      update(speed, dt) {
        if (!this.ctx || this.isMuted) return;
        if (this.ctx.state === 'suspended') {
          this.ctx.resume();
        }

        // Wind/Tire sound volume tied to speed
        if (this.windGain) {
          const targetGain = Math.min(0.25, (speed / CONFIG.MAX_SPEED) * 0.22);
          this.windGain.gain.setTargetAtTime(targetGain, this.ctx.currentTime, 0.08);
        }

        // Chain clicking sound triggered per tooth pass
        const anglePerTooth = (2 * Math.PI) / CONFIG.CHAINRING_TEETH;
        if (state.crankAngle - this.lastClickAngle >= anglePerTooth) {
          if (speed > 0.05) this.playClick();
          this.lastClickAngle = state.crankAngle;
        }

        // Occasional pelican low squawk call
        if (speed > 0.5) {
          this.squawkTimer += dt;
          if (this.squawkTimer > 10.0) {
            if (Math.random() < 0.35) {
              this.playPelicanCall();
              this.squawkTimer = 0;
            }
          }
        }
      }

      playClick() {
        if (!this.ctx || this.isMuted) return;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(1100 + Math.random() * 250, this.ctx.currentTime);
        gain.gain.setValueAtTime(0.03, this.ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + 0.012);
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start();
        osc.stop(this.ctx.currentTime + 0.015);
      }

      playPelicanCall() {
        if (!this.ctx || this.isMuted) return;
        const now = this.ctx.currentTime;
        const osc1 = this.ctx.createOscillator();
        const osc2 = this.ctx.createOscillator();
        const gain = this.ctx.createGain();

        osc1.type = 'sawtooth';
        osc2.type = 'square';

        osc1.frequency.setValueAtTime(160, now);
        osc1.frequency.exponentialRampToValueAtTime(105, now + 0.35);
        osc2.frequency.setValueAtTime(165, now);
        osc2.frequency.exponentialRampToValueAtTime(108, now + 0.35);

        gain.gain.setValueAtTime(0.01, now);
        gain.gain.linearRampToValueAtTime(0.1, now + 0.06);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.4);

        osc1.connect(gain);
        osc2.connect(gain);
        gain.connect(this.ctx.destination);

        osc1.start(now);
        osc2.start(now);
        osc1.stop(now + 0.42);
        osc2.stop(now + 0.42);
      }

      toggleMute() {
        this.isMuted = !this.isMuted;
        if (this.windGain && this.isMuted) {
          this.windGain.gain.setValueAtTime(0, this.ctx.currentTime);
        }
        return this.isMuted;
      }
    }

    /* =========================================================================
       3. INVERSE KINEMATICS (IK) SOLVER FOR BIRD LEGS
       ========================================================================= */
    function solveLegIK2D(hip, pedal, l1, l2) {
      const dx = pedal.x - hip.x;
      const dy = pedal.y - hip.y;
      const dist = Math.sqrt(dx * dx + dy * dy);

      // Clamp distance within reachable bounds
      const d = Math.min(Math.max(dist, Math.abs(l1 - l2) + 0.001), (l1 + l2) * 0.998);
      const phi = Math.atan2(dy, dx);

      // Law of Cosines angle at hip joint
      const cosAlpha = (l1 * l1 + d * d - l2 * l2) / (2 * l1 * d);
      const alpha = Math.acos(Math.max(-1, Math.min(1, cosAlpha)));

      // For bird limb: ankle (hock) bends BACKWARDS
      const ankleAngle = phi - alpha;
      const ankle = {
        x: hip.x + l1 * Math.cos(ankleAngle),
        y: hip.y + l1 * Math.sin(ankleAngle)
      };

      return { hip, ankle, foot: pedal };
    }

    function update3DScene() {
      if (!scene || !renderer) return;

      // Wheel Rotations
      rearWheel3D.rotation.z = -state.wheelAngle;
      frontWheel3D.rotation.z = -state.wheelAngle;

      // Ground Texture Movement simulation
      if (groundGrid3D) {
        groundGrid3D.position.x = (-state.wheelAngle * CONFIG.WHEEL_RADIUS) % 1.0;
      }

      // Crank Rotation
      crank3DGroup.rotation.z = -state.crankAngle;

      // Keep Pedals Horizontal (counter-rotate pedal mesh)
      leftPedal3D.rotation.z = state.crankAngle;
      rightPedal3D.rotation.z = state.crankAngle;

      // Gular Pouch Swaying Motion
      if (gularPouch3DMesh) {
        const sway = Math.sin(state.wheelAngle * 1.5) * 0.08;
        gularPouch3DMesh.rotation.z = sway;
      }

      // Solve 3D Leg IK & Update Cylinder Bone Transformations
      const hipLeft = new THREE.Vector3(-0.18, 0.85, -0.10);
      const hipRight = new THREE.Vector3(-0.18, 0.85, 0.10);

      // Get pedal world locations
      const leftPedalWorld = new THREE.Vector3();
      leftPedal3D.getWorldPosition(leftPedalWorld);
      bike3DGroup.worldToLocal(leftPedalWorld);

      const rightPedalWorld = new THREE.Vector3();
      rightPedal3D.getWorldPosition(rightPedalWorld);
      bike3DGroup.worldToLocal(rightPedalWorld);

      const l1 = 0.35, l2 = 0.34;

      // Left Leg IK
      const ikL = solveLegIK2D({ x: hipLeft.x, y: hipLeft.y }, { x: leftPedalWorld.x, y: leftPedalWorld.y }, l1, l2);
      const ankleL = new THREE.Vector3(ikL.ankle.x, ikL.ankle.y, -0.10);

      updateBoneMesh(leftLegThigh3D, hipLeft, ankleL);
      updateBoneMesh(leftLegShin3D, ankleL, leftPedalWorld);

      // Right Leg IK
      const ikR = solveLegIK2D({ x: hipRight.x, y: hipRight.y }, { x: rightPedalWorld.x, y: rightPedalWorld.y }, l1, l2);
      const ankleR = new THREE.Vector3(ikR.ankle.x, ikR.ankle.y, 0.10);

      updateBoneMesh(rightLegThigh3D, hipRight, ankleR);
      updateBoneMesh(rightLegShin3D, ankleR, rightPedalWorld);

      controls.update();
      renderer.render(scene, camera);
    }

    function updateBoneMesh(mesh, p1, p2) {
      const dir = new THREE.Vector3().subVectors(p2, p1);
      const len = dir.length();
      mesh.scale.set(1, len, 1);
      mesh.position.copy(p1).add(dir.clone().multiplyScalar(0.5));
      mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), dir.normalize());
    }

        view3D.classList.add('hidden');
        view2D.classList.remove('hidden');
        view2D.classList.add('active');
        if (btnText) btnText.innerText = '切换至 3D 模式';
      }
    }

    /* =========================================================================
       7. MAIN ANIMATION LOOP & FPS COUNTER
       ========================================================================= */
    let lastTimestamp = performance.now();
    let frameCount = 0;
    let fpsTimer = performance.now();

    function animate(timestamp) {
      requestAnimationFrame(animate);

      const dt = Math.min((timestamp - lastTimestamp) / 1000, 0.1);
      lastTimestamp = timestamp;

      // Update Physics & Accumulate Angles
      if (state.speed > 0) {
        const deltaWheel = (state.speed / CONFIG.WHEEL_RADIUS) * dt;
        state.wheelAngle += deltaWheel;
        state.crankAngle += deltaWheel / GEAR_RATIO;
      }

      // Render Active Mode
      if (state.mode === '2.5d') {
        update2DScene();
      } else if (state.mode === '3d') {
        update3DScene();
      }

      // Audio Synthesis Update
      audioEngine.update(state.speed, dt);

      // UI & FPS Update
      frameCount++;
      if (timestamp - fpsTimer >= 1000) {
        const fpsDisplay = document.getElementById('fps-display');
        if (fpsDisplay) fpsDisplay.innerText = `${frameCount} FPS`;
        
        const cadence = ((state.speed / (CONFIG.WHEEL_RADIUS * GEAR_RATIO)) * 60 / (2 * Math.PI)).toFixed(0);
        const infoCadence = document.getElementById('info-cadence');
        if (infoCadence) infoCadence.innerText = `${cadence} RPM`;

        frameCount = 0;
        fpsTimer = timestamp;
      }
    }

    /* =========================================================================
       8. EVALUATION INTERFACE (__bench)
       ========================================================================= */
    window.__bench = {
      setMode(m) {
        if (m === '2.5d' || m === '3d') {
          switchMode(m);
        }
      },
      getMode() {
        return state.mode;
      },
      setSpeed(v) {
        const speed = Math.max(0, Math.min(CONFIG.MAX_SPEED, Number(v) || 0));
        state.speed = speed;
        const slider = document.getElementById('speed-slider');
        if (slider) slider.value = speed;
        const speedVal = document.getElementById('speed-val');
        if (speedVal) speedVal.innerText = `${speed.toFixed(1)} m/s (${(speed * 3.6).toFixed(1)} km/h)`;
      },
      getState() {
        return {
          mode: state.mode,
          speed: state.speed,
          wheelAngle: state.wheelAngle,
          crankAngle: state.crankAngle,
          wheelRadius: CONFIG.WHEEL_RADIUS,
          chainringTeeth: CONFIG.CHAINRING_TEETH,
          cogTeeth: CONFIG.COG_TEETH,
          pedalAngles: [0, 0],
          footToPedal: [0, 0],
          hipToSaddle: CONFIG.HIP_SADDLE_OFFSET
        };
      },
      species: CONFIG.SPECIES
    };

    /* =========================================================================
       9. EVENT LISTENERS & DOM INITIALIZATION
       ========================================================================= */
    document.addEventListener('DOMContentLoaded', () => {
      // Speed Slider Listener
      const slider = document.getElementById('speed-slider');
      slider.addEventListener('input', (e) => {
        window.__bench.setSpeed(e.target.value);
      });

      // Mode Toggle Button
      const btnMode = document.getElementById('btn-mode-toggle');
      btnMode.addEventListener('click', () => {
        audioEngine.init();
        switchMode(state.mode === '2.5d' ? '3d' : '2.5d');
      });

      // Mute Toggle Button
      const btnAudio = document.getElementById('btn-audio-toggle');
      btnAudio.addEventListener('click', () => {
        audioEngine.init();
        const muted = audioEngine.toggleMute();
        document.getElementById('audio-btn-icon').innerText = muted ? '🔇' : '🔊';
        document.getElementById('audio-btn-text').innerText = muted ? '静音中' : '音效开启';
      });

      // Global User Click to Unlock Audio Context
      window.addEventListener('pointerdown', () => {
        audioEngine.init();
      }, { once: true });

      // Window Resize Listener
      window.addEventListener('resize', () => {
        if (renderer && camera) {
          camera.aspect = window.innerWidth / window.innerHeight;
          camera.updateProjectionMatrix();
          renderer.setSize(window.innerWidth, window.innerHeight);
        }
      });

      // Initialize 3D Scene ahead of time for smooth transition
      init3DScene();

      // Start Animation Loop
      requestAnimationFrame(animate);
    });
  </script>
</body>
</html>
