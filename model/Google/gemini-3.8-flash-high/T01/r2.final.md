### 建模与物理实现思路

本项目以美洲白鹈鹕（*Pelecanus erythrorhynchos*）为原型，实现 2.5D SVG 与 3D Three.js 双模式无缝切换。
1. **物理传动**：无滑滚动遵循 $\omega_{wheel} = v / R_{wheel}$（轮径 $0.34\text{ m}$）；48T 牙盘与 16T 飞轮构成 $3:1$ 传动比（$\omega_{crank} = \omega_{wheel} / 3$）。
2. **生物与IK**：鸟类踝关节向后反曲，采用双骨节解析逆运动学（IK），足部全蹼（4 趾相连）平贴水平踏板，髋关节紧贴车座；喉囊伴随踏频自适应弹性颤动。
3. **音效与性能**：Web Audio 实时合成链条咔哒、风噪及偶发喉音鸣叫；统一物理状态驱动双视图，稳定 60 FPS。

        f1.connect(f2);
        f2.connect(gain);

    /**
     * In birds, the visible "knee" flexes forward, while the prominent
     * backward-bending joint is the intertarsal ankle joint (hock).
     * The upper link (femur + tibiotarsus) meets the lower link (tarsometatarsus)
     * at this backward-oriented ankle joint.
     */
    function solveBirdLegIK(hipX, hipY, footX, footY, lenUpper, lenLower) {
      const dx = footX - hipX;
      const dy = footY - hipY;
      const d = Math.hypot(dx, dy);

      // Clamp distance within valid range to avoid singularities
      const clampedD = Math.max(Math.abs(lenUpper - lenLower) + 1e-4, Math.min(lenUpper + lenLower - 1e-4, d));
      const baseAngle = Math.atan2(dy, dx);

      // Law of Cosines for the angle between hip-foot line and upper leg segment
      const cosAngle = (lenUpper * lenUpper + clampedD * clampedD - lenLower * lenLower) / (2 * lenUpper * clampedD);
      const angleOffset = Math.acos(Math.max(-1, Math.min(1, cosAngle)));

      // In birds, the ankle joint points BACKWARD (towards -X, so subtract angleOffset)
      const upperAngle = baseAngle - angleOffset;
      const ankleX = hipX + lenUpper * Math.cos(upperAngle);
      const ankleY = hipY + lenUpper * Math.sin(upperAngle);

      return { ankleX, ankleY, footX, footY };
    }

    /**
     * Analytical 3D Inverse Kinematics for Bird Leg
     * Exactly locks foot to pedal center with backward-bending hock
     */
    function update3DLegIK(legParts, hipPos, pedalWorldPos) {
      const D = new THREE.Vector3().subVectors(pedalWorldPos, hipPos);
      const d = D.length();

      // Ensure valid triangle closure
      const clampedD = Math.max(Math.abs(legUpperLen3D - legLowerLen3D) + 0.001, Math.min(legUpperLen3D + legLowerLen3D - 0.001, d));
      const u = D.clone().normalize();

      // Midpoint distance along hip-to-pedal vector
      const s = (legUpperLen3D * legUpperLen3D - legLowerLen3D * legLowerLen3D + clampedD * clampedD) / (2 * clampedD);
      const r = Math.sqrt(Math.max(0, legUpperLen3D * legUpperLen3D - s * s));

      // In birds, the ankle bends backward (-X)
      const backwardRef = new THREE.Vector3(-1, 0, 0);
      const v = backwardRef.clone().sub(u.clone().multiplyScalar(backwardRef.dot(u))).normalize();

      // Exact Ankle position
      const anklePos = hipPos.clone().add(u.clone().multiplyScalar(s)).add(v.clone().multiplyScalar(r));

      // Position & Orient Upper Leg Segment
      const upperMid = hipPos.clone().add(anklePos).multiplyScalar(0.5);
      legParts.upperBone.position.copy(upperMid);
      legParts.upperBone.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), anklePos.clone().sub(hipPos).normalize());

      // Position Ankle Joint
      legParts.ankleJoint.position.copy(anklePos);

      // Position & Orient Lower Leg Segment
      const lowerMid = anklePos.clone().add(pedalWorldPos).multiplyScalar(0.5);
      legParts.lowerBone.position.copy(lowerMid);
      legParts.lowerBone.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), pedalWorldPos.clone().sub(anklePos).normalize());

      // Position Foot right on Pedal (footToPedal distance is identically 0.0)
      legParts.footGroup.position.copy(pedalWorldPos);
      // Foot stays horizontal (relative to ground)
      legParts.footGroup.rotation.set(0, 0, 0);
    }
