### 建模与物理实现思路
1. **物理约束**：以 `wheelAngle += speed/radius*dt` 与 `crankAngle = wheelAngle/gearRatio` 驱动双模式同步联动。通过保持踏板水平作为逆运动学（IK）的端点，以双连杆模型结合余弦定理实时计算鹈鹕腿部（大腿、小腿）的关节转角，使脚部完美贴合踏板，且呈现出鸟类特有的后屈踝关节姿态。
2. **2.5D与3D建模**：2.5D模式基于SVG绘制，利用图层遮挡、渐变和 `feDropShadow` 制造阴影纵深感；3D模式使用 Three.js 基础几何体构建自行车框架与鹈鹕特征（如喉囊摆动、翅膀羽毛分层），结合定向光源与阴影实现立体场景。
3. **音频与交互**：用 Web Audio API 根据速度与踏频实时生成风噪与链条咔哒声；双模式共用状态机并通过 CSS `opacity` 实现无缝过渡。

        function calculateIK(hip, target, l1, l2, flip) {
            let dx = target.x - hip.x;
            let dy = target.y - hip.y;
            let d = Math.sqrt(dx*dx + dy*dy);
            let clampedD = Math.max(0.001, Math.min(d, l1 + l2 - 0.001));

            let a = (l1*l1 + clampedD*clampedD - l2*l2) / (2*l1*clampedD);
            let alpha = Math.acos(Math.max(-1, Math.min(1, a)));
            let theta = Math.atan2(dy, dx);

            let angle1 = flip ? theta - alpha : theta + alpha;
            let knee = {
                x: hip.x + l1 * Math.cos(angle1),
                y: hip.y + l1 * Math.sin(angle1)
            };

            let dx2 = target.x - knee.x;
            let dy2 = target.y - knee.y;
            let angle2 = Math.atan2(dy2, dx2);

            const endX = knee.x + l2 * Math.cos(angle2);
            const endY = knee.y + l2 * Math.sin(angle2);
            const dist = Math.hypot(target.x - endX, target.y - endY);

            return { knee, angle1, angle2, dist };
