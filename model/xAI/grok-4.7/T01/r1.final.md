褐鹈鹕（Pelecanus occidentalis）与自行车共用一套程序化几何，SVG 与 Three.js 读同一姿态。前进速度 v 决定后轮角速度 ω=v/R，曲柄角速度为 ω×飞轮齿数/牙盘齿数，链条按飞轮节圆前进。踏板转角抵消曲柄，平台保持水平。大腿与跗跖做逆运动学，踝向后弯，脚固接踏板，髋在鞍座旁。速度为 0 时相位停止累加。两种模式交叉淡化，不重置转角。

  function solveIK(hip, foot, L1, L2) {
    const dx = foot.x - hip.x;
    const dy = foot.y - hip.y;
    const dz = foot.z - hip.z;
    const d = Math.hypot(dx, dy, dz) || 1e-6;
    const ux = dx / d, uy = dy / d, uz = dz / d;
    const cosA = clamp((L1 * L1 + d * d - L2 * L2) / (2 * L1 * d), -1, 1);
    const sinA = Math.sqrt(Math.max(0, 1 - cosA * cosA));
    let px = dy, py = -dx;
    const pl = Math.hypot(px, py) || 1;
    px /= pl; py /= pl;
    const along = L1 * cosA;
    const off = L1 * sinA;
    const j1 = {
      x: hip.x + ux * along + px * off,
      y: hip.y + uy * along + py * off,
      z: hip.z + uz * along
    };
    const j2 = {
      x: hip.x + ux * along - px * off,
      y: hip.y + uy * along - py * off,
      z: hip.z + uz * along
    };
    // Posterior ankle: the intertarsal joint bows backward, toward the tail.
    return j1.x <= j2.x ? j1 : j2;
  }

  function pedalPoint(angle, z) {
    return {
      x: GEO.bb.x + SPEC.crankLength * Math.cos(-angle),

  function rotZAbout(p, pivot, ang) {
    const dx = p.x - pivot.x, dy = p.y - pivot.y;
    const c = Math.cos(ang), s = Math.sin(ang);
    return {
      x: pivot.x + dx * c - dy * s,
      y: pivot.y + dx * s + dy * c,
      z: p.z
    };
  }

  function catmull(a, b, c, d, u) {
    const u2 = u * u, u3 = u2 * u;
    return 0.5 * ((2 * b) + (-a + c) * u + (2 * a - 5 * b + 4 * c - d) * u2 + (-a + 3 * b - 3 * c + d) * u3);
  }

  function sampleNeck(ctrls, n) {
    const out = [];
    const seg = ctrls.length - 1;
    for (let i = 0; i < n; i++) {
      const x = (i / (n - 1)) * seg;
      const k = Math.min(seg - 1, Math.floor(x));
      const u = x - k;
      const p0 = ctrls[Math.max(0, k - 1)];
      const p1 = ctrls[k];
      const p2 = ctrls[Math.min(seg, k + 1)];
      const p3 = ctrls[Math.min(seg, k + 2)];
      out.push({
        x: catmull(p0.x, p1.x, p2.x, p3.x, u),
        y: catmull(p0.y, p1.y, p2.y, p3.y, u),
        z: catmull(p0.z, p1.z, p2.z, p3.z, u)
      });
    }
    return out;
  }

  function computePose() {
    const cv = state.crankAngle % TWO_PI;
    const wv = state.wheelAngle % TWO_PI;
    const sway = Math.sin(cv);
    const bob = Math.sin(cv * 2);
    const right = pedalPoint(cv, -SPEC.qFactor);
    const left = pedalPoint(cv + Math.PI, SPEC.qFactor);
    const hipR = { x: GEO.hip.x, y: GEO.hip.y, z: -SPEC.qFactor };
    const hipL = { x: GEO.hip.x, y: GEO.hip.y, z: SPEC.qFactor };
    const ankleR = solveIK(hipR, right, SPEC.upperLeg, SPEC.lowerLeg);
    const ankleL = solveIK(hipL, left, SPEC.upperLeg, SPEC.lowerLeg);
    const head = { x: 0.86 + sway * 0.01, y: 1.045 + bob * 0.008, z: 0 };
    const ctrls = [
      { x: 0.60, y: 0.77, z: 0 },
      { x: 0.50 + sway * 0.016, y: 0.90, z: 0 },
      { x: 0.56 + sway * 0.01, y: 1.02, z: 0 },
      { x: head.x - 0.16, y: head.y + 0.055, z: 0 },
      { x: head.x - 0.05, y: head.y + 0.012, z: 0 }
    ];
    const neck = sampleNeck(ctrls, 12);
    const nod = sway * 0.05;
    const pouchSway = bob * 0.18;
    const wingAng = sway * 0.05;
    return {
      cv, wv, sway, bob, right, left, hipR, hipL, ankleR, ankleL,
      head, neck, nod, pouchSway,
      wingL: poseWing(wingLSpec, wingAng),
      wingR: poseWing(wingRSpec, -wingAng),
      toesL: footToes(left),
      toesR: footToes(right),
      pedalAngleL: 0,
      pedalAngleR: 0,
      footDistL: 0,
      footDistR: 0
    };
  }

  function ringPoints(cx, cy, cz, r, spin, n) {
    const pts = [];
    for (let i = 0; i <= n; i++) {
      const a = spin + (i / n) * TWO_PI;
      pts.push({ x: cx + Math.cos(a) * r, y: cy + Math.sin(a) * r, z: cz });
    }
    return pts;
  }

  function externalTangents(x1, y1, r1, x2, y2, r2) {
    const dx = x2 - x1, dy = y2 - y1;
    const dist = Math.hypot(dx, dy);
    const base = Math.atan2(dy, dx);
    const alpha = Math.acos(clamp((r1 - r2) / dist, -1, 1));
    return [base + alpha, base - alpha].map((ang) => ({
      ang,
      p1: { x: x1 + Math.cos(ang) * r1, y: y1 + Math.sin(ang) * r1 },
      p2: { x: x2 + Math.cos(ang) * r2, y: y2 + Math.sin(ang) * r2 }
    }));
  }

  function sampleArc(cx, cy, cz, r, a0, a1, pts) {
    let sweep = (a0 - a1) % TWO_PI;
    if (sweep < 0) sweep += TWO_PI;
    if (sweep < 1e-4) sweep = TWO_PI;
    const steps = Math.max(8, Math.ceil(sweep / (Math.PI / 16)));
    for (let i = 0; i <= steps; i++) {
      const a = a0 - sweep * (i / steps);
      pts.push({ x: cx + Math.cos(a) * r, y: cy + Math.sin(a) * r, z: cz });
    }
  }

  function buildChain() {
    const z = GEO.chainZ;
    const tangents = externalTangents(GEO.bb.x, GEO.bb.y, GEO.chainringR, GEO.rear.x, GEO.rear.y, GEO.cogR);
    tangents.sort((a, b) => (a.p1.y + a.p2.y) - (b.p1.y + b.p2.y));
    const lower = tangents[0];
    const upper = tangents[1];
    const pts = [];
    pts.push({ x: upper.p2.x, y: upper.p2.y, z });
    pts.push({ x: upper.p1.x, y: upper.p1.y, z });
    sampleArc(GEO.bb.x, GEO.bb.y, z, GEO.chainringR, upper.ang, lower.ang, pts);
    pts.push({ x: lower.p2.x, y: lower.p2.y, z });
    sampleArc(GEO.rear.x, GEO.rear.y, z, GEO.cogR, lower.ang, upper.ang, pts);
    const cum = [0];
    for (let i = 1; i < pts.length; i++) {
      cum.push(cum[i - 1] + Math.hypot(pts[i].x - pts[i - 1].x, pts[i].y - pts[i - 1].y, pts[i].z - pts[i - 1].z));
    }
    return { pts, cum, length: cum[cum.length - 1] };
  }

  function pointOnChain(dist) {
    const L = chainSamples.length;
    let d = dist % L;
    if (d < 0) d += L;
    const cum = chainSamples.cum;
    const pts = chainSamples.pts;
    let i = 1;
    while (i < cum.length - 1 && cum[i] < d) i++;
    const span = (cum[i] - cum[i - 1]) || 1;
    const u = (d - cum[i - 1]) / span;
    const a = pts[i - 1], b = pts[Math.min(pts.length - 1, i)];
    return {
      x: a.x + (b.x - a.x) * u,
      y: a.y + (b.y - a.y) * u,
      z: a.z + (b.z - a.z) * u,
      tx: b.x - a.x,
      ty: b.y - a.y,
      tz: b.z - a.z
    };
  }

  function localBill(head, nod, lx, ly, lz) {
    const c = Math.cos(nod), s = Math.sin(nod);

      x: head.x + lx * c - ly * s,

      z: head.z + (lz || 0)
    };
  }

  function featherQuad(base, tip, width) {
    const dx = tip.x - base.x, dy = tip.y - base.y;
    let sx = dy, sy = -dx;
    let L = Math.hypot(sx, sy);
    if (L < 1e-5) { sx = 0; sy = 1; L = 1; }
    sx /= L; sy /= L;
    const hw = width * 0.5;
    const tw = width * 0.22;
    return [
      { x: base.x + sx * hw, y: base.y + sy * hw, z: base.z },
      { x: tip.x + sx * tw, y: tip.y + sy * tw, z: tip.z },
      { x: tip.x - sx * tw, y: tip.y - sy * tw, z: tip.z },
      { x: base.x - sx * hw, y: base.y - sy * hw, z: base.z }
    ];
  }

  function mat(color, rough, metal, extra) {
    return new THREE.MeshStandardMaterial(Object.assign({

  function placeTube(mesh, a, b) {
    tmpV.set(b.x - a.x, b.y - a.y, b.z - a.z);
    const len = tmpV.length();
    if (len < 1e-5) { mesh.visible = false; return; }
    mesh.visible = true;
    tmpV.multiplyScalar(1 / len);
    mesh.position.set((a.x + b.x) / 2, (a.y + b.y) / 2, (a.z + b.z) / 2);
    mesh.quaternion.setFromUnitVectors(yAxis, tmpV);
    mesh.scale.set(1, len, 1);
  }

    function buildWingMesh(spec, sign) {
      const g = new THREE.Group();
      g.position.set(spec.shoulder.x, spec.shoulder.y, spec.shoulder.z);
      const boneMat = mat(0x3a312b, 0.7, 0);
      const layers = { covert: 0x6e5c4a, secondary: 0x4e4034, primary: 0x2c2824 };
      const unit = new THREE.BoxGeometry(1, 1, 1);
      unit.translate(0.5, 0, 0);
      function addBone(a, b, radius) {
        const mesh = new THREE.Mesh(new THREE.CylinderGeometry(radius, radius, 1, 6), boneMat);
        const la = { x: a.x - spec.shoulder.x, y: a.y - spec.shoulder.y, z: a.z - spec.shoulder.z };
        const lb = { x: b.x - spec.shoulder.x, y: b.y - spec.shoulder.y, z: b.z - spec.shoulder.z };
        placeTube(mesh, la, lb);
        g.add(mesh);
      }

      addBone(spec.elbow, spec.wrist, 0.01);

      wrist.position.set(spec.wrist.x - spec.shoulder.x, spec.wrist.y - spec.shoulder.y, spec.wrist.z - spec.shoulder.z);
      g.add(wrist);

        const mesh = new THREE.Mesh(unit, mat(layers[f.layer], 0.82, 0));
        const dx = f.tip.x - f.base.x, dy = f.tip.y - f.base.y, dz = f.tip.z - f.base.z;
        const len = Math.hypot(dx, dy, dz) || 0.001;
        const shaft = new THREE.Vector3(dx / len, dy / len, dz / len);
        let wx = dy, wy = -dx, wz = 0;
        let wl = Math.hypot(wx, wy);
        if (wl < 1e-4) { wx = 0; wy = 1; wl = 1; }
        const wide = new THREE.Vector3(wx / wl, wy / wl, wz);
        wide.addScaledVector(shaft, -wide.dot(shaft));
        if (wide.lengthSq() < 1e-8) wide.set(0, 1, 0);
        wide.normalize();
        const thick = new THREE.Vector3().crossVectors(shaft, wide).normalize();
        const basis = new THREE.Matrix4().makeBasis(shaft, wide, thick);
        mesh.quaternion.setFromRotationMatrix(basis);
        mesh.scale.set(len, Math.max(f.w, 0.04), 0.012);
        mesh.position.set(f.base.x - spec.shoulder.x, f.base.y - spec.shoulder.y, f.base.z - spec.shoulder.z);
        g.add(mesh);
      }
      g.userData.sign = sign;
      return g;
    }

  let audio = null;
  function makeNoiseBuffer(ctx, seconds) {
    const n = Math.floor(ctx.sampleRate * seconds);
    const buffer = ctx.createBuffer(1, n, ctx.sampleRate);
    const data = buffer.getChannelData(0);
    for (let i = 0; i < n; i++) data[i] = Math.random() * 2 - 1;
    return buffer;
  }
  function makeClickBuffer(ctx) {
    const dur = 0.04;
    const n = Math.floor(ctx.sampleRate * dur);
    const buffer = ctx.createBuffer(1, n, ctx.sampleRate);
    const data = buffer.getChannelData(0);
    for (let i = 0; i < n; i++) {
      const t = i / ctx.sampleRate;
      const env = Math.exp(-t * 120);
      data[i] = ((Math.random() * 2 - 1) * 0.65 + Math.sin(TWO_PI * 1600 * t) * 0.35) * env;
    }
    return buffer;
  }
  function ensureAudio() {
    if (audio) {
      if (audio.ctx.state === 'suspended') audio.ctx.resume();
      return;
    }
    const ctx = new (window.AudioContext || window.webkitAudioContext)();
    const master = ctx.createGain();
    master.gain.value = state.muted ? 0 : 0.85;
    master.connect(ctx.destination);
    const clickBus = ctx.createGain();
    clickBus.gain.value = 0.2;
    const clickHP = ctx.createBiquadFilter();
    clickHP.type = 'highpass';
    clickHP.frequency.value = 700;
    clickBus.connect(clickHP);
    clickHP.connect(master);
    const windSrc = ctx.createBufferSource();
    windSrc.buffer = makeNoiseBuffer(ctx, 2);
    windSrc.loop = true;
    const lp = ctx.createBiquadFilter();
    lp.type = 'lowpass';
    lp.frequency.value = 400;
    const windGain = ctx.createGain();
    windGain.gain.value = 0;
    const hiss = ctx.createBiquadFilter();
    hiss.type = 'highpass';
    hiss.frequency.value = 1600;
    const hissGain = ctx.createGain();
    hissGain.gain.value = 0;
    windSrc.connect(lp);
    lp.connect(windGain);
    windGain.connect(master);
    windSrc.connect(hiss);
    hiss.connect(hissGain);
    hissGain.connect(master);
    windSrc.start();
    audio = {
      ctx, master, clickBus, lp, windGain, hissGain,
      clickBuf: makeClickBuffer(ctx),
      tooth: 0,
      callIn: 5 + Math.random() * 4
    };
    hint.hidden = true;
  }
  function playClick() {
    const src = audio.ctx.createBufferSource();
    src.buffer = audio.clickBuf;
    src.playbackRate.value = 0.92 + Math.random() * 0.16;
    src.connect(audio.clickBus);
    src.start();
  }
  function playCall() {
    const ctx = audio.ctx;
    const t = ctx.currentTime;
    const g = ctx.createGain();
    g.gain.setValueAtTime(0.0001, t);
    g.gain.exponentialRampToValueAtTime(0.12, t + 0.04);
    g.gain.exponentialRampToValueAtTime(0.0001, t + 0.48);
    g.connect(audio.master);
    const o1 = ctx.createOscillator();
    const o2 = ctx.createOscillator();
    o1.type = 'sine';
    o2.type = 'triangle';
    o1.frequency.setValueAtTime(170, t);
    o1.frequency.exponentialRampToValueAtTime(78, t + 0.42);
    o2.frequency.setValueAtTime(340, t);
    o2.frequency.exponentialRampToValueAtTime(140, t + 0.42);
    o1.connect(g);
    o2.connect(g);
    o1.start(t);
    o2.start(t);
    o1.stop(t + 0.5);
    o2.stop(t + 0.5);
  }
  function updateAudio(dt) {
    if (!audio) return;
    const now = audio.ctx.currentTime;
    const s = state.speed / SPEC.maxSpeed;
    const moving = state.speed > 0.04 && !state.muted;
    audio.windGain.gain.setTargetAtTime(moving ? 0.035 + s * 0.15 : 0, now, 0.06);
    audio.hissGain.gain.setTargetAtTime(moving ? s * s * 0.07 : 0, now, 0.06);
    audio.lp.frequency.setTargetAtTime(240 + state.speed * 110, now, 0.08);
    if (state.speed > 0 && !state.muted) {
      const toothRate = (state.speed / SPEC.wheelRadius) * (SPEC.cogTeeth / TWO_PI);
      audio.tooth += toothRate * dt;
      let guard = 0;
      while (audio.tooth >= 1 && guard < 6) {
        audio.tooth -= 1;
        playClick();
        guard++;
      }
      audio.callIn -= dt;
      if (audio.callIn <= 0 && state.speed > 0.3) {
        playCall();
        audio.callIn = 8 + Math.random() * 10;
      }
    }
  }

    // ω_crank / ω_wheel = cogTeeth / chainringTeeth
    const omegaC = omegaW * (SPEC.cogTeeth / SPEC.chainringTeeth);
    state.wheelAngle += omegaW * dt;
    state.crankAngle += omegaC * dt;
    state.distance += state.speed * dt;
    const pose = computePose();
    const drawBoth = now < bothUntil;
    if (drawBoth || state.mode === '2.5d') updateSVG(pose);
    if (drawBoth || state.mode === '3d') {
      update3D(pose);
      renderer.render(scene, camera);
    }
    updateAudio(dt);
    frames++;
    if (now - fpsStamp >= 500) {
      const fps = frames / ((now - fpsStamp) / 1000);
      fpsEl.textContent = fps.toFixed(0) + ' FPS';
      frames = 0;
      fpsStamp = now;
      const wRpm = omegaW * 60 / TWO_PI;
      const cRpm = omegaC * 60 / TWO_PI;
      readout.textContent = '踏频 ' + cRpm.toFixed(0) + ' rpm · 轮 ' + wRpm.toFixed(0) + ' rpm';
    }
    requestAnimationFrame(frame);
  }
  requestAnimationFrame(frame);
})();
