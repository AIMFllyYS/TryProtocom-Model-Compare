建模与物理实现思路：
本作采用统一的物理状态机驱动2.5D(SVG)与3D(Three.js)双引擎。通过时间积分更新全局偏移量，并按严格的传动比(16/44)与车轮半径(0.35m)计算后轮与曲柄角度。鸟腿运动链使用两段式逆运动学(IK)解析计算，确保脚掌精准贴合水平踏板。生物特征上，不仅运用曲线模拟S型颈部和可动态摆动的喉囊，且翅膀分层渲染覆羽与初级飞羽。音效则用Web Audio API合成依赖转速的链条咔哒声与风噪。切换模式时共享该状态机，利用CSS opacity实现平滑过渡。

function initAudio() {
  audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  const noiseBuf = audioCtx.createBuffer(1, audioCtx.sampleRate*2, audioCtx.sampleRate);
  const out = noiseBuf.getChannelData(0);
  for(let i=0; i<out.length; i++) out[i] = Math.random()*2-1;
  const windSrc = audioCtx.createBufferSource(); windSrc.buffer = noiseBuf; windSrc.loop = true;
  const windFilter = audioCtx.createBiquadFilter(); windFilter.type = 'lowpass'; windFilter.frequency.value = 400;
  windGain = audioCtx.createGain(); windGain.gain.value = 0;
  windSrc.connect(windFilter); windFilter.connect(windGain); windGain.connect(audioCtx.destination);
  windSrc.start();
}

function updateAudio(dt) {
  if(!audioCtx || !audioEnabled) return;
  windGain.gain.setTargetAtTime(Math.min(state.speed/20, 0.3), audioCtx.currentTime, 0.1);
  const clickAngle = (Math.PI*2)/T_CHAINRING;
  if(state.crankAngle - lastClickAngle >= clickAngle) { lastClickAngle += clickAngle; playClick(); }
  if(state.speed > 0 && Math.random() < 0.001*state.speed) playPelicanSound();
}

function playClick() {
  if(!audioCtx || !audioEnabled) return;
  const osc = audioCtx.createOscillator(), gain = audioCtx.createGain();
  osc.type = 'square'; osc.frequency.setValueAtTime(800+Math.random()*200, audioCtx.currentTime); osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime+0.05);
  gain.gain.setValueAtTime(0.05, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime+0.05);
  osc.connect(gain); gain.connect(audioCtx.destination); osc.start(); osc.stop(audioCtx.currentTime+0.05);
}

function playPelicanSound() {
  if(!audioCtx || !audioEnabled) return;
  const osc = audioCtx.createOscillator(), gain = audioCtx.createGain();
  osc.type = 'sawtooth'; osc.frequency.setValueAtTime(150, audioCtx.currentTime); osc.frequency.linearRampToValueAtTime(100, audioCtx.currentTime+0.2);
  gain.gain.setValueAtTime(0.1, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime+0.2);
  osc.connect(gain); gain.connect(audioCtx.destination); osc.start(); osc.stop(audioCtx.currentTime+0.2);
}

function solveIK(x1, y1, x2, y2, l1, l2, flip) {
  let dx=x2-x1, dy=y2-y1, d=Math.sqrt(dx*dx+dy*dy);
  d = Math.max(Math.abs(l1-l2)+0.001, Math.min(d, l1+l2-0.001));
  let a = (l1*l1-l2*l2+d*d)/(2*d), h = Math.sqrt(Math.max(0, l1*l1-a*a));
  let cx = x1+a*dx/d, cy = y1+a*dy/d, sign = flip?-1:1;
  return {x: cx+sign*h*dy/d, y: cy-sign*h*dx/d};
}

function updateLeg3D(leg1, leg2, hip, knee, pedal, zOffset) {

}
